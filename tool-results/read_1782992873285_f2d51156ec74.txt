    14→// ===== Document type mapping =====
    15→const DOC_TYPE_MAP: Record<string, string> = {
    16→  "/prd": "prd",
    17→  "/trd": "trd",
    18→  "/flow": "flow",
    19→  "/ux": "ux",
    20→  "/schema": "schema",
    21→  "/plan": "plan",
    22→};
    23→
    24→// Document keyword signatures for detecting update-type messages
    25→const DOC_SIGNATURES = [
    26→  { label: "prd", keywords: ["Product Requirements Document", "Executive Summary", "Feature Requirements", "User Stories", "Target Audience", "Product Vision", "Non-Functional Requirements"] },
    27→  { label: "trd", keywords: ["Technical Requirements Document", "Architecture Overview", "Technology Stack", "API Specification", "Security Requirements", "Performance Requirements"] },
    28→  { label: "flow", keywords: ["Application Flow Document", "User Journey", "Screen Flow", "Navigation Architecture", "State Management", "Interaction Patterns"] },
    29→  { label: "ux", keywords: ["UI/UX Design Brief", "Design Principles", "Design System", "Color Palette", "Typography", "Component Guidelines", "Accessibility"] },
    30→  { label: "schema", keywords: ["Backend Schema Document", "Entity Relationship", "Schema Definitions", "Enum Types", "Data Integrity Rules", "Seed Data"] },
    31→  { label: "plan", keywords: ["Implementation Plan", "Phase Breakdown", "Task Breakdown", "Sprint Planning", "Resource Requirements", "Risk Register", "Quality Gates"] },
    32→];
    33→
    34→function detectDocTypeFromContent(content: string): string | null {
    35→  if (content.length < 1500) return null;
    36→  const lower = content.toLowerCase();
    37→  let bestMatch = "";
    38→  let bestCount = 0;
    39→  for (const sig of DOC_SIGNATURES) {
    40→    const count = sig.keywords.filter(kw => lower.includes(kw.toLowerCase())).length;
    41→    if (count >= 3 && count > bestCount) {
    42→      bestCount = count;
    43→      bestMatch = sig.label;
    44→    }
    45→  }
    46→  return bestMatch || null;
    47→}

   173→// GET /api/ai/chat — Paginated chat history
   174→export async function GET(request: NextRequest) {
   175→  try {
   176→    const user = await getAuthUser(request);
   177→    if (!user) {
   178→      return NextResponse.json({ success: false, error: "Authentication required" }, { status: 401 });
   179→    }
   180→
   181→    const { searchParams } = new URL(request.url);
   182→    const projectId = searchParams.get("projectId");
   183→    if (!projectId) {
   184→      return NextResponse.json({ success: false, error: "projectId is required" }, { status: 400 });
   185→    }
   186→
   187→    // Check project access
   188→    const canAccess = await hasProjectAccess(user.id, user.role, projectId);
   189→    if (!canAccess) {
   190→      return NextResponse.json({ success: false, error: "You don't have access to this project" }, { status: 403 });
   191→    }
   192→
   193→    const page = Math.max(1, parseInt(searchParams.get("page") || "1"));
   194→    const limit = Math.min(100, Math.max(1, parseInt(searchParams.get("limit") || "50")));
   195→
   196→    const client = getTursoClient();
   197→    const offset = (page - 1) * limit;
   198→
   199→    let messagesResult;
   200→
   201→    if (user.role === "SUPERADMIN") {
   202→      // SUPERADMIN sees all messages — newest first (client will reverse for display)
   203→      messagesResult = await client.execute({
   204→        sql: `SELECT m.*, u.name as userName
   205→              FROM "AiChat" m
   206→              LEFT JOIN "User" u ON m."userId" = u.id
   207→              WHERE m."projectId" = ?
   208→              ORDER BY m."timestamp" DESC
   209→              LIMIT ? OFFSET ?`,
   210→        args: [projectId, limit, offset],
   211→      });
   212→    } else {
   213→      // Others see only own messages + assistant messages — newest first
   214→      messagesResult = await client.execute({
   215→        sql: `SELECT m.*, u.name as userName
   216→              FROM "AiChat" m
   217→              LEFT JOIN "User" u ON m."userId" = u.id
   218→              WHERE m."projectId" = ? AND (m."userId" = ? OR m."role" = 'assistant')
   219→              ORDER BY m."timestamp" DESC
   220→              LIMIT ? OFFSET ?`,
   221→        args: [projectId, user.id, limit, offset],
   222→      });
   223→    }
   224→
   225→    const countResult = await client.execute({
   226→      sql: `SELECT COUNT(*) as total FROM "AiChat" WHERE "projectId" = ?`,
   227→      args: [projectId],
   228→    });
   229→    const total = Number(countResult.rows[0].total);
   230→
   231→    const messages = messagesResult.rows.map((row) => ({
   232→      id: row.id,
   233→      userId: row.userId,
   234→      projectId: row.projectId,
   235→      role: row.role,
   236→      content: row.content,
   237→      timestamp: row.timestamp,
   238→      userName: row.userName,
   239→    }));
   240→
   241→    return NextResponse.json({
   242→      success: true,
   243→      data: {
   244→        messages,
   245→        pagination: {
   246→          page,
   247→          limit,
   248→          total,
   249→          totalPages: Math.ceil(total / limit),
   250→        },
   251→      },
   252→    });
   253→  } catch (error) {
   254→    console.error("[GET /api/ai/chat] Error:", error);
   255→    return NextResponse.json({ success: false, error: "Internal server error" }, { status: 500 });
   256→  }
   257→}

   263→/**
   264→ * Parse tasks from a markdown document. Uses doc-type-specific patterns
   265→ * to extract actionable items from PRD, TRD, Flow, UX, Schema, and Plan docs.
   266→ */
   267→async function autoGenerateTodosFromDoc(
   268→  dbClient: ReturnType<typeof getTursoClient>,
   269→  projectId: string,
   270→  docContent: string,
   271→  docType: string,
   272→  userId: string,
   273→): Promise<number> {
   274→  // Get current max sortOrder for the project
   275→  const maxOrder = await dbClient.execute({
   276→    sql: `SELECT COALESCE(MAX("sortOrder"), -1) as maxOrder FROM "ProjectTodo" WHERE "projectId" = ?`,
   277→    args: [projectId],
   278→  });
   279→  let sortOrder = Number(maxOrder.rows[0].maxOrder) + 1;
   280→
   281→  const tasks: { title: string; description: string; priority: string }[] = [];
   282→  const lines = docContent.split('\n');
   283→  let currentSection = '';
   284→  let currentPhase = '';
   285→
   286→  // Doc-type-specific section prefixes for richer descriptions
   287→  const DOC_LABELS: Record<string, string> = {
   288→    prd: "Product Requirements",
   289→    trd: "Technical Requirements",
   290→    flow: "Application Flow",
   291→    ux: "UI/UX Design",
   292→    schema: "Database Schema",
   293→    plan: "Implementation Plan",
   294→  };
   295→  const docLabel = DOC_LABELS[docType] || docType.toUpperCase();
   296→
   297→  for (let i = 0; i < lines.length; i++) {
   298→    const line = lines[i].trim();
   299→
   300→    // Track current section/phase for context
   301→    if (line.match(/^#{1,3}\s/)) {
   302→      currentSection = line.replace(/^#+\s*/, '').replace(/\*\*/g, '').trim();
   303→      if (/phase/i.test(currentSection)) {
   304→        currentPhase = currentSection;
   305→      }
   306→      continue;
   307→    }
   308→
   309→    // Match markdown list items: "- [ ] task", "- task", "* task", numbered lists
   310→    const taskMatch = line.match(/^[-*]\s+\[?\s*\]?\s+(.+)/) || line.match(/^\d+\.\s+(.+)/);
   311→    if (taskMatch) {
   312→      let taskText = taskMatch[1].trim();
   313→      // Skip if it's a sub-heading or too short
   314→      if (taskText.length < 5) continue;
   315→      // Skip if it's just a bold title (like a section header in a list)
   316→      if (taskText.match(/^\*\*[^*]+\*\*$/) && taskText.length < 80) continue;
   317→
   318→      // Determine priority from context
   319→      let priority = 'MEDIUM';
   320→      if (/\b(critical|urgent|high.?priority|must.?have|blocker|p0)\b/i.test(taskText) ||
   321→          /\b(critical|urgent|high.?priority|must.?have|blocker|p0)\b/i.test(currentSection)) {
   322→        priority = 'HIGH';
   323→      } else if (/\b(low.?priority|nice.?to.?have|optional|future|p[2-3])\b/i.test(taskText) ||
   324→                 /\b(low.?priority|nice.?to.?have|optional|future|p[2-3])\b/i.test(currentSection)) {
   325→        priority = 'LOW';
   326→      }
   327→
   328→      // Build description from context
   329→      let description = `Source: ${docLabel}`;
   330→      if (currentPhase) {
   331→        description += `\nPhase: ${currentPhase}`;
   332→      }
   333→      if (currentSection && currentSection !== currentPhase) {
   334→        description += `\nSection: ${currentSection}`;
   335→      }
   336→
   337→      // Try to grab a sub-description line (next non-empty, non-list line)
   338→      for (let j = i + 1; j < Math.min(i + 3, lines.length); j++) {
   339→        const nextLine = lines[j].trim();
   340→        if (nextLine && !nextLine.match(/^[-*#]/) && !nextLine.match(/^\d+\./)) {
   341→          description += `\n${nextLine}`;
   342→          break;
   343→        }
   344→      }
   345→
   346→      tasks.push({
   347→        title: taskText,
   348→        description: description.trim(),
   349→        priority,
   350→      });
   351→    }
   352→  }
   353→
   354→  if (tasks.length === 0) return 0;
   355→
   356→  // Check for existing todos to avoid duplicates
   357→  const existingTodos = await dbClient.execute({
   358→    sql: `SELECT title FROM "ProjectTodo" WHERE "projectId" = ?`,
   359→    args: [projectId],
   360→  });
   361→  const existingTitles = new Set(existingTodos.rows.map(r => (r.title as string).toLowerCase()));
   362→
   363→  const newTasks = tasks.filter(t => !existingTitles.has(t.title.toLowerCase()));
   364→  if (newTasks.length === 0) {
   365→    console.log(`[autoGenerateTodos] All ${tasks.length} tasks from ${docType} already exist, skipping.`);
   366→    return 0;
   367→  }
   368→
   369→  // Batch insert all tasks
   370→  for (const task of newTasks) {
   371→    const id = crypto.randomUUID();
   372→    await dbClient.execute({
   373→      sql: `INSERT INTO "ProjectTodo" (id, "projectId", title, description, status, priority, "sortOrder", "createdBy", "createdAt", "updatedAt")
   374→            VALUES (?, ?, ?, ?, 'PENDING', ?, ?, ?, datetime('now'), datetime('now'))`,
   375→      args: [id, projectId, task.title, task.description || null, task.priority, sortOrder++, userId],
   376→    });
   377→  }
   378→
   379→  console.log(`[autoGenerateTodos] Created ${newTasks.length} tasks from ${docType} (out of ${tasks.length} parsed, ${existingTitles.size} existing)`);
   380→  return newTasks.length;
   381→}

   383→// POST /api/ai/chat — Send message and get AI response (with agentic tool-calling loop)
   384→export async function POST(request: NextRequest) {
   385→  try {
   386→    const user = await getAuthUser(request);
   387→    if (!user) {
   388→      return NextResponse.json({ success: false, error: "Authentication required" }, { status: 401 });
   389→    }
   390→
   391→    const rawBody = await request.json();
   392→    const { projectId, content, files, fileData, fileName, fileType } = rawBody as { projectId: string; content: string; files?: unknown[]; fileData?: string; fileName?: string; fileType?: string };
   393→
   394→    if (!projectId || !content) {
   395→      return NextResponse.json({ success: false, error: "projectId and content are required" }, { status: 400 });
   396→    }
   397→
   398→    const attachedImages: { data: string; type: string; name: string }[] = [];
   399→    if (files && Array.isArray(files) && files.length > 0) {
   400→      // New format: array of { data, type, name }
   401→      for (const f of files) {
   402→        if (f.data && f.type && f.type.startsWith("image/")) {
   403→          attachedImages.push({ data: f.data, type: f.type, name: f.name || "image" });
   404→        }
   405→      }
   406→    } else if (fileData && fileType && fileType.startsWith("image/")) {
   407→      // Legacy single-file format
   408→      attachedImages.push({ data: fileData, type: fileType, name: fileName || "image" });
   409→    }
   410→    const hasImages = attachedImages.length > 0;
   411→
   412→    // Check project access
   413→    const canAccess = await hasProjectAccess(user.id, user.role, projectId);
   414→    if (!canAccess) {
   415→      return NextResponse.json({ success: false, error: "You don't have access to this project" }, { status: 403 });
   416→    }
   417→
   418→    const client = getTursoClient();
   419→    const ip = getClientIp(request);
   420→
   421→    // Auto-migrate AI tables if they don't exist
   422→    await ensureAiTables(client);
   423→
   424→    // Save user message
   425→    const userMsgId = crypto.randomUUID();
   426→    await client.execute({
   427→      sql: `INSERT INTO "AiChat" (id, "userId", "projectId", "role", "content", "timestamp")
   428→            VALUES (?, ?, ?, 'user', ?, datetime('now'))`,
   429→      args: [userMsgId, user.id, projectId, content],
   430→    });
   431→
   432→    // Detect command
   433→    const trimmed = content.trim();
   434→    const command = trimmed.startsWith("/") ? trimmed.split(/[\s\n]/)[0] : undefined;
   435→
   436→    // Detect command type (needed before loading history to determine history limit)
   437→    const isDocCommand = !!command && ["/docs", "/prd", "/trd", "/flow", "/ux", "/schema", "/plan", "/init"].includes(command);
   438→    const isSkillCommand = !!command && ["/standup", "/risks", "/summarize", "/code-review", "/knowledge"].includes(command);
   439→
   440→    // Load recent chat messages for context
   441→    // For doc commands: only last 6 messages (saves tokens, avoids bloated context)
   442→    // For skill commands & regular chat: last 20 messages for context-aware analysis
   443→    const historyLimit = isDocCommand ? 6 : 20;
   444→    const historyResult = await client.execute({
   445→      sql: `SELECT role, content FROM "AiChat"
   446→            WHERE "projectId" = ?
   447→            ORDER BY "timestamp" DESC
   448→            LIMIT ?`,
   449→      args: [projectId, historyLimit],
   450→    });
   451→    const chatHistory = historyResult.rows.reverse().map((row) => ({
   452→      role: row.role,
   453→      content: row.content,
   454→    }));
   455→

   457→    let userName = user.name;
   458→    try {
   459→      const userResult = await client.execute({
   460→        sql: `SELECT name FROM "User" WHERE id = ?`,
   461→        args: [user.id],
   462→      });
   463→      if (userResult.rows.length > 0) {
   464→        userName = userResult.rows[0].name as string;
   465→      }
   466→    } catch {
   467→      // Use token name as fallback
   468→    }
   469→
   470→    // Load project info (with status, priority, deadline)
   471→    const projectResult = await client.execute({
   472→      sql: `SELECT name, description, "clientName", status, priority, deadline FROM "Project" WHERE id = ?`,
   473→      args: [projectId],
   474→    });
   475→    const project = projectResult.rows[0];
   476→
   477→    // Load team member count for this project
   478→    let teamCount: number | undefined;
   479→    try {
   480→      const teamResult = await client.execute({
   481→        sql: `SELECT COUNT(*) as count FROM "ProjectMember" WHERE "projectId" = ? AND "removedAt" IS NULL`,
   482→        args: [projectId],
   483→      });
   484→      teamCount = Number(teamResult.rows[0].count);
   485→    } catch {
   486→      // Non-critical
   487→    }
   488→
   489→    // Load protocol steps for /docs command

   611→        modelRouteReason = route.reason;

   616→      console.error("[AI Auto-Route] Routing check failed, using selected model:", routeErr);

   619→    // ===== Agentic Loop Setup =====

   624→      label: string;

   632→      userId: user.id,

   636→    };

   639→    const aiMessages: AiMessage[] = [
   640→      { role: "system", content: systemPrompt },
   641→    ];

   643→    for (const msg of chatHistory) {
   644→      aiMessages.push({
   645→        role: msg.role as "user" | "assistant",

   683→          aiText = result.content;
   684→        }
   685→      } else {
   686→        // ===== Agentic Tool-Calling Loop with Provider Fallback =====
   687→        // Helper: call AI with automatic fallback on 429/413/401 errors
   688→        const callWithFallback = async (
   689→          messages: AiMessage[],
   690→          model: string,
   691→          maxTok: number,

   696→          let result = await chatCompletion({
   697→            messages,
   698→            model,
   699→            maxTokens: maxTok,
   700→            tools,
   701→            tool_choice: toolChoice,
   702→          });

   707→              tools: !!tools,
   708→              vision: false,

   716→                messages,
   717→                model: fallbackModel,
   718→                maxTokens: maxTok,
   719→                tools,
   720→                tool_choice: toolChoice,
   721→              });

   723→                console.log(`[AI Fallback] Success with ${fallbackModel}!`);

   727→              const errStr = fbResult.error || "";

   754→            aiMessages,
   755→            activeModel,
   756→            maxTokens,
   757→            shouldSendTools ? availableTools as unknown as NonNullable<Parameters<typeof chatCompletion>[0]["tools"]> : undefined,

   770→              shouldSendTools = false;

   772→            }

   774→

" (auto-routed)" : ""} command=${command || "none"} has_tools=${!!result.tool_calls} content_len=${result.content?.length || 0} success=${result.success} error=${result.error || "none"}`);

   792→            finalContent = result.content;

   798→          aiMessages.push({
   799→            role: "assistant",

   808→            const toolIcon = getToolIcon(toolName);

   815→              icon: toolIcon,

   828→            aiMessages.push({
   829→              role: "tool",
   830→              tool_call_id: toolCall.id,
   831→              content: toolResult.result,
   832→              name: toolName,
   833→            });
   834→          }
   835→
   836→          // Loop continues — AI will see tool results and decide next action
   837→        }
   838→
   839→        // If we exhausted rounds without final content, generate a fallback
   840→        if (!finalContent && round >= MAX_TOOL_ROUNDS) {

   842→            toolExecutions.map((t) => `- ${t.icon} ${t.displayMessage}`).join("\n") +
   843→            "\n\nIs there anything else you'd like me to do?";

   847→        }
   848→
   849→        // Better fallback: retry without tools (with provider fallback) for commands that produced no content
   850→        if (!finalContent) {

   855→              const retryMessages: AiMessage[] = [

   866→                finalContent = retryResult.content;

   881→        aiText = typeof finalContent === "string" ? finalContent : JSON.stringify(finalContent);
   882→      }

   889→    // ===== Anti-Hallucination Filter =====

   893→      const fakeActionPatterns = [
   894→        { pattern: /(?:I[''](ve| have| will| am| just))?\s*(?:pushed|pushing|committed|deployed|deploying|created|creating|built|building|cloned|cloning|merged|merging)\s+(?:the\s+)?(?:code|changes|files|project|application|app|repo|repository|to\s+GitHub|to\s+the\s+repo)/gi,

   898→        { pattern: /(?:Build\s+passed|Deployment\s+successful|Successfully\s+deployed|CI\/CD\s+passed)/gi,
   899→          replacement: "[Note: This was not actually executed.]" },
   900→      ];

   902→      // Only apply filter if NO github/file tools were used
   903→      const usedGitTools = toolNames.has("github_pull") || toolNames.has("github_push_code") || toolNames.has("create_or_update_file");
   904→      if (!usedGitTools) {
   905→        for (const { pattern, replacement } of fakeActionPatterns) {
   906→          aiText = aiText.replace(pattern, replacement);

   908→        // Remove references to other projects/applications
   909→        const otherProjectPattern = /(?:(?:the|a|an)\s+)?(?:Share\s+Sathi|[A-Z][a-z]+(?:\s+[A-Z][a-z]+)+\s+application)(?!\s+(?:project|in\s+KarmaBoard))/g;
   910→        aiText = aiText.replace(otherProjectPattern, "[irrelevant project reference removed]");
   911→      }
   912→    }

   914→    // Save AI response
   915→    const aiMsgId = crypto.randomUUID();

   917→      sql: `INSERT INTO "AiChat" (id, "userId", "projectId", "role", "content", "timestamp")

   920→    });
   921→
   922→    // ===== Auto-save document to ProjectDocument =====

   925→      try {
   926→        let docType: string | undefined;
   927→
   928→        // Check if this is a direct doc command

   930→          docType = DOC_TYPE_MAP[command];
   931→        }
   932→        // Or if the AI response matches a document signature (update flow)
   933→        if (!docType) {
   934→          const detected = detectDocTypeFromContent(aiText);
   935→          docType = detected || undefined;
   936→        }
   937→
   938→        if (docType) {
   939→          // Extract title from markdown heading
   940→          const title = aiText.split("\n").find(l => l.trim().startsWith("#"))
   941→            ?.replace(/^#+\s*/, "").replace(/\*\*/g, "").trim().slice(0, 100)
   942→            || docType.toUpperCase() + " Document";

   947→            pdfBase64 = await generatePdfBase64(aiText);

   957→
   958→          if (existing.rows.length > 0) {

   963→              sql: `UPDATE "ProjectDocument" SET title = ?, content = ?, "pdfData" = ?, version = ?, "updatedAt" = datetime('now') WHERE id = ?`,
   964→              args: [title, aiText, pdfBase64, newVersion, existingId],

   966→            documentInfo = { id: existingId, docType, title, version: newVersion };

   968→          } else {
   969→            // Create new
   970→            const newDocId = crypto.randomUUID();

   978→        }
   979→      } catch (docErr) {

   981→      }
   982→    }
   983→
   984→    // ===== Auto-generate todos from pre-coding documents =====
   985→    const PRE_CODING_DOCS = ['prd', 'trd', 'flow', 'ux', 'schema', 'plan'];

   997→      }
   998→    }
   999→
  1000→    // Log activity
  1001→    const toolSummary = toolExecutions.length > 0
  1002→      ? ` (tools used: ${toolExecutions.map((t) => t.toolName).join(", ")})`
  1003→      : "";

  1005→      userId: user.id,
  1006→      action: "AI_CHAT_MESSAGE",
  1007→      details: `Sent AI chat message in project${project?.name ? ` "${project.name}"` : ""}${command ? ` (command: ${command})` : ""}${toolSummary}`,
  1008→      entity: "ai_chat",
  1009→      entityId: projectId,
  1010→      ipAddress: ip,
  1011→      tursoClient: client,
  1012→    });
  1013→
  1014→    // ===== Extract TRD theme colors (runs after doc save if TRD was generated) =====
  1015→    if ((command === "/trd" || (documentInfo && documentInfo.docType === "trd")) && aiText && aiText.length > 500) {
  1016→      try {
  1017→        const settingsKey = "PROJECT_THEME:" + projectId;
  1018→        const existingTheme = await client.execute({
  1019→          sql: `SELECT key FROM "Settings" WHERE key = ?`,
  1020→          args: [settingsKey],
  1021→        });
  1022→        if (existingTheme.rows.length === 0) {

  1025→            const themeColors = { colors: [...new Set(colorMatches)] };
  1026→            await client.execute({
  1027→              sql: `INSERT OR REPLACE INTO "Settings" (key, value, "updatedAt") VALUES (?, ?, datetime('now'))`,
  1028→              args: [settingsKey, JSON.stringify(themeColors)],
  1029→            });

  1031→          }
  1032→        }
  1033→      } catch (themeErr) {
  1034→        console.error("[Chat] Theme extraction error (non-fatal):", themeErr);
  1035→      }
  1036→    }
  1037→
  1038→    // ===== Auto-push document to GitHub (fire-and-forget, non-blocking) =====
  1039→    if (documentInfo && !aiError) {
  1040→      // Fire-and-forget: push to GitHub in background without blocking the response
  1041→      (async () => {
  1042→        try {
  1043→          const repoResult = await client.execute({
  1044→            sql: `SELECT value FROM "Settings" WHERE key = 'GITHUB_REPO_URL'`,
  1045→            args: [],
  1046→          });

  1048→            sql: `SELECT value FROM "Settings" WHERE key = 'GITHUB_PAT'`,
  1049→            args: [],
  1050→          });
  1051→
  1052→          if (repoResult.rows.length === 0 || patResult.rows.length === 0) {
  1053→            console.log("[Chat] No GitHub config, skipping auto-push for", documentInfo.docType);
  1054→            return;
  1055→          }
  1056→
  1057→          const repoUrl = repoResult.rows[0].value as string;
  1058→          let token: string;

  1060→            token = decrypt(patResult.rows[0].value as string);
  1061→          } catch {
  1062→            token = patResult.rows[0].value as string; // Legacy unencrypted
  1063→          }
  1064→
  1065→          const githubConfig = { repoUrl, token };
  1066→          const docLabel = documentInfo.docType.toUpperCase();
  1067→          const commitMsg = `docs: update ${docLabel} document (v${documentInfo.version})`;
  1068→
  1069→          // Push markdown file
  1070→          const mdPath = "docs/pre-coding/" + documentInfo.docType + ".md";
  1071→          await pushFile(githubConfig, mdPath, aiText, commitMsg);
  1072→          console.log("[Chat] GitHub push: markdown", mdPath);
  1073→

  1075→          if (documentInfo) {
  1076→            const pdfResult = await client.execute({
  1077→              sql: `SELECT "pdfData" FROM "ProjectDocument" WHERE id = ? AND "pdfData" IS NOT NULL AND "pdfData" != ''`,
  1078→              args: [documentInfo.id],
  1079→            });
  1080→            if (pdfResult.rows.length > 0 && pdfResult.rows[0].pdfData) {
  1081→              const pdfPath = "docs/pre-coding/" + documentInfo.docType + ".pdf";
  1082→              await pushBinaryFile(githubConfig, pdfPath, pdfResult.rows[0].pdfData as string, commitMsg);
  1083→              console.log("[Chat] GitHub push: PDF", pdfPath);
  1084→            }
  1085→          }
  1086→
  1087→          console.log("[Chat] GitHub auto-push complete for", documentInfo.docType, "v" + documentInfo.version);
  1088→        } catch (ghErr) {
  1089→          console.error("[Chat] GitHub auto-push failed (non-fatal):", ghErr);
  1090→        }
  1091→      })();
  1092→    }
  1093→
  1094→    // ===== z.ai Bridge: Check if /init should include bridge info =====
  1095→    // If command is /init and project has documents, check z.ai config and add bridge data
  1096→    let zaiBridge: {
  1097→      chatId: string;
  1098→      chatUrl: string;
  1099→      context: string;
  1100→      modelName: string;
  1101→      documentsFound: number;
  1102→      isNewChat: boolean;
  1103→      aiResponse?: string;
  1104→      apiError?: string;
  1105→      chatMessagesFound?: number;
  1106→      docsSource?: string;
  1107→    } | undefined;
  1108→
  1109→    if (command === "/init") {
  1110→      try {
  1111→        // Count project documents (no minimum required — send whatever exists)
  1112→        const docsCountResult = await client.execute({
  1113→          sql: `SELECT COUNT(*) as count FROM "ProjectDocument" WHERE "projectId" = ? AND "docType" IN ('prd', 'trd', 'flow', 'ux', 'schema', 'plan')`,
  1114→          args: [projectId],
  1115→        });
  1116→        const docsCount = Number(docsCountResult.rows[0]?.count || 0);
  1117→
  1118→        // Check z.ai API key (primary auth method)
  1119→        const zaiApiKeyResult = await client.execute({
  1120→          sql: `SELECT value FROM "Settings" WHERE key = 'ZAI_BRIDGE_API_KEY'`,
  1121→          args: [],
  1122→        });
  1123→
  1124→        let zaiBearerToken = "";
  1125→        if (zaiApiKeyResult.rows.length > 0 && zaiApiKeyResult.rows[0].value) {
  1126→          try { zaiBearerToken = decrypt(zaiApiKeyResult.rows[0].value as string); }
  1127→          catch { zaiBearerToken = zaiApiKeyResult.rows[0].value as string; }
  1128→        }
  1129→
  1130→        if (zaiBearerToken) {
  1131→          // Fetch all documents and build context
  1132→          const docsResult = await client.execute({
  1133→            sql: `SELECT "docType", title, content, version FROM "ProjectDocument" WHERE "projectId" = ? AND "docType" IN ('prd', 'trd', 'flow', 'ux', 'schema', 'plan') ORDER BY "docType"`,
  1134→            args: [projectId],
  1135→          });
  1136→
  1137→          const DOC_LABELS: Record<string, string> = {
  1138→            prd: "Product Requirements Document",
  1139→            trd: "Technical Requirements Document",
  1140→            flow: "Application Flow Document",
  1141→            ux: "UI/UX Design Brief",
  1142→            schema: "Backend Schema Document",
  1143→            plan: "Implementation Plan",
  1144→          };
  1145→
  1146→          // Check for existing chat mapping
  1147→          const chatKey = `ZAI_CHAT:${projectId}`;
  1148→          const existingChat = await client.execute({
  1149→            sql: `SELECT value FROM "Settings" WHERE key = ?`,
  1150→            args: [chatKey],
  1151→          });
  1152→
  1153→          let chatId: string;
  1154→          let isNewChat = false;
  1155→          if (existingChat.rows.length > 0 && existingChat.rows[0].value) {
  1156→            chatId = existingChat.rows[0].value as string;

  1168→            sql: `SELECT value FROM "Settings" WHERE key = 'ZAI_BRIDGE_BASE_URL'`,

  1177→            ? (zaiBaseUrlResult.rows[0].value as string)

  1266→        }

  1271→
  1272→    return NextResponse.json({
