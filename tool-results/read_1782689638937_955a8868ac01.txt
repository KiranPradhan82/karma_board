    83→    83→
    84→    84→export default function ProjectsPage() {
    85→    85→  const { data: session } = useSession();
    86→    86→  const router = useRouter();
    87→    87→  const user = session?.user as { role?: string; id?: string } | undefined;
    88→    88→  const isAdmin = user?.role === "ADMIN" || user?.role === "SUPERADMIN";
    89→    89→  const isSuperAdmin = user?.role === "SUPERADMIN";
    90→    90→
    91→    91→  const [projects, setProjects] = useState<Project[]>([]);
    92→    92→  const [loading, setLoading] = useState(true);
    93→    93→  const [search, setSearch] = useState("");
    94→    94→  const [statusFilter, setStatusFilter] = useState("");
    95→    95→  const [priorityFilter, setPriorityFilter] = useState("");
    96→    96→  const [sortBy, setSortBy] = useState("createdAt");
    97→    97→  const [page, setPage] = useState(1);
    98→    98→  const [totalPages, setTotalPages] = useState(1);
    99→    99→  const [total, setTotal] = useState(0);
   100→   100→
   101→   101→  // Create dialog
   102→   102→  const [createOpen, setCreateOpen] = useState(false);
   103→   103→  const [createLoading, setCreateLoading] = useState(false);
   104→   104→  const [createForm, setCreateForm] = useState({
   105→   105→    name: "",
   106→   106→    description: "",
   107→   107→    priority: "MEDIUM",
   108→   108→    clientName: "",
   109→   109→    clientId: "",
   110→   110→    color: "#6366f1",
   111→   111→    deadline: "",
   112→   112→  });
   113→   113→  const [clients, setClients] = useState<{ id: string; name: string; email: string; company: string | null }[]>([]);
   114→   114→  const [showNewClientForm, setShowNewClientForm] = useState(false);
   115→   115→  const [newClientForm, setNewClientForm] = useState({ name: "", email: "", company: "", address: "", phone: "" });
   116→   116→
   117→   117→  // Fetch clients for dropdown
   118→   118→  const fetchClients = useCallback(async () => {
   119→   119→    try {
   120→   120→      const res = await fetch('/api/clients?limit=100');
   121→   121→      const data = await res.json();
   122→   122→      if (data.success) {
   123→   123→        setClients(data.data.clients);
   124→   124→      }
   125→   125→    } catch {
   126→   126→      // Ignore errors on client fetch
   127→   127→    }
   128→   128→  }, []);
   129→   129→
   130→   130→  useEffect(() => {
   131→   131→    if (createOpen) fetchClients();
   132→   132→  }, [createOpen, fetchClients]);
   133→   133→
   134→   134→  const fetchProjects = useCallback(async () => {
   135→   135→    setLoading(true);
   136→   136→    try {
   137→   137→      const params = new URLSearchParams({
   138→   138→        page: String(page),
   139→   139→        limit: "20",
   140→   140→        sortBy,
   141→   141→        sortOrder: "desc",

   150→   150→        setProjects(data.data.projects);
   151→   151→        setTotalPages(data.data.pagination.totalPages);
   152→   152→        setTotal(data.data.pagination.total);

   155→   155→      errorToast({ error: "Failed to load projects" });

   157→   157→      setLoading(false);
   158→   158→    }
   159→   159→  }, [page, search, statusFilter, priorityFilter, sortBy]);
   160→   160→
   161→   161→  useEffect(() => {
   162→   162→    fetchProjects();

   164→   164→
   165→   165→  useEffect(() => {
   166→   166→    setPage(1);
   167→   167→  }, [search, statusFilter, priorityFilter, sortBy]);
   168→   168→

   170→   170→    if (!createForm.name.trim()) {
   171→   171→      toast.error("Project name is required");
   172→   172→      return;
   173→   173→    }
   174→   174→    setCreateLoading(true);
   175→   175→    try {
   176→   176→      const payload: Record<string, unknown> = {
   177→   177→        ...createForm,
   178→   178→        deadline: createForm.deadline || undefined,
   179→   179→        clientName: createForm.clientName || undefined,
   180→   180→      };
   181→   181→
   182→   182→      if (createForm.clientId === "new" && showNewClientForm) {
   183→   183→        payload.clientId = "new";
   184→   184→        payload.newClient = newClientForm;
   185→   185→      } else if (createForm.clientId) {
   186→   186→        payload.clientId = createForm.clientId;
   187→   187→        // Auto-fill clientName from selected client
   188→   188→        const sel = clients.find(c => c.id === createForm.clientId);
   189→   189→        if (sel) payload.clientName = sel.name;
   190→   190→      }
   191→   191→
   192→   192→      const res = await fetch("/api/projects", {
   193→   193→        method: "POST",
   194→   194→        headers: { "Content-Type": "application/json" },
   195→   195→        body: JSON.stringify(payload),
   196→   196→      });
   197→   197→      const data = await res.json();
   198→   198→      if (data.success) {
   199→   199→        toast.success("Project created successfully");
   200→   200→        setCreateOpen(false);
   201→   201→        setCreateForm({ name: "", description: "", priority: "MEDIUM", clientName: "", clientId: "", color: "#6366f1", deadline: "" });
   202→   202→        setShowNewClientForm(false);
   203→   203→        setNewClientForm({ name: "", email: "", company: "", address: "", phone: "" });
   204→   204→        fetchProjects();
   205→   205→        // Navigate to project detail to manage team
   206→   206→        router.push(`/dashboard/projects/${data.data.project.id}`);
   207→   207→      } else {
   208→   208→        errorToast({ error: data.error, title: "Failed to create project" });
   209→   209→      }
   210→   210→    } catch {
   211→   211→      errorToast({ error: "Failed to create project" });
   212→   212→    } finally {
   213→   213→      setCreateLoading(false);
   214→   214→    }
   215→   215→  };
   216→   216→
   217→   217→  const handleDelete = async (project: Project) => {
   218→   218→    if (!confirm(`Are you sure you want to archive "${project.name}"?`)) return;
   219→   219→    try {
   220→   220→      const res = await fetch(`/api/projects/${project.id}`, { method: "DELETE" });
   221→   221→      const data = await res.json();
   222→   222→      if (data.success) {
   223→   223→        toast.success("Project archived");
   224→   224→        fetchProjects();
   225→   225→      } else {
   226→   226→        errorToast({ error: data.error, title: "Failed to archive project" });
   227→   227→      }
   228→   228→    } catch {
   229→   229→      errorToast({ error: "Failed to archive project" });
   230→   230→    }
   231→   231→  };
   232→   232→
   233→   233→  function formatDate(dateStr: string | null) {
   234→   234→    if (!dateStr) return null;
   235→   235→    try {
   236→   236→      return new Date(dateStr).toLocaleDateString("en-GB", {
   237→   237→        day: "numeric",
   238→   238→        month: "short",
   239→   239→        year: "numeric",
   240→   240→      });
   241→   241→    } catch {
   242→   242→      return dateStr;
   243→   243→    }
   244→   244→  }
   245→   245→
   246→   246→  function isOverdue(deadline: string | null, status: string) {
   247→   247→    if (!deadline || status === "COMPLETED" || status === "ARCHIVED") return false;
   248→   248→    return new Date(deadline) < new Date();
   249→   249→  }
   250→   250→
   251→   251→  return (
   252→   252→    <div className="space-y-6">
   253→   253→      {/* Header */}
   254→   254→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   255→   255→        <div>
   256→   256→          <h1 className="text-2xl font-bold tracking-tight">Projects</h1>
   257→   257→          <p className="text-sm text-muted-foreground mt-1">
   258→   258→            {total} project{total !== 1 ? "s" : ""} total
   259→   259→          </p>
   260→   260→        </div>
   261→   261→        {isAdmin && (
   262→   262→          <Button onClick={() => setCreateOpen(true)}>
   263→   263→            <Plus className="mr-2 h-4 w-4" />
   264→   264→            New Project
   265→   265→          </Button>
   266→   266→        )}
   267→   267→      </div>
   268→   268→
   269→   269→      {/* Filters */}
   270→   270→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   271→   271→        <div className="relative flex-1">
   272→   272→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   273→   273→          <Input
   274→   274→            placeholder="Search projects..."
   275→   275→            value={search}
   276→   276→            onChange={(e) => setSearch(e.target.value)}
   277→   277→            className="pl-9"
   278→   278→          />
   279→   279→        </div>
   280→   280→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   281→   281→          <SelectTrigger className="w-full sm:w-[140px]">
   282→   282→            <SelectValue placeholder="All Status" />
   283→   283→          </SelectTrigger>
   284→   284→          <SelectContent>
   285→   285→            <SelectItem value="all">All Status</SelectItem>
   286→   286→            {Object.entries(PROJECT_STATUSES).map(([key, val]) => (
   287→   287→              <SelectItem key={key} value={val}>
   288→   288→                {val.charAt(0) + val.slice(1).toLowerCase().replace("_", " ")}
   289→   289→              </SelectItem>
   290→   290→            ))}
   291→   291→          </SelectContent>
   292→   292→        </Select>
   293→   293→        <Select value={priorityFilter} onValueChange={setPriorityFilter}>
   294→   294→          <SelectTrigger className="w-full sm:w-[140px]">
   295→   295→            <SelectValue placeholder="All Priority" />
   296→   296→          </SelectTrigger>
   297→   297→          <SelectContent>
   298→   298→            <SelectItem value="all">All Priority</SelectItem>
   299→   299→            <SelectItem value="HIGH">High</SelectItem>
   300→   300→            <SelectItem value="MEDIUM">Medium</SelectItem>
   301→   301→            <SelectItem value="LOW">Low</SelectItem>
   302→   302→          </SelectContent>
   303→   303→        </Select>
   304→   304→        <Select value={sortBy} onValueChange={setSortBy}>
   305→   305→          <SelectTrigger className="w-full sm:w-[160px]">
   306→   306→            <ArrowUpDown className="mr-2 h-4 w-4" />
   307→   307→            <SelectValue />
   308→   308→          </SelectTrigger>
   309→   309→          <SelectContent>
   310→   310→            <SelectItem value="createdAt">Newest First</SelectItem>
   311→   311→            <SelectItem value="updatedAt">Recently Updated</SelectItem>
   312→   312→            <SelectItem value="name">Name (A-Z)</SelectItem>
   313→   313→            <SelectItem value="deadline">Deadline</SelectItem>
   314→   314→          </SelectContent>
   315→   315→        </Select>
   316→   316→      </div>
   317→   317→
   318→   318→      {/* Loading */}
   319→   319→      {loading && (
   320→   320→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   321→   321→          {[...Array(6)].map((_, i) => (
   322→   322→            <Card key={i} className="animate-pulse">
   323→   323→              <CardContent className="p-6">
   324→   324→                <div className="h-4 bg-muted rounded w-3/4 mb-3" />
   325→   325→                <div className="h-3 bg-muted rounded w-1/2 mb-6" />
   326→   326→                <div className="flex gap-2">
   327→   327→                  <div className="h-6 bg-muted rounded w-16" />
   328→   328→                  <div className="h-6 bg-muted rounded w-16" />
   329→   329→                </div>
   330→   330→              </CardContent>
   331→   331→            </Card>
   332→   332→          ))}
   333→   333→        </div>
   334→   334→      )}
   335→   335→
   336→   336→      {/* Empty State */}
   337→   337→      {!loading && projects.length === 0 && (
   338→   338→        <Card>
   339→   339→          <CardContent className="flex flex-col items-center justify-center py-16">
   340→   340→            <FolderKanban className="h-12 w-12 text-muted-foreground mb-4" />
   341→   341→            <h3 className="text-lg font-semibold">No projects found</h3>
   342→   342→            <p className="text-sm text-muted-foreground mt-1 mb-4">
   343→   343→              {search || (statusFilter && statusFilter !== "all") || (priorityFilter && priorityFilter !== "all")
   344→   344→                ? "Try adjusting your search or filters"
   345→   345→                : isAdmin
   346→   346→                  ? "Create your first project to get started"
   347→   347→                  : "No projects have been created yet"}
   348→   348→            </p>
   349→   349→            {isAdmin && !search && (!statusFilter || statusFilter === "all") && (!priorityFilter || priorityFilter === "all") && (
   350→   350→              <Button onClick={() => setCreateOpen(true)}>
   351→   351→                <Plus className="mr-2 h-4 w-4" />
   352→   352→                Create Project
   353→   353→              </Button>
   354→   354→            )}
   355→   355→          </CardContent>
   356→   356→        </Card>
   357→   357→      )}
   358→   358→
   359→   359→      {/* Projects Grid */}
   360→   360→      {!loading && projects.length > 0 && (
   361→   361→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   362→   362→          {projects.map((project) => {
   363→   363→            const overdue = isOverdue(project.deadline, project.status);
   364→   364→            const priority = priorityConfig[project.priority] || priorityConfig.MEDIUM;
   365→   365→            const status = statusConfig[project.status] || statusConfig.ACTIVE;
   366→   366→
   367→   367→            return (
   368→   368→              <Card
   369→   369→                key={project.id}
   370→   370→                className="group relative hover:shadow-md transition-shadow cursor-pointer border-l-4"
   371→   371→                style={{ borderLeftColor: project.color || "#6366f1" }}
   372→   372→                onClick={() => router.push(`/dashboard/projects/${project.id}`)}
   373→   373→              >
   374→   374→                <CardContent className="p-5">
   375→   375→                  {/* Top row: name + actions */}
   376→   376→                  <div className="flex items-start justify-between gap-2 mb-3">
   377→   377→                    <div className="min-w-0 flex-1">
   378→   378→                      <h3 className="font-semibold text-sm truncate">{project.name}</h3>
   379→   379→                      {project.clientName && (
   380→   380→                        <p className="text-xs text-muted-foreground mt-0.5 truncate">
   381→   381→                          {project.clientName}
   382→   382→                        </p>
   383→   383→                      )}
   384→   384→                    </div>
   385→   385→                    {isAdmin && (
   386→   386→                      <DropdownMenu>
   387→   387→                        <DropdownMenuTrigger asChild>
   388→   388→                          <Button
   389→   389→                            variant="ghost"
   390→   390→                            size="icon"
   391→   391→                            className="h-7 w-7 shrink-0 opacity-0 group-hover:opacity-100"
   392→   392→                            onClick={(e) => e.stopPropagation()}
   393→   393→                          >
   394→   394→                            <MoreHorizontal className="h-4 w-4" />
   395→   395→                          </Button>
   396→   396→                        </DropdownMenuTrigger>
   397→   397→                        <DropdownMenuContent align="end">
   398→   398→                          <DropdownMenuItem
   399→   399→                            onClick={(e) => {
   400→   400→                              e.stopPropagation();
   401→   401→                              router.push(`/dashboard/projects/${project.id}`);
   402→   402→                            }}
   403→   403→                          >
   404→   404→                            <Pencil className="mr-2 h-4 w-4" />
   405→   405→                            Edit
   406→   406→                          </DropdownMenuItem>
   407→   407→                          {isSuperAdmin && (
   408→   408→                            <DropdownMenuItem
   409→   409→                              className="text-destructive"
   410→   410→                              onClick={(e) => {
   411→   411→                                e.stopPropagation();
   412→   412→                                handleDelete(project);
   413→   413→                              }}
   414→   414→                            >
   415→   415→                              <Trash2 className="mr-2 h-4 w-4" />
   416→   416→                              Archive
   417→   417→                            </DropdownMenuItem>
   418→   418→                          )}
   419→   419→                        </DropdownMenuContent>
   420→   420→                      </DropdownMenu>
   421→   421→                    )}
   422→   422→                  </div>
   423→   423→
   424→   424→                  {/* Description */}
   425→   425→                  {project.description && (
   426→   426→                    <p className="text-xs text-muted-foreground line-clamp-2 mb-3">
   427→   427→                      {project.description}
   428→   428→                    </p>
   429→   429→                  )}
   430→   430→
   431→   431→                  {/* Badges */}
   432→   432→                  <div className="flex flex-wrap items-center gap-1.5 mb-3">
   433→   433→                    <Badge variant="secondary" className={`text-[10px] px-1.5 py-0 ${priority.className}`}>
   434→   434→                      {priority.label}
   435→   435→                    </Badge>
   436→   436→                    <Badge variant="secondary" className={`text-[10px] px-1.5 py-0 ${status.className}`}>
   437→   437→                      {status.label}
   438→   438→                    </Badge>
   439→   439→                  </div>
   440→   440→
   441→   441→                  {/* Bottom row: deadline + team */}
   442→   442→                  <div className="flex items-center justify-between text-xs text-muted-foreground">
   443→   443→                    <div className="flex items-center gap-1">
   444→   444→                      {project.deadline ? (
   445→   445→                        <>
   446→   446→                          <Calendar className="h-3.5 w-3.5" />
   447→   447→                          <span className={overdue ? "text-destructive font-medium" : ""}>
   448→   448→                            {formatDate(project.deadline)}
   449→   449→                          </span>
   450→   450→                          {overdue && <AlertCircle className="h-3 w-3 text-destructive" />}
   451→   451→                        </>
   452→   452→                      ) : (
   453→   453→                        <span>No deadline</span>
   454→   454→                      )}
   455→   455→                    </div>
   456→   456→                    <div className="flex items-center gap-1">
   457→   457→                      <Users className="h-3.5 w-3.5" />
   458→   458→                      <span>{project.memberCount}</span>
   459→   459→                    </div>
   460→   460→                  </div>
   461→   461→
