   128→  // Logo upload handler
   129→  async function handleLogoUpload(e: React.ChangeEvent<HTMLInputElement>) {
   130→    const file = e.target.files?.[0];
   131→    if (!file) return;
   132→    if (file.size > 500 * 1024) {
   133→      errorToast({ error: "File too large. Maximum size is 500 KB.", title: "Upload failed" });
   134→      return;
   135→    }
   136→    setLogoSaving(true);
   137→    try {
   138→      const dataUrl = await new Promise<string>((resolve, reject) => {
   139→        const reader = new FileReader();
   140→        reader.onload = () => resolve(reader.result as string);
   141→        reader.onerror = reject;
   142→        reader.readAsDataURL(file);
   143→      });
   144→      const res = await fetch("/api/settings", {
   145→        method: "PUT",
   146→        headers: { "Content-Type": "application/json" },
   147→        body: JSON.stringify({ settings: { BRANDING_LOGO: dataUrl } }),
   148→      });
   149→      const json = await res.json();
   150→      if (json.success) {
   151→        setBrandingLogo(dataUrl);
   152→        toast.success("Logo updated successfully");
   153→      } else {
   154→        errorToast({ error: json.error, title: "Failed to save logo" });
   155→      }
   156→    } catch (err) {
   157→      errorToast({ error: err, title: "Logo upload failed" });
   158→    } finally {
   159→      setLogoSaving(false);
   160→      // Reset input so same file can be re-selected
   161→      if (logoInputRef.current) logoInputRef.current.value = "";
   162→    }
   163→  }

   311→    setSaving(true);
   312→    try {
   313→      const updateSettings: Record<string, string> = {};
   314→
   315→      if (emailProvider !== (settings.EMAIL_PROVIDER?.value || "")) {
   316→        updateSettings.EMAIL_PROVIDER = emailProvider;
   317→      }
   318→      if (resendKeyChanged && resendApiKey) {
   319→        updateSettings.RESEND_API_KEY = resendApiKey;
   320→      }
   321→      if (fromEmail !== (settings.RESEND_FROM_EMAIL?.value || "")) {
   322→        updateSettings.RESEND_FROM_EMAIL = fromEmail;
   323→      }
   324→      if (fromName !== (settings.RESEND_FROM_NAME?.value || "")) {
   325→        updateSettings.RESEND_FROM_NAME = fromName;
   326→      }
   327→      if (smtpUser !== (settings.SMTP_USER?.value || "")) {
   328→        updateSettings.SMTP_USER = smtpUser;
   329→      }
   330→      if (smtpPassChanged && smtpPassword) {
   331→        updateSettings.SMTP_PASSWORD = smtpPassword;
   332→      }
   333→
   334→      if (Object.keys(updateSettings).length === 0) {
   335→        toast.info("No changes to save");
   336→        setSaving(false);
   337→        return;
   338→      }

   340→      const res = await fetch("/api/settings", {
   341→        method: "PUT",
   342→        headers: { "Content-Type": "application/json" },

   344→      });
   345→      const json = await res.json();
   346→
   347→      if (json.success) {
   348→        toast.success(json.message || "Settings saved successfully");
   349→        setResendKeyChanged(false);
   350→        setSmtpPassChanged(false);
   351→
   352→        // Refresh settings
   353→        const refreshRes = await fetch("/api/settings");
   354→        const refreshJson = await refreshRes.json();
   355→        if (refreshJson.success) {
   356→          setSettings(refreshJson.data);
   357→          const data = refreshJson.data as Record<string, SettingItem>;
   358→          if (data.RESEND_API_KEY && !resendKeyChanged) setResendApiKey(data.RESEND_API_KEY.value);
   359→          if (data.RESEND_FROM_EMAIL) setFromEmail(data.RESEND_FROM_EMAIL.value);
   360→          if (data.RESEND_FROM_NAME) setFromName(data.RESEND_FROM_NAME.value);
   361→          if (data.EMAIL_PROVIDER) setEmailProvider(data.EMAIL_PROVIDER.value);

   364→          if (data.ZAI_BRIDGE_API_KEY && !zaiApiKeyChanged) setZaiApiKey(data.ZAI_BRIDGE_API_KEY.value);
   365→          if (data.ZAI_BRIDGE_BASE_URL) setZaiBaseUrl(data.ZAI_BRIDGE_BASE_URL.value);
