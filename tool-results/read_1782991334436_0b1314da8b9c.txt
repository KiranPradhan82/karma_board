   701→function getDocWorkflowFooter(
   702→  currentDocLabel: string,
   703→  currentDocIndex: number,
   704→  nextCommand: string | null,
   705→  firstName: string,
   706→  isLastDoc: boolean,

   720→    if (i < currentDocIndex - 1) {
   721→      docs[i][2] = "Done";
