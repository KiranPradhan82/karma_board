    34→    34→function detectDocTypeFromContent(content: string): string | null {
    35→    35→  if (content.length < 1500) return null;
    36→    36→  const lower = content.toLowerCase();
    37→    37→  let bestMatch = "";
    38→    38→  let bestCount = 0;
    39→    39→  for (const sig of DOC_SIGNATURES) {
    40→    40→    const count = sig.keywords.filter(kw => lower.includes(kw.toLowerCase())).length;
    41→    41→    if (count >= 3 && count > bestCount) {
    42→    42→      bestCount = count;
    43→    43→      bestMatch = sig.label;
    44→    44→    }
    45→    45→  }
    46→    46→  return bestMatch || null;
    47→    47→}

   176→   176→    const user = await getAuthUser(request);

   182→   182→    const projectId = searchParams.get("projectId");
   183→   183→    if (!projectId) {
   184→   184→      return NextResponse.json({ success: false, error: "projectId is required" }, { status: 400 });
   185→   185→    }

   189→   189→    if (!canAccess) {

   192→   192→
   193→   193→    const page = Math.max(1, parseInt(searchParams.get("page") || "1"));
   194→   194→    const limit = Math.min(100, Math.max(1, parseInt(searchParams.get("limit") || "50")));
   195→   195→
   196→   196→    const client = getTursoClient();
   197→   197→    const offset = (page - 1) * limit;
   198→   198→
   199→   199→    let messagesResult;
   200→   200→

   202→   202→      // SUPERADMIN sees all messages — newest first (client will reverse for display)

   206→   206→              LEFT JOIN "User" u ON m."userId" = u.id
   207→   207→              WHERE m."projectId" = ?
   208→   208→              ORDER BY m."timestamp" DESC
   209→   209→              LIMIT ? OFFSET ?`,
   210→   210→        args: [projectId, limit, offset],
   211→   211→      });
   212→   212→    } else {
   213→   213→      // Others see only own messages + assistant messages — newest first
   214→   214→      messagesResult = await client.execute({
   215→   215→        sql: `SELECT m.*, u.name as userName
   216→   216→              FROM "AiChat" m
   217→   217→              LEFT JOIN "User" u ON m."userId" = u.id
   218→   218→              WHERE m."projectId" = ? AND (m."userId" = ? OR m."role" = 'assistant')
   219→   219→              ORDER BY m."timestamp" DESC
   220→   220→              LIMIT ? OFFSET ?`,
   221→   221→        args: [projectId, user.id, limit, offset],
   222→   222→      });
   223→   223→    }
   224→   224→
   225→   225→    const countResult = await client.execute({
   226→   226→      sql: `SELECT COUNT(*) as total FROM "AiChat" WHERE "projectId" = ?`,
   227→   227→      args: [projectId],
   228→   228→    });
   229→   229→    const total = Number(countResult.rows[0].total);
   230→   230→
   231→   231→    const messages = messagesResult.rows.map((row) => ({
   232→   232→      id: row.id,
   233→   233→      userId: row.userId,
   234→   234→      projectId: row.projectId,
   235→   235→      role: row.role,
   236→   236→      content: row.content,
   237→   237→      timestamp: row.timestamp,
   238→   238→      userName: row.userName,
   239→   239→    }));
   240→   240→
   241→   241→    return NextResponse.json({
   242→   242→      success: true,
   243→   243→      data: {
   244→   244→        messages,
   245→   245→        pagination: {
   246→   246→          page,
   247→   247→          limit,
   248→   248→          total,
   249→   249→          totalPages: Math.ceil(total / limit),
   250→   250→        },
   251→   251→      },
   252→   252→    });
   253→   253→  } catch (error) {
   254→   254→    console.error("[GET /api/ai/chat] Error:", error);
   255→   255→    return NextResponse.json({ success: false, error: "Internal server error" }, { status: 500 });
   256→   256→  }

   263→   263→/**
   264→   264→ * Parse tasks from a markdown document. Uses doc-type-specific patterns
   265→   265→ * to extract actionable items from PRD, TRD, Flow, UX, Schema, and Plan docs.
   266→   266→ */
   267→   267→async function autoGenerateTodosFromDoc(
   268→   268→  dbClient: ReturnType<typeof getTursoClient>,
   269→   269→  projectId: string,
   270→   270→  docContent: string,
   271→   271→  docType: string,
   272→   272→  userId: string,
   273→   273→): Promise<number> {
   274→   274→  // Get current max sortOrder for the project
   275→   275→  const maxOrder = await dbClient.execute({
   276→   276→    sql: `SELECT COALESCE(MAX("sortOrder"), -1) as maxOrder FROM "ProjectTodo" WHERE "projectId" = ?`,
   277→   277→    args: [projectId],
   278→   278→  });
   279→   279→  let sortOrder = Number(maxOrder.rows[0].maxOrder) + 1;
   280→   280→
   281→   281→  const tasks: { title: string; description: string; priority: string }[] = [];
   282→   282→  const lines = docContent.split('\n');
   283→   283→  let currentSection = '';
   284→   284→  let currentPhase = '';
   285→   285→
   286→   286→  // Doc-type-specific section prefixes for richer descriptions
   287→   287→  const DOC_LABELS: Record<string, string> = {
   288→   288→    prd: "Product Requirements",
   289→   289→    trd: "Technical Requirements",
   290→   290→    flow: "Application Flow",
   291→   291→    ux: "UI/UX Design",
   292→   292→    schema: "Database Schema",
   293→   293→    plan: "Implementation Plan",
   294→   294→  };
   295→   295→  const docLabel = DOC_LABELS[docType] || docType.toUpperCase();
   296→   296→
   297→   297→  for (let i = 0; i < lines.length; i++) {
   298→   298→    const line = lines[i].trim();
   299→   299→
   300→   300→    // Track current section/phase for context
   301→   301→    if (line.match(/^#{1,3}\s/)) {
   302→   302→      currentSection = line.replace(/^#+\s*/, '').replace(/\*\*/g, '').trim();
   303→   303→      if (/phase/i.test(currentSection)) {
   304→   304→        currentPhase = currentSection;
   305→   305→      }
   306→   306→      continue;
   307→   307→    }
   308→   308→
   309→   309→    // Match markdown list items: "- [ ] task", "- task", "* task", numbered lists
   310→   310→    const taskMatch = line.match(/^[-*]\s+\[?\s*\]?\s+(.+)/) || line.match(/^\d+\.\s+(.+)/);
   311→   311→    if (taskMatch) {
   312→   312→      let taskText = taskMatch[1].trim();
   313→   313→      // Skip if it's a sub-heading or too short
   314→   314→      if (taskText.length < 5) continue;
   315→   315→      // Skip if it's just a bold title (like a section header in a list)
   316→   316→      if (taskText.match(/^\*\*[^*]+\*\*$/) && taskText.length < 80) continue;
   317→   317→
   318→   318→      // Determine priority from context
   319→   319→      let priority = 'MEDIUM';
   320→   320→      if (/\b(critical|urgent|high.?priority|must.?have|blocker|p0)\b/i.test(taskText) ||
   321→   321→          /\b(critical|urgent|high.?priority|must.?have|blocker|p0)\b/i.test(currentSection)) {
   322→   322→        priority = 'HIGH';
   323→   323→      } else if (/\b(low.?priority|nice.?to.?have|optional|future|p[2-3])\b/i.test(taskText) ||
   324→   324→                 /\b(low.?priority|nice.?to.?have|optional|future|p[2-3])\b/i.test(currentSection)) {
   325→   325→        priority = 'LOW';
   326→   326→      }
   327→   327→
   328→   328→      // Build description from context
   329→   329→      let description = `Source: ${docLabel}`;
   330→   330→      if (currentPhase) {
   331→   331→        description += `\nPhase: ${currentPhase}`;
   332→   332→      }
   333→   333→      if (currentSection && currentSection !== currentPhase) {
   334→   334→        description += `\nSection: ${currentSection}`;
   335→   335→      }
   336→   336→
   337→   337→      // Try to grab a sub-description line (next non-empty, non-list line)
   338→   338→      for (let j = i + 1; j < Math.min(i + 3, lines.length); j++) {
   339→   339→        const nextLine = lines[j].trim();
   340→   340→        if (nextLine && !nextLine.match(/^[-*#]/) && !nextLine.match(/^\d+\./)) {
   341→   341→          description += `\n${nextLine}`;
   342→   342→          break;
   343→   343→        }
   344→   344→      }
   345→   345→
   346→   346→      tasks.push({
   347→   347→        title: taskText,
   348→   348→        description: description.trim(),
   349→   349→        priority,
   350→   350→      });
   351→   351→    }
   352→   352→  }
   353→   353→
   354→   354→  if (tasks.length === 0) return 0;
   355→   355→
   356→   356→  // Check for existing todos to avoid duplicates
   357→   357→  const existingTodos = await dbClient.execute({
   358→   358→    sql: `SELECT title FROM "ProjectTodo" WHERE "projectId" = ?`,
   359→   359→    args: [projectId],
   360→   360→  });
   361→   361→  const existingTitles = new Set(existingTodos.rows.map(r => (r.title as string).toLowerCase()));
   362→   362→
   363→   363→  const newTasks = tasks.filter(t => !existingTitles.has(t.title.toLowerCase()));
   364→   364→  if (newTasks.length === 0) {
   365→   365→    console.log(`[autoGenerateTodos] All ${tasks.length} tasks from ${docType} already exist, skipping.`);
   366→   366→    return 0;
   367→   367→  }
   368→   368→
   369→   369→  // Batch insert all tasks
   370→   370→  for (const task of newTasks) {
   371→   371→    const id = crypto.randomUUID();
   372→   372→    await dbClient.execute({
   373→   373→      sql: `INSERT INTO "ProjectTodo" (id, "projectId", title, description, status, priority, "sortOrder", "createdBy", "createdAt", "updatedAt")
   374→   374→            VALUES (?, ?, ?, ?, 'PENDING', ?, ?, ?, datetime('now'), datetime('now'))`,
   375→   375→      args: [id, projectId, task.title, task.description || null, task.priority, sortOrder++, userId],
   376→   376→    });
   377→   377→  }
   378→   378→
   379→   379→  console.log(`[autoGenerateTodos] Created ${newTasks.length} tasks from ${docType} (out of ${tasks.length} parsed, ${existingTitles.size} existing)`);
   380→   380→  return newTasks.length;
   381→   381→}

   383→   383→// POST /api/ai/chat — Send message and get AI response (with agentic tool-calling loop)
   384→   384→export async function POST(request: NextRequest) {
   385→   385→  try {
   386→   386→    const user = await getAuthUser(request);
   387→   387→    if (!user) {
   388→   388→      return NextResponse.json({ success: false, error: "Authentication required" }, { status: 401 });
   389→   389→    }
   390→   390→
   391→   391→    const rawBody = await request.json();
   392→   392→    const { projectId, content, files, fileData, fileName, fileType } = rawBody as { projectId: string; content: string; files?: unknown[]; fileData?: string; fileName?: string; fileType?: string };
   393→   393→
   394→   394→    if (!projectId || !content) {
   395→   395→      return NextResponse.json({ success: false, error: "projectId and content are required" }, { status: 400 });
   396→   396→    }
   397→   397→
   398→   398→    const attachedImages: { data: string; type: string; name: string }[] = [];
   399→   399→    if (files && Array.isArray(files) && files.length > 0) {
   400→   400→      // New format: array of { data, type, name }
   401→   401→      for (const f of files) {
   402→   402→        if (f.data && f.type && f.type.startsWith("image/")) {
   403→   403→          attachedImages.push({ data: f.data, type: f.type, name: f.name || "image" });
   404→   404→        }
   405→   405→      }
