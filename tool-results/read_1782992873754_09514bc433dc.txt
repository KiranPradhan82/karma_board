   153→/** Ensure content is always a string — prevents React error #31 when API returns objects */
   154→function safeContent(content: unknown): string {
   155→  return typeof content === "string" ? content : content != null ? JSON.stringify(content) : "";
   156→}
   157→
   158→/** Ensure error is always a string — prevents React error #31 when API returns error objects like {code, id, message} */
   159→function safeError(err: unknown, fallback: string): string {
   160→  if (typeof err === "string") return err;
   161→  if (err && typeof err === "object") {
   162→    const obj = err as Record<string, unknown>;
   163→    if (typeof obj.message === "string") return obj.message;
   164→    try { return JSON.stringify(err); } catch { /* fall through */ }
   165→  }
   166→  return fallback;
   167→}
