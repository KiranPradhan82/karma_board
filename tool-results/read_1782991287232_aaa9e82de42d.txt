   106→const ROLE_STYLES: Record<string, string> = {
   107→  SUPERADMIN: 'bg-red-100 text-red-700 border-red-200',
   108→  ADMIN: 'bg-amber-100 text-amber-700 border-amber-200',
   109→  MEMBER: 'bg-slate-100 text-slate-700 border-slate-200',
   110→};
   111→
   112→const STATUS_STYLES: Record<string, string> = {
   113→  ACTIVE: 'bg-emerald-100 text-emerald-700 border-emerald-200',
   114→  INACTIVE: 'bg-gray-100 text-gray-600 border-gray-200',
   115→  ON_LEAVE: 'bg-sky-100 text-sky-700 border-sky-200',
   116→};
   117→
   118→function getInitials(name: string) {
   119→  return name
   120→    .split(' ')
   121→    .map((n) => n[0])
   122→    .join('')
   123→    .toUpperCase()
   124→    .slice(0, 2);
   125→}
   126→
   127→function parseSkills(skills: string | null | undefined): string[] {
   128→  if (!skills) return [];
   129→  try {
   130→    return JSON.parse(skills);
   131→  } catch {
   132→    return skills.split(',').map((s) => s.trim()).filter(Boolean);
   133→  }
   134→}

   162→function MemberDialog({
   163→  open,
   164→  onOpenChange,
   165→  mode,
   166→  member,
   167→  onSubmit,
   168→  loading,
   169→}: {
   170→  open: boolean;
   171→  onOpenChange: (open: boolean) => void;
   172→  mode: 'add' | 'edit';
   173→  member: Member | null;
   174→  onSubmit: (data: MemberFormData) => void;
   175→  loading: boolean;
   176→}) {
   177→  const dialogKey = `${mode}-${member?.id || 'new'}`;
   178→
   179→  return open ? (
   180→    <div key={dialogKey}>
   181→      <MemberDialogInner
   182→        mode={mode}
   183→        member={member}
   184→        onSubmit={onSubmit}
   185→        loading={loading}
   186→        onClose={() => onOpenChange(false)}
   187→      />
   188→    </div>
   189→  ) : null;
   190→}
   191→
   192→function MemberDialogInner({
   193→  mode,
   194→  member,
   195→  onSubmit,
   196→  loading,
   197→  onClose,
   198→}: {
   199→  open: boolean;
   200→  onOpenChange: (open: boolean) => void;
   201→  mode: 'add' | 'edit';
   202→  member: Member | null;
   203→  onSubmit: (data: MemberFormData) => void;
   204→  loading: boolean;
   205→}) {
   206→  const [form, setForm] = useState<MemberFormData>(() => {
   207→    if (mode === 'edit' && member) {
   208→      return {
   209→        name: member.name,
   210→        email: member.email,
   211→        password: '',
   212→        jobTitle: member.jobTitle || '',
   213→        phone: member.phone || '',
   214→        skills: parseSkills(member.skills).join(', '),
   215→        role: member.role,
   216→        status: member.status,
   217→      };
   218→    }
   219→    return EMPTY_FORM;
   220→  });
   221→  const [errors, setErrors] = useState<Record<string, string>>({});
   222→
   223→
   224→  function validate(): boolean {
   225→    const errs: Record<string, string> = {};
   226→    if (!form.name || form.name.length < 2) errs.name = 'Name must be at least 2 characters';
   227→    if (!form.email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) errs.email = 'Valid email is required';
   228→    // Password is auto-generated for new members — no validation needed
   229→    setErrors(errs);
   230→    return Object.keys(errs).length === 0;
   231→  }
   232→
   233→  function handleSubmit(e: React.FormEvent) {
   234→    e.preventDefault();
   235→    if (validate()) onSubmit(form);
   236→  }
   237→
   238→  return (
   239→    <Dialog open onOpenChange={(v) => { if (!v) onClose(); }}>
   240→      <DialogContent className="sm:max-w-md max-h-[90vh] overflow-y-auto">
   241→        <DialogHeader>
   242→          <DialogTitle>{mode === 'add' ? 'Add New Member' : 'Edit Member'}</DialogTitle>
   243→          <DialogDescription>
   244→            {mode === 'add'
   245→              ? 'Create a new team member account.'
   246→              : `Edit ${member?.name || 'member'} details.`}
   247→          </DialogDescription>
   248→        </DialogHeader>
   249→
   250→        <form onSubmit={handleSubmit} className="space-y-4">
   251→          {/* Name */}
   252→          <div className="space-y-1.5">
   253→            <Label htmlFor="name">Name *</Label>
   254→            <Input
   255→              id="name"
   256→              placeholder="John Doe"
   257→              value={form.name}
   258→              onChange={(e) => setForm({ ...form, name: e.target.value })}
   259→            />
   260→            {errors.name && <p className="text-sm text-destructive">{errors.name}</p>}
   261→          </div>
   262→
   263→          {/* Email */}
   264→          <div className="space-y-1.5">
   265→            <Label htmlFor="email">Email *</Label>
   266→            <Input
   267→              id="email"
   268→              type="email"
   269→              placeholder="john@example.com"
   270→              value={form.email}
   271→              onChange={(e) => setForm({ ...form, email: e.target.value })}
   272→              disabled={mode === 'edit'}
   273→            />
   274→            {errors.email && <p className="text-sm text-destructive">{errors.email}</p>}
   275→          </div>
   276→
   277→          {/* Password is auto-generated — no field shown */}
   278→          {mode === 'add' && (
   279→            <div className="rounded-lg border bg-muted/50 px-3 py-2">
   280→              <p className="text-xs text-muted-foreground">
   281→                A temporary password will be automatically generated and sent to the member&apos;s email. They will be required to change it on first login.
   282→              </p>
   283→            </div>
   284→          )}
   285→
   286→          {/* Job Title */}
   287→          <div className="space-y-1.5">
   288→            <Label htmlFor="jobTitle">Job Title</Label>
   289→            <Input
   290→              id="jobTitle"
   291→              placeholder="Software Engineer"
   292→              value={form.jobTitle}
   293→              onChange={(e) => setForm({ ...form, jobTitle: e.target.value })}
   294→            />
   295→          </div>
   296→
   297→          {/* Phone */}
   298→          <div className="space-y-1.5">
   299→            <Label htmlFor="phone">Phone</Label>
   300→            <Input
   301→              id="phone"
   302→              placeholder="+1 (555) 123-4567"
   303→              value={form.phone}
   304→              onChange={(e) => setForm({ ...form, phone: e.target.value })}
   305→            />
   306→          </div>
   307→
   308→          {/* Skills */}
   309→          <div className="space-y-1.5">
   310→            <Label htmlFor="skills">Skills</Label>
   311→            <Input
   312→              id="skills"
   313→              placeholder="React, Node.js, TypeScript"
   314→              value={form.skills}
   315→              onChange={(e) => setForm({ ...form, skills: e.target.value })}
   316→            />
   317→            <p className="text-xs text-muted-foreground">Separate skills with commas</p>
   318→          </div>
   319→
   320→          {/* Role + Status */}
   321→          <div className="grid grid-cols-2 gap-3">
   322→            <div className="space-y-1.5">
   323→              <Label>Role</Label>
   324→              <Select
   325→                value={form.role}
   326→                onValueChange={(v) => setForm({ ...form, role: v })}
   327→              >
   328→                <SelectTrigger className="w-full">
   329→                  <SelectValue />
   330→                </SelectTrigger>
   331→                <SelectContent>
   332→                  <SelectItem value="MEMBER">Member</SelectItem>
   333→                  <SelectItem value="ADMIN">Admin</SelectItem>
   334→                  <SelectItem value="SUPERADMIN">Superadmin</SelectItem>
   335→                </SelectContent>
   336→              </Select>
   337→            </div>
   338→
   339→            {mode === 'edit' && (
   340→              <div className="space-y-1.5">
   341→                <Label>Status</Label>
   342→                <Select
   343→                  value={form.status}
   344→                  onValueChange={(v) => setForm({ ...form, status: v })}
   345→                >
   346→                  <SelectTrigger className="w-full">
   347→                    <SelectValue />
   348→                  </SelectTrigger>
   349→                  <SelectContent>
   350→                    <SelectItem value="ACTIVE">Active</SelectItem>
   351→                    <SelectItem value="INACTIVE">Inactive</SelectItem>
   352→                    <SelectItem value="ON_LEAVE">On Leave</SelectItem>
   353→                  </SelectContent>
   354→                </Select>
   355→              </div>
   356→            )}
   357→          </div>
   358→
   359→          <DialogFooter className="pt-2">
   360→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   361→              Cancel
   362→            </Button>
   363→            <Button type="submit" disabled={loading}>
   364→              {loading && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   365→              {mode === 'add' ? 'Create Member' : 'Save Changes'}
   366→            </Button>
   367→          </DialogFooter>
   368→        </form>
   369→      </DialogContent>
   370→    </Dialog>
   371→  );
   372→}

   424→            </div>
   425→            <div>
   426→              <p className="text-muted-foreground">Job Title</p>

   428→            </div>
   429→            <div>
   430→              <p className="text-muted-foreground">Phone</p>
   431→              <p className="mt-1 font-medium">{member.phone || '—'}</p>
   432→            </div>

   438→              <p className="text-muted-foreground">Joined</p>
   439→              <p className="mt-1 font-medium">

   443→              </p>
   444→            </div>
   445→          </div>
   446→

   452→                  <Badge key={skill} variant="secondary" className="text-xs">
   453→                    {skill}
   454→                  </Badge>
   455→                ))}
   456→              </div>
   457→            </div>
   458→          )}

   460→
   461→        <DialogFooter>
   462→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   463→            Close
   464→          </Button>
   465→        </DialogFooter>
   466→      </DialogContent>
   467→    </Dialog>
   468→  );

   628→
   629→export default function TeamPage() {
   630→  const { data: session } = useSession();
   631→  const userRole = (session?.user as { role?: string })?.role || 'MEMBER';
   632→  const userId = (session?.user as { id?: string })?.id || '';
   633→  const isAdmin = userRole === 'ADMIN' || userRole === 'SUPERADMIN';
   634→  const isSuperAdmin = userRole === 'SUPERADMIN';
   635→
   636→  // State
   637→  const [members, setMembers] = useState<Member[]>([]);
   638→  const [pagination, setPagination] = useState<Pagination>({ page: 1, limit: 20, total: 0, totalPages: 0 });
   639→  const [loading, setLoading] = useState(true);
   640→
   641→  // Filters
   642→  const [search, setSearch] = useState('');
   643→  const [statusFilter, setStatusFilter] = useState('');
   644→  const [roleFilter, setRoleFilter] = useState('');
   645→  const [sortBy, setSortBy] = useState('createdAt');
   646→  const [sortOrder, setSortOrder] = useState('desc');
   647→
   648→  // Selection
   649→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   650→
   651→  // Dialogs
   652→  const [memberDialogOpen, setMemberDialogOpen] = useState(false);
   653→  const [memberDialogMode, setMemberDialogMode] = useState<'add' | 'edit'>('add');
   654→  const [editingMember, setEditingMember] = useState<Member | null>(null);
   655→  const [dialogLoading, setDialogLoading] = useState(false);
   656→
   657→  const [detailDialogOpen, setDetailDialogOpen] = useState(false);
   658→  const [detailMember, setDetailMember] = useState<Member | null>(null);
   659→
   660→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   661→  const [deletingMember, setDeletingMember] = useState<Member | null>(null);
   662→  const [deleteLoading, setDeleteLoading] = useState(false);
   663→
   664→  const [bulkDeleteDialogOpen, setBulkDeleteDialogOpen] = useState(false);
   665→  const [bulkDeleteLoading, setBulkDeleteLoading] = useState(false);
   666→
   667→  // Fetch members
   668→  const fetchMembers = useCallback(async () => {
   669→    setLoading(true);
   670→    try {
   671→      const params = new URLSearchParams();
   672→      if (search) params.set('search', search);
   673→      if (statusFilter) params.set('status', statusFilter);
   674→      if (roleFilter) params.set('role', roleFilter);
   675→      params.set('page', pagination.page.toString());
   676→      params.set('limit', pagination.limit.toString());
   677→      params.set('sortBy', sortBy);
   678→      params.set('sortOrder', sortOrder);
   679→
   680→      const res = await fetch(`/api/members?${params.toString()}`);
   681→      const json = await res.json();
   682→
   683→      if (json.success) {
   684→        setMembers(json.data.members);
   685→        setPagination(json.data.pagination);
   686→      } else {
   687→        errorToast({ error: json.error, title: 'Failed to load members' });
   688→      }
   689→    } catch {
   690→      errorToast({ error: 'Failed to load members' });
   691→    } finally {
   692→      setLoading(false);
   693→    }
   694→  }, [search, statusFilter, roleFilter, pagination.page, pagination.limit, sortBy, sortOrder]);
   695→
   696→  useEffect(() => {
   697→    fetchMembers();
   698→  }, [fetchMembers]);
   699→
   700→  // Reset page when filters change
   701→  useEffect(() => {
   702→    setPagination((prev) => ({ ...prev, page: 1 }));
   703→  }, [search, statusFilter, roleFilter, sortBy, sortOrder]);
   704→
   705→  // Clear filters
   706→  function clearFilters() {
   707→    setSearch('');
   708→    setStatusFilter('');
   709→    setRoleFilter('');
   710→    setSortBy('createdAt');
   711→    setSortOrder('desc');
   712→  }
   713→
   714→  const hasActiveFilters = !!(search || statusFilter || roleFilter);
   715→
   716→  // Selection
   717→  function toggleSelect(id: string) {
   718→    setSelectedIds((prev) => {
   719→      const next = new Set(prev);
   720→      if (next.has(id)) next.delete(id);
   721→      else next.add(id);
   722→      return next;
   723→    });
   724→  }
   725→
   726→  function toggleSelectAll() {
   727→    if (selectedIds.size === members.length && members.length > 0) {
   728→      setSelectedIds(new Set());
   729→    } else {
   730→      setSelectedIds(new Set(members.map((m) => m.id)));
   731→    }
   732→  }
   733→
   734→  // Create / Update
   735→  async function handleFormSubmit(data: MemberFormData) {
   736→    setDialogLoading(true);
   737→    try {
   738→      // Convert skills to JSON array
   739→      const skillsArr = data.skills
   740→        .split(',')
   741→        .map((s) => s.trim())
   742→        .filter(Boolean);
   743→
   744→      if (memberDialogMode === 'add') {
   745→        const res = await fetch('/api/members', {
   746→          method: 'POST',
   747→          headers: { 'Content-Type': 'application/json' },
   748→          body: JSON.stringify({
   749→            ...data,
   750→            skills: skillsArr.length > 0 ? JSON.stringify(skillsArr) : null,
   751→          }),
   752→        });
   753→        const json = await res.json();
   754→        if (json.success) {
   755→          if (json.data.emailSent) {
   756→            toast.success('Member created! A welcome email with temporary password has been sent.');
   757→          } else {
   758→            const emailErr = json.data.emailError || 'Unknown email error';
   759→            toast.success(`Member created! Email not sent: ${emailErr}`);
   760→          }
   761→          setMemberDialogOpen(false);
   762→          fetchMembers();
   763→        } else {
   764→          errorToast({ error: json.error, title: 'Failed to create member' });
   765→        }
   766→      } else if (editingMember) {
   767→        const updateData: Record<string, unknown> = {};
   768→        if (data.name !== editingMember.name) updateData.name = data.name;
   769→        if (data.jobTitle !== (editingMember.jobTitle || '')) updateData.jobTitle = data.jobTitle || null;
   770→        if (data.phone !== (editingMember.phone || '')) updateData.phone = data.phone || null;
   771→        const existingSkills = parseSkills(editingMember.skills);
   772→        if (JSON.stringify(skillsArr) !== JSON.stringify(existingSkills)) {
   773→          updateData.skills = skillsArr.length > 0 ? JSON.stringify(skillsArr) : null;
   774→        }
   775→        if (data.role !== editingMember.role) updateData.role = data.role;
   776→        if (data.status !== editingMember.status) updateData.status = data.status;
   777→
   778→        if (Object.keys(updateData).length === 0) {
   779→          toast.info('No changes to save');
   780→          setMemberDialogOpen(false);
   781→          return;
   782→        }
   783→
   784→        const res = await fetch(`/api/members/${editingMember.id}`, {
   785→          method: 'PATCH',
   786→          headers: { 'Content-Type': 'application/json' },
   787→          body: JSON.stringify(updateData),
   788→        });
   789→        const json = await res.json();
   790→        if (json.success) {
   791→          toast.success('Member updated successfully');
   792→          setMemberDialogOpen(false);
   793→          fetchMembers();
   794→        } else {
   795→          errorToast({ error: json.error, title: 'Failed to update member' });
   796→        }
   797→      }
   798→    } catch (formError) {
   799→      const errMsg = formError instanceof Error ? formError.message : String(formError);
   800→      errorToast({ error: errMsg, title: 'Request failed' });
   801→      console.error('[team] Member form error:', formError);
   802→    } finally {
   803→      setDialogLoading(false);
   804→    }
   805→  }
   806→
   807→  // Delete single
   808→  async function handleDelete() {
   809→    if (!deletingMember) return;
   810→    setDeleteLoading(true);
   811→    try {
   812→      const res = await fetch(`/api/members/${deletingMember.id}`, { method: 'DELETE' });
   813→      const json = await res.json();
   814→      if (json.success) {
   815→        toast.success('Member deleted successfully');
   816→        setDeleteDialogOpen(false);
   817→        setDeletingMember(null);
   818→        setSelectedIds((prev) => {
   819→          const next = new Set(prev);
   820→          next.delete(deletingMember.id);
   821→          return next;
   822→        });
   823→        fetchMembers();
   824→      } else {
   825→        errorToast({ error: json.error, title: 'Failed to delete member' });
   826→      }
   827→    } catch (err) {
   828→      errorToast({ error: err, title: 'Delete member failed' });
   829→    } finally {
   830→      setDeleteLoading(false);
   831→    }
   832→  }
   833→
   834→  // Bulk delete
   835→  async function handleBulkDelete() {
   836→    if (selectedIds.size === 0) return;
   837→    setBulkDeleteLoading(true);
   838→    try {
   839→      const res = await fetch('/api/members/bulk-delete', {
   840→        method: 'POST',
   841→        headers: { 'Content-Type': 'application/json' },
   842→        body: JSON.stringify({ ids: Array.from(selectedIds) }),
   843→      });
   844→      const json = await res.json();
   845→      if (json.success) {
   846→        toast.success(`Deleted ${json.data.deleted} member(s) successfully`);
   847→        setBulkDeleteDialogOpen(false);
   848→        setSelectedIds(new Set());
   849→        fetchMembers();
   850→      } else {
   851→        errorToast({ error: json.error, title: 'Failed to delete members' });
   852→      }
   853→    } catch (err) {
   854→      errorToast({ error: err, title: 'Bulk delete failed' });
   855→    } finally {
   856→      setBulkDeleteLoading(false);
   857→    }
   858→  }
   859→
   860→  // Page navigation
   861→  function goToPage(page: number) {
   862→    setPagination((prev) => ({ ...prev, page }));
   863→  }
   864→
   865→  // Render pagination numbers
   866→  function renderPageNumbers() {
   867→    const pages: number[] = [];
   868→    const { totalPages, page: current } = pagination;
   869→    const maxVisible = 5;
   870→
   871→    let start = Math.max(1, current - Math.floor(maxVisible / 2));
   872→    const end = Math.min(totalPages, start + maxVisible - 1);
   873→    if (end - start + 1 < maxVisible) {
   874→      start = Math.max(1, end - maxVisible + 1);
   875→    }
   876→
   877→    for (let i = start; i <= end; i++) pages.push(i);
   878→    return pages;
   879→  }
   880→
   881→  return (
   882→    <TooltipProvider>
   883→      <div className="space-y-6">
   884→        {/* Header */}
   885→        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   886→          <div>
   887→            <h1 className="text-2xl font-bold tracking-tight">Team Members</h1>
   888→            <p className="text-muted-foreground text-sm mt-1">
   889→              Manage your team members, roles, and permissions.
   890→            </p>
   891→          </div>
   892→          {isAdmin && (
   893→            <Button
   894→              onClick={() => {
   895→                setMemberDialogMode('add');
   896→                setEditingMember(null);
   897→                setMemberDialogOpen(true);
   898→              }}
   899→            >
   900→              <Plus className="mr-2 h-4 w-4" />

   905→

   907→        <div className="flex flex-col sm:flex-row gap-3">

   909→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   910→            <Input
   911→              placeholder="Search by name or email..."
   912→              value={search}
   913→              onChange={(e) => setSearch(e.target.value)}
   914→              className="pl-9"
   915→            />
   916→          </div>
   917→
   918→          <Select value={statusFilter} onValueChange={(v) => setStatusFilter(v === 'all' ? '' : v)}>
   919→            <SelectTrigger className="w-[140px]">
   920→              <SelectValue placeholder="All Status" />
   921→            </SelectTrigger>
   922→            <SelectContent>
   923→              <SelectItem value="all">All Status</SelectItem>
   924→              <SelectItem value="ACTIVE">Active</SelectItem>
   925→              <SelectItem value="INACTIVE">Inactive</SelectItem>
   926→              <SelectItem value="ON_LEAVE">On Leave</SelectItem>
   927→            </SelectContent>
   928→          </Select>
   929→
   930→          <Select value={roleFilter} onValueChange={(v) => setRoleFilter(v === 'all' ? '' : v)}>
   931→            <SelectTrigger className="w-[140px]">
   932→              <SelectValue placeholder="All Roles" />
   933→            </SelectTrigger>
   934→            <SelectContent>
   935→              <SelectItem value="all">All Roles</SelectItem>
   936→              <SelectItem value="SUPERADMIN">Superadmin</SelectItem>
   937→              <SelectItem value="ADMIN">Admin</SelectItem>
   938→              <SelectItem value="MEMBER">Member</SelectItem>
   939→            </SelectContent>
   940→          </Select>
   941→
   942→          <Select value={sortBy} onValueChange={setSortBy}>
   943→            <SelectTrigger className="w-[130px]">
   944→              <SelectValue />
   945→            </SelectTrigger>
   946→            <SelectContent>
   947→              <SelectItem value="name">Name</SelectItem>
   948→              <SelectItem value="createdAt">Created</SelectItem>
   949→              <SelectItem value="joinDate">Join Date</SelectItem>
   950→              <SelectItem value="role">Role</SelectItem>
   951→            </SelectContent>
   952→          </Select>
   953→
   954→          <Button
   955→            variant="outline"
   956→            size="icon"
   957→            onClick={() => setSortOrder((prev) => (prev === 'asc' ? 'desc' : 'asc'))}
   958→            title={`Sort ${sortOrder === 'asc' ? 'ascending' : 'descending'}`}
   959→          >
   960→            <ChevronLeft className={`h-4 w-4 transition-transform ${sortOrder === 'desc' ? 'rotate-180' : ''}`} />
   961→          </Button>
   962→        </div>
   963→
   964→        {/* Bulk actions bar */}
   965→        {selectedIds.size > 0 && (
   966→          <div className="flex items-center gap-3 rounded-lg border bg-muted/50 px-4 py-2.5">
   967→            <span className="text-sm font-medium">
   968→              {selectedIds.size} selected
   969→            </span>
   970→            <div className="flex-1" />
   971→            {isSuperAdmin && (
   972→              <Button
   973→                variant="destructive"
   974→                size="sm"
   975→                onClick={() => setBulkDeleteDialogOpen(true)}
   976→              >
   977→                <Trash2 className="mr-2 h-4 w-4" />
   978→                Delete Selected
   979→              </Button>
   980→            )}
   981→            <Button variant="outline" size="sm" onClick={() => setSelectedIds(new Set())}>
   982→              Clear
   983→            </Button>
   984→          </div>
   985→        )}
   986→
   987→        {/* Content */}
   988→        {loading ? (
   989→          <TeamSkeleton />
   990→        ) : members.length === 0 ? (
   991→          <EmptyState hasFilters={hasActiveFilters} onClearFilters={clearFilters} />
   992→        ) : (
   993→          <>
   994→            {/* Desktop Table */}
   995→            <div className="hidden md:block rounded-lg border">
   996→              <Table>
   997→                <TableHeader>
   998→                  <TableRow>
   999→                    <TableHead className="w-10">
  1000→                      <Checkbox
  1001→                        checked={selectedIds.size === members.length && members.length > 0}
  1002→                        onCheckedChange={toggleSelectAll}
  1003→                        aria-label="Select all members"
  1004→                      />
  1005→                    </TableHead>
  1006→                    <TableHead>Member</TableHead>
  1007→                    <TableHead>Role</TableHead>
  1008→                    <TableHead>Status</TableHead>
  1009→                    <TableHead>Projects</TableHead>
  1010→                    <TableHead>Skills</TableHead>
  1011→                    <TableHead className="w-12" />
  1012→                  </TableRow>
  1013→                </TableHeader>
  1014→                <TableBody>
  1015→                  {members.map((member) => (
  1016→                    <TableRow key={member.id} data-selected={selectedIds.has(member.id)}>
  1017→                      <TableCell>
  1018→                        <Checkbox
  1019→                          checked={selectedIds.has(member.id)}
  1020→                          onCheckedChange={() => toggleSelect(member.id)}
  1021→                          aria-label={`Select ${member.name}`}
  1022→                        />
  1023→                      </TableCell>
  1024→                      <TableCell>
  1025→                        <div className="flex items-center gap-3">
  1026→                          <Avatar className="h-8 w-8">
  1027→                            <AvatarFallback className="text-xs bg-primary/10 text-primary font-medium">
  1028→                              {getInitials(member.name)}
  1029→                            </AvatarFallback>
  1030→                          </Avatar>
  1031→                          <div className="min-w-0">
  1032→                            <p className="font-medium text-sm truncate max-w-[180px]">{member.name}</p>
  1033→                            <p className="text-xs text-muted-foreground truncate max-w-[180px]">{member.email}</p>
  1034→                            {member.jobTitle && (
  1035→                              <p className="text-xs text-muted-foreground truncate max-w-[180px]">{member.jobTitle}</p>
  1036→                            )}
  1037→                          </div>
  1038→                        </div>
  1039→                      </TableCell>
  1040→                      <TableCell>
  1041→                        <Badge className={`text-[10px] px-1.5 py-0 ${ROLE_STYLES[member.role] || ''}`} variant="outline">
  1042→                          {member.role}
  1043→                        </Badge>
  1044→                      </TableCell>
  1045→                      <TableCell>
  1046→                        <Badge className={`text-[10px] px-1.5 py-0 ${STATUS_STYLES[member.status] || ''}`} variant="outline">
  1047→                          {member.status}
  1048→                        </Badge>

  1050→                      <TableCell className="text-sm">{member.projectCount}</TableCell>
  1051→                      <TableCell>

  1058→                          {parseSkills(member.skills).length > 3 && (
  1059→                            <Tooltip>
