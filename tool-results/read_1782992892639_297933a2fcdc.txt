   407→/**
   408→ * Get the provider configuration (baseUrl + apiKey) for a given model.
   409→ *
   410→ * Lookup priority:
   411→ *   1. Provider-specific env var (e.g., GROQ_API_KEY)
   412→ *   2. Generic fallback (AI_API_KEY)
   413→ *   3. Provider's default base URL for baseUrl
   414→ */
   415→export function getProviderConfig(modelId: string): ProviderConfig {
   416→  const cap = MODEL_MAP[modelId];
   417→
   418→  if (!cap) {
   419→    // Unknown model — use generic config
   420→    const baseUrl = process.env.AI_API_BASE_URL || "https://api.openai.com/v1";
   421→    const apiKey = process.env.AI_API_KEY || "";
   422→    return { baseUrl, apiKey, provider: "unknown" };
   423→  }
   424→
   425→  // API key: provider-specific first, then generic fallback
   426→  const apiKey = process.env[cap.providerEnvKey] || process.env.AI_API_KEY || "";
   427→
   428→  // Base URL: provider-specific first, then generic fallback, then provider default
   429→  const baseUrl =
   430→    process.env[cap.providerEnvBaseUrl] ||
   431→    process.env.AI_API_BASE_URL ||
   432→    cap.defaultBaseUrl;
   433→
   434→  return { baseUrl, apiKey, provider: cap.provider };
   435→}

   446→/**
   447→ * Get the vision model to use.
   448→ * Priority:
   449→ *   1. AI_VISION_MODEL env var (explicit override)
   450→ *   2. Same-provider vision model (avoids cross-provider quota issues)
   451→ *   3. First configured vision-capable model
   452→ *   4. Fallback to gpt-4o-mini
   453→ */
   454→export function getVisionModel(preferredModel?: string): string {
   455→  // 1. If AI_VISION_MODEL is set, use it
   456→  if (process.env.AI_VISION_MODEL) return process.env.AI_VISION_MODEL;
   457→  // 2. If preferred model supports multimodal, use it directly
   458→  if (preferredModel && supportsMultimodal(preferredModel)) return preferredModel;
   459→  // 3. Try to find a vision model from the SAME provider as the preferred model
   460→  //    This avoids cross-provider issues (e.g., GLM user hitting Gemini quota)
   461→  if (preferredModel) {
   462→    const preferredCap = MODEL_MAP[preferredModel];
   463→    if (preferredCap) {
   464→      for (const m of MODEL_REGISTRY) {
   465→        if (m.provider === preferredCap.provider && m.supportsMultimodal) {
   466→          const config = getProviderConfig(m.id);
   467→          if (config.apiKey) return m.id;
   468→        }
   469→      }
   470→    }
   471→  }
   472→  // 4. Find any vision-capable model that has its provider configured
   473→  for (const m of MODEL_REGISTRY) {
   474→    if (m.supportsMultimodal) {
   475→      const config = getProviderConfig(m.id);
   476→      if (config.apiKey) return m.id;
   477→    }
   478→  }
   479→  // 5. Default to gpt-4o-mini (best cost/vision quality)
   480→  return "gpt-4o-mini";
   481→}

   522→// ===== Token Estimation =====
   523→
   524→/**
   525→ * Estimate the number of tokens in a text string.
   526→ * Rough heuristic: ~4 characters per token for English, ~1.5 for CJK.
   527→ */
   528→function estimateTextTokens(text: string): number {
   529→  if (!text) return 0;
   530→  let tokens = 0;
   531→  for (const char of text) {
   532→    const code = char.charCodeAt(0);
   533→    if (code > 0x2e7f) {
   534→      // CJK and other non-Latin: ~1.5 chars per token
   535→      tokens += 0.67;
   536→    } else {
   537→      // Latin: ~4 chars per token
   538→      tokens += 0.25;
   539→    }
   540→  }
   541→  return Math.ceil(tokens);
   542→}

   544→/**
   545→ * Estimate total prompt tokens for an array of AI messages.
   546→ */
   547→export function estimatePromptTokens(
   548→  messages: Array<{ role: string; content: string | unknown }>,
   549→  tools?: unknown[]
   550→): number {
   551→  let totalTokens = 0;
   552→
   553→  // Estimate message tokens
   554→  for (const msg of messages) {
   555→    totalTokens += 4; // overhead per message (role, formatting)
   556→    if (typeof msg.content === "string") {
   557→      totalTokens += estimateTextTokens(msg.content);
   558→    } else if (Array.isArray(msg.content)) {
   559→      // Multimodal content
   560→      for (const part of msg.content) {
   561→        if (typeof part === "object" && part !== null) {
   562→          if ("text" in part && typeof part.text === "string") {
   563→            totalTokens += estimateTextTokens(part.text);
   564→          }
   565→          if ("image_url" in part) {
   566→            totalTokens += 85; // low detail image ~85 tokens
   567→          }
   568→        }
   569→      }
   570→    }
   571→  }
   572→
   573→  // Estimate tool tokens (each tool definition adds ~100-300 tokens)
   574→  if (tools && tools.length > 0) {
   575→    for (const tool of tools) {
   576→      totalTokens += estimateTextTokens(JSON.stringify(tool));
   577→    }
   578→  }
   579→
   580→  return totalTokens;
   581→}

   583→// ===== Automatic Model Routing =====
   584→
   585→/**
   586→ * Routing result from findBestModelForPrompt().
   587→ */
   588→export interface ModelRouteResult {
   589→  model: string;
   590→  reason: string;
   591→  autoRouted: boolean;
   592→}
   593→
   594→/**
   595→ * Find the best available model for a given estimated prompt size.
   596→ *
   597→ * Logic:
   598→ *   1. If preferred model can handle it AND is configured -> use it
   599→ *   2. Search for the smallest configured model with enough context
   600→ *   3. Fallback to preferred model (let it fail with its own error message)
   601→ *
   602→ * @param estimatedTokens  - Estimated token count of the full prompt
   603→ * @param preferredModel   - The model the user/system originally selected
   604→ * @param requiredFeatures - Optional feature requirements (tools, vision)
   605→ */
   606→export function findBestModelForPrompt(
   607→  estimatedTokens: number,
   608→  preferredModel: string,
   609→  requiredFeatures?: { tools?: boolean; vision?: boolean }
   610→): ModelRouteResult {
   611→  const preferred = MODEL_MAP[preferredModel];
   612→
   613→  // 1. Check if preferred model can handle the prompt
   614→  if (preferred && isModelConfigured(preferredModel)) {
   615→    // Leave 20% headroom for the response tokens
   616→    const maxPromptTokens = Math.floor(preferred.contextWindowTokens * 0.8);
   617→    if (estimatedTokens <= maxPromptTokens) {
   618→      return {
   619→        model: preferredModel,
   620→        reason: "Preferred model has sufficient capacity",
   621→        autoRouted: false,
   622→      };
   623→    }
   624→    // Preferred model is too small — we need to find an alternative
   625→  }
   626→
   627→  // 2. Find the best alternative model
   628→  const candidates = MODEL_REGISTRY.filter((m) => {
   629→    // Must have API key configured
   630→    if (!isModelConfigured(m.id)) return false;
   631→    // Must support required features
   632→    if (requiredFeatures?.tools && !m.supportsTools) return false;
   633→    if (requiredFeatures?.vision && !m.supportsVision) return false;
   634→    // Must have enough context (80% of window reserved for prompt)
   635→    const maxPromptTokens = Math.floor(m.contextWindowTokens * 0.8);
   636→    return estimatedTokens <= maxPromptTokens;
   637→  });
   638→
   639→  if (candidates.length > 0) {
   640→    // Sort: prefer models that are "just big enough" (avoid wasteful 1M context for small prompts)
   641→    // but break ties by preferring well-known quality models
   642→    const qualityOrder: Record<string, number> = {
   643→      "DeepSeek-V3.1": 1,
   644→      "gpt-4o": 2,
   645→      "glm-4-flash": 3,
   646→      "Meta-Llama-3.3-70B-Instruct": 4,
   647→      "gpt-4o-mini": 5,
   648→      "llama-3.3-70b-versatile": 6,
   649→      "glm-4-plus": 7,
   650→      "glm-4-air": 8,
   651→      "Llama-4-Maverick-17B-128E-Instruct": 9,
   652→      "gemini-2.0-flash": 10,
   653→      "gemini-1.5-flash": 11,
   654→      "gemini-1.5-pro": 12,
   655→      "llama-3.1-8b-instant": 13,
   656→      "gpt-3.5-turbo": 14,
   657→    };
   658→
   659→    // For tool-calling (doc generation), prefer models with 16K+ output tokens
   660→    // For regular chat, prefer by quality then context fit
   661→    candidates.sort((a, b) => {
   662→      const aQ = qualityOrder[a.id] ?? 50;
   663→      const bQ = qualityOrder[b.id] ?? 50;
   664→      const aOut = a.maxOutputTokens;
   665→      const bOut = b.maxOutputTokens;
   666→
   667→      if (requiredFeatures?.tools) {
   668→        // For docs: strongly prefer 16K+ output models
   669→        if (aOut >= 16000 && bOut < 16000) return -1;
   670→        if (bOut >= 16000 && aOut < 16000) return 1;
   671→      }
   672→
   673→      // Both have similar output capacity — prefer by quality
   674→      if (Math.abs(a.contextWindowTokens - estimatedTokens) < estimatedTokens) {
   675→        return aQ - bQ;
   676→      }
   677→      // Otherwise prefer context closer to our needs
   678→      return a.contextWindowTokens - b.contextWindowTokens;
   679→    });
   680→
   681→    const best = candidates[0];
   682→    return {
   683→      model: best.id,
   684→      reason: `Preferred model "${preferredModel}" may not handle ~${estimatedTokens.toLocaleString()} tokens well. Auto-routed to "${best.name}" (${best.contextWindow} context, ${best.category})`,
   685→      autoRouted: true,
   686→    };
   687→  }
   688→
   689→  // 3. No suitable model found — return a helpful error message
   690→  const configuredProviders: string[] = [];
   691→  if (process.env.GROQ_API_KEY) configuredProviders.push("Groq");
   692→  if (process.env.OPENAI_API_KEY) configuredProviders.push("OpenAI");
   693→  if (process.env.GOOGLE_AI_API_KEY) configuredProviders.push("Google");
   694→  if (process.env.TOGETHER_API_KEY) configuredProviders.push("Together");
   695→  if (process.env.SAMBANOVA_API_KEY) configuredProviders.push("SambaNova");
   696→  if (process.env.OPENROUTER_API_KEY) configuredProviders.push("OpenRouter");
   697→  if (process.env.ZAI_API_KEY) configuredProviders.push("Z.ai (GLM)");
   698→  if (process.env.AI_API_KEY) configuredProviders.push("Generic (AI_API_KEY)");
   699→
   700→  const providerHint =
   701→    configuredProviders.length > 0
   702→      ? ` Configured providers: ${configuredProviders.join(", ")}.`
   703→      : " No AI providers are configured.";
   704→
   705→  return {
   706→    model: preferredModel,
   707→    reason: `No configured model can handle ~${estimatedTokens.toLocaleString()} tokens.${providerHint} Add ZAI_API_KEY for free GLM-4 (200K context), GOOGLE_AI_API_KEY for Gemini, or OPENAI_API_KEY for GPT-4o.`,
   708→    autoRouted: false,
   709→  };
   710→}
   711→
   712→/**
   713→ * Get a list of fallback models for a given model, ordered by quality and reliability.
   714→ * Used for retrying when a provider returns 429 (rate limit), 413 (too large), or any 4xx/5xx error.
   715→ * Excludes models from the same provider as the failed model.
   716→ *
   717→ * Ordering priority:
   718→ *   1. Quality (smarter models first — DeepSeek, GPT-4o, etc.)
   719→ *   2. Larger context windows (for handling big prompts)
   720→ *   3. NOT by context window alone (Google 1M models would always win but 429 due to low RPM)
   721→ */
   722→export function getFallbackModels(
   723→  failedModelId: string,
   724→  requiredFeatures?: { tools?: boolean; vision?: boolean }
   725→): string[] {
   726→  const failedCap = MODEL_MAP[failedModelId];
   727→  const failedProvider = failedCap?.provider;
   728→
   729→  // Quality-based priority (lower number = higher priority)
   730→  // SambaNova models are high priority (free, high RPM, good quality)
   731→  // OpenAI models are great fallback (paid, very reliable)
   732→  // Groq models are good (free, fast)
   733→  // Google models are lower priority despite 1M context (free tier only 15 RPM, frequent 429)
   734→  const fallbackQuality: Record<string, number> = {
   735→    "DeepSeek-V3.1": 1,
   736→    "glm-4-flash": 2,
   737→    "Meta-Llama-3.3-70B-Instruct": 3,
   738→    "gpt-4o": 4,
   739→    "Llama-4-Maverick-17B-128E-Instruct": 5,
   740→    "gpt-4o-mini": 6,
   741→    "llama-3.3-70b-versatile": 7,
   742→    "glm-4-air": 8,
   743→    "glm-4-plus": 9,
   744→    "llama-3.1-8b-instant": 10,
   745→    "gemini-2.0-flash": 11,
   746→    "gemini-1.5-flash": 12,
   747→    "gemini-1.5-pro": 13,
   748→    "gpt-3.5-turbo": 14,
   749→    // Together / older models — lowest priority
   750→  };
   751→
   752→  return MODEL_REGISTRY
   753→    .filter((m) => {
   754→      // Must have API key configured
   755→      if (!isModelConfigured(m.id)) return false;
   756→      // Skip same provider (they'll have same rate limits)
   757→      if (m.provider === failedProvider) return false;
   758→      // Must support required features
   759→      if (requiredFeatures?.tools && !m.supportsTools) return false;
   760→      if (requiredFeatures?.vision && !m.supportsVision) return false;
   761→      return true;
   762→    })
   763→    .sort((a, b) => {
   764→      // Primary: quality-based priority
   765→      const aQ = fallbackQuality[a.id] ?? 50;
   766→      const bQ = fallbackQuality[b.id] ?? 50;
   767→      if (aQ !== bQ) return aQ - bQ;
   768→      // Secondary: larger context window (for handling big prompts)
   769→      if (b.contextWindowTokens !== a.contextWindowTokens) {
   770→        return b.contextWindowTokens - a.contextWindowTokens;
   771→      }
   772→      return 0;
   773→    })
   774→    .map((m) => m.id);
   775→}
