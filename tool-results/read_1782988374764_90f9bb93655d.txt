   148→/** Ensure content is always a string — prevents React error #31 when API returns objects */
   149→function safeContent(content: unknown): string {
   150→  return typeof content === "string" ? content : content != null ? JSON.stringify(content) : "";
   151→}
   152→
   153→/** Ensure error is always a string — prevents React error #31 when API returns error objects like {code, id, message} */
   154→function safeError(err: unknown, fallback: string): string {
   155→  if (typeof err === "string") return err;
   156→  if (err && typeof err === "object") {
   157→    const obj = err as Record<string, unknown>;
   158→    if (typeof obj.message === "string") return obj.message;
   159→    try { return JSON.stringify(err); } catch { /* fall through */ }
   160→  }
   161→  return fallback;
   162→}
