    48→  DropdownMenuTrigger,
    49→} from "@/components/ui/dropdown-menu";

    51→
    52→const priorityConfig: Record<string, { label: string; className: string }> = {
    53→  HIGH: { label: "High", className: "bg-red-100 text-red-700 dark:bg-red-900/30 dark:text-red-400" },
    54→  MEDIUM: { label: "Medium", className: "bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-400" },
    55→  LOW: { label: "Low", className: "bg-green-100 text-green-700 dark:bg-green-900/30 dark:text-green-400" },
    56→};
    57→
    58→const statusConfig: Record<string, { label: string; className: string }> = {
    59→  ACTIVE: { label: "Active", className: "bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-400" },
    60→  ON_HOLD: { label: "On Hold", className: "bg-yellow-100 text-yellow-700 dark:bg-yellow-900/30 dark:text-yellow-400" },

    62→  ARCHIVED: { label: "Archived", className: "bg-gray-100 text-gray-600 dark:bg-gray-800/30 dark:text-gray-400" },
    63→};
    64→

    66→  LEAD: "Team Lead",
    67→  DEVELOPER: "Developer",
    68→  MARKETER: "Marketer",
    69→  VIEWER: "Viewer",
    70→  MEMBER: "Member",
    71→};
    72→

    74→  LEAD: "bg-purple-100 text-purple-700 dark:bg-purple-900/30 dark:text-purple-400",
    75→  DEVELOPER: "bg-sky-100 text-sky-700 dark:bg-sky-900/30 dark:text-sky-400",
    76→  MARKETER: "bg-orange-100 text-orange-700 dark:bg-orange-900/30 dark:text-orange-400",
    77→  VIEWER: "bg-gray-100 text-gray-600 dark:bg-gray-800/30 dark:text-gray-400",
    78→  MEMBER: "bg-indigo-100 text-indigo-700 dark:bg-indigo-900/30 dark:text-indigo-400",
    79→};

    81→interface ProjectDetail {
    82→  id: string;
    83→  name: string;
    84→  description: string | null;
    85→  status: string;
    86→  priority: string;
    87→  clientName: string | null;
    88→  color: string | null;
    89→  deadline: string | null;
    90→  createdAt: string;
    91→  updatedAt: string;
    92→  memberCount: number;
    93→}
    94→
    95→interface TeamMember {
    96→  id: string;
    97→  userId: string;
    98→  role: string;
    99→  joinedAt: string;
   100→  assignedBy: string | null;
   101→  assignedByName?: string;
   102→  user: {
   103→    name: string;
   104→    email: string;
   105→    avatar: string | null;
   106→    jobTitle: string | null;
   107→    status: string;
   108→  };
   109→}
   110→
   111→interface AvailableMember {
   112→  id: string;
   113→  name: string;
   114→  email: string;
   115→  role: string; // global role
   116→  jobTitle: string | null;
   117→}
   118→
   119→export default function ProjectDetailPage() {
   120→  const { data: session } = useSession();
   121→  const router = useRouter();
   122→  const params = useParams();
   123→  const projectId = params.id as string;
   124→  const user = session?.user as { role?: string; id?: string } | undefined;
   125→  const isAdmin = user?.role === "ADMIN" || user?.role === "SUPERADMIN";
   126→  const isSuperAdmin = user?.role === "SUPERADMIN";
   127→
   128→  const [project, setProject] = useState<ProjectDetail | null>(null);
   129→  const [team, setTeam] = useState<TeamMember[]>([]);
   130→  const [loading, setLoading] = useState(true);
   131→
   132→  // Edit project dialog
   133→  const [editOpen, setEditOpen] = useState(false);
   134→  const [editLoading, setEditLoading] = useState(false);
   135→  const [editForm, setEditForm] = useState({
   136→    name: "",
   137→    description: "",
   138→    priority: "MEDIUM",
   139→    status: "ACTIVE",
   140→    clientName: "",
   141→    color: "#6366f1",
   142→    deadline: "",
   143→  });
   144→
   145→  // Add member dialog
   146→  const [addMemberOpen, setAddMemberOpen] = useState(false);
   147→  const [addMemberLoading, setAddMemberLoading] = useState(false);
   148→  const [availableMembers, setAvailableMembers] = useState<AvailableMember[]>([]);
   149→  const [selectedUserId, setSelectedUserId] = useState("");
   150→  const [selectedRole, setSelectedRole] = useState("MEMBER");
   151→  const [loadingMembers, setLoadingMembers] = useState(false);
   152→
   153→  const fetchProject = useCallback(async () => {
   154→    try {
   155→      const res = await fetch(`/api/projects/${projectId}`);
   156→      const data = await res.json();
   157→      if (data.success) {
   158→        setProject(data.data.project);
   159→      } else {
   160→        errorToast({ error: data.error, title: "Failed to load project" });
   161→        router.push("/dashboard/projects");
   162→      }
   163→    } catch {
   164→      errorToast({ error: "Failed to load project" });
   165→      router.push("/dashboard/projects");
   166→    } finally {
   167→      setLoading(false);
   168→    }
   169→  }, [projectId, router]);
   170→
   171→  const fetchTeam = useCallback(async () => {
   172→    try {
   173→      const res = await fetch(`/api/projects/${projectId}/team`);
   174→      const data = await res.json();
   175→      if (data.success) {
   176→        setTeam(data.data.members);
   177→      }
   178→    } catch {
   179→      // Silently fail — team data is secondary
   180→    }
   181→  }, [projectId]);
   182→
   183→  useEffect(() => {
   184→    setLoading(true);
   185→    fetchProject();
   186→    fetchTeam();
   187→  }, [fetchProject, fetchTeam]);
   188→
   189→  const handleEditSave = async () => {
   190→    if (!editForm.name.trim()) {
   191→      toast.error("Project name is required");
   192→      return;
   193→    }
   194→    setEditLoading(true);
   195→    try {
   196→      const body: Record<string, unknown> = {
   197→        name: editForm.name,
   198→        description: editForm.description || null,
   199→        priority: editForm.priority,
   200→        status: editForm.status,
   201→        clientName: editForm.clientName || null,
   202→        color: editForm.color,
   203→        deadline: editForm.deadline ? new Date(editForm.deadline).toISOString() : null,
   204→      };
   205→
   206→      const res = await fetch(`/api/projects/${projectId}`, {
   207→        method: "PATCH",
   208→        headers: { "Content-Type": "application/json" },
   209→        body: JSON.stringify(body),
   210→      });
   211→      const data = await res.json();
   212→      if (data.success) {
   213→        toast.success("Project updated");
   214→        setEditOpen(false);
   215→        fetchProject();
   216→      } else {
   217→        errorToast({ error: data.error, title: "Failed to update project" });
   218→      }
   219→    } catch {
   220→      errorToast({ error: "Failed to update project" });
   221→    } finally {
   222→      setEditLoading(false);
   223→    }
   224→  };
   225→
   226→  const openEditDialog = () => {
   227→    if (!project) return;
   228→    setEditForm({
   229→      name: project.name,
   230→      description: project.description || "",
   231→      priority: project.priority,
   232→      status: project.status,
   233→      clientName: project.clientName || "",
   234→      color: project.color || "#6366f1",
   235→      deadline: project.deadline ? project.deadline.split("T")[0] : "",
   236→    });
   237→    setEditOpen(true);
   238→  };
   239→
   240→  const openAddMemberDialog = async () => {
   241→    setAddMemberOpen(true);
   242→    setSelectedUserId("");
   243→    setSelectedRole("MEMBER");
   244→    setLoadingMembers(true);
   245→    try {
   246→      // Fetch all active members (not deleted)
   247→      const res = await fetch("/api/members?limit=200&status=ACTIVE");
   248→      const data = await res.json();
   249→      if (data.success) {
   250→        // Filter out members already in this project
   251→        const existingIds = new Set(team.map((m) => m.userId));
   252→        const available = data.data.members.filter(
   253→          (m: AvailableMember) => !existingIds.has(m.id)
   254→        );
   255→        setAvailableMembers(available);
   256→      }
   257→    } catch {
   258→      errorToast({ error: "Failed to load available members" });
   259→    } finally {
   260→      setLoadingMembers(false);
   261→    }
   262→  };
   263→
   264→  const handleAddMember = async () => {
   265→    if (!selectedUserId) {
   266→      toast.error("Please select a team member");
   267→      return;
   268→    }
   269→    setAddMemberLoading(true);
   270→    try {
   271→      const res = await fetch(`/api/projects/${projectId}/team`, {
   272→        method: "POST",
   273→        headers: { "Content-Type": "application/json" },
   274→        body: JSON.stringify({ userId: selectedUserId, role: selectedRole }),
   275→      });
   276→      const data = await res.json();
   277→      if (data.success) {
   278→        toast.success("Team member added");
   279→        setAddMemberOpen(false);
   280→        fetchTeam();
   281→        fetchProject(); // Update member count
   282→      } else {
   283→        errorToast({ error: data.error, title: "Failed to add member" });
   284→      }
   285→    } catch {
   286→      errorToast({ error: "Failed to add member" });
   287→    } finally {
   288→      setAddMemberLoading(false);
   289→    }
   290→  };
   291→
   292→  const handleRemoveMember = async (member: TeamMember) => {
   293→    if (member.role === "LEAD" && !isSuperAdmin) {
   294→      toast.error("Only super admin can remove a team lead");
   295→      return;
   296→    }
   297→    if (!confirm(`Remove ${member.user.name} from this project?`)) return;
   298→    try {
   299→      const res = await fetch(`/api/projects/${projectId}/team`, {
   300→        method: "DELETE",
   301→        headers: { "Content-Type": "application/json" },
   302→        body: JSON.stringify({ userId: member.userId }),
   303→      });
   304→      const data = await res.json();
   305→      if (data.success) {
   306→        toast.success("Member removed");
   307→        fetchTeam();
   308→        fetchProject();
   309→      } else {
   310→        errorToast({ error: data.error, title: "Failed to remove member" });
   311→      }
   312→    } catch {
   313→      errorToast({ error: "Failed to remove member" });
   314→    }
   315→  };
   316→
   317→  const handleChangeRole = async (member: TeamMember, newRole: string) => {
   318→    // Multiple LEADs allowed — no need to check for existing lead
   319→    // Only ADMIN+ global role can be LEAD in a project
   320→    try {
   321→      const memberRes = await fetch(`/api/members/${member.userId}`);
   322→      const memberData = await memberRes.json();
   323→      if (memberData.success && newRole === "LEAD") {
   324→        const globalRole = memberData.data.role;
   325→        if (globalRole !== "ADMIN" && globalRole !== "SUPERADMIN") {
   326→          toast.error("Only admins can be assigned as Team Lead");
   327→          return;
   328→        }
   329→      }
   330→    } catch {
   331→      // If fetch fails, let the backend handle the validation
   332→    }
   333→
   334→    try {
   335→      const res = await fetch(`/api/projects/${projectId}/team/${member.userId}/role`, {
   336→        method: "PATCH",
   337→        headers: { "Content-Type": "application/json" },
   338→        body: JSON.stringify({ role: newRole }),
   339→      });
   340→      const data = await res.json();
   341→      if (data.success) {
   342→        toast.success("Role updated");
   343→        fetchTeam();
   344→        fetchProject(); // Refresh member count in header
   345→      } else {
   346→        errorToast({ error: data.error, title: "Failed to change role" });
   347→      }
   348→    } catch {
   349→      errorToast({ error: "Failed to change role" });
   350→    }
   351→  };
   352→
   353→  const handleArchive = async () => {
   354→    if (!project || !confirm(`Archive "${project.name}"?`)) return;
   355→    try {
   356→      const res = await fetch(`/api/projects/${projectId}`, { method: "DELETE" });
   357→      const data = await res.json();
   358→      if (data.success) {
   359→        toast.success("Project archived");
   360→        router.push("/dashboard/projects");
   361→      } else {
   362→        errorToast({ error: data.error, title: "Failed to archive project" });
   363→      }
   364→    } catch {
   365→      errorToast({ error: "Failed to archive project" });
   366→    }
   367→  };
   368→
   369→  function formatDate(dateStr: string | null) {
   370→    if (!dateStr) return "—";
   371→    try {
   372→      return new Date(dateStr).toLocaleDateString("en-GB", {
   373→        day: "numeric",
   374→        month: "short",
   375→        year: "numeric",
   376→      });
   377→    } catch {
   378→      return dateStr;
   379→    }
   380→  }
   381→
   382→  function isOverdue() {
   383→    if (!project?.deadline || project.status === "COMPLETED" || project.status === "ARCHIVED") return false;
   384→    return new Date(project.deadline) < new Date();
   385→  }
   386→
   387→  if (loading) {
   388→    return (
   389→      <div className="flex items-center justify-center py-20">
   390→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   391→      </div>
   392→    );
   393→  }
   394→
   395→  if (!project) {
   396→    return (
   397→      <div className="flex flex-col items-center justify-center py-20">
   398→        <p className="text-muted-foreground">Project not found</p>
   399→        <Button variant="link" onClick={() => router.push("/dashboard/projects")}>
   400→          Back to Projects
   401→        </Button>
   402→      </div>
   403→    );
   404→  }
   405→
   406→  const overdue = isOverdue();
   407→  const priority = priorityConfig[project.priority] || priorityConfig.MEDIUM;
   408→  const status = statusConfig[project.status] || statusConfig.ACTIVE;
   409→  const leads = team.filter((m) => m.role === "LEAD");
   410→  const otherMembers = team.filter((m) => m.role !== "LEAD");
   411→
   412→  const handlePromoteToLead = async (member: TeamMember) => {
   413→    // Multiple leads allowed — just promote directly
   414→    await handleChangeRole(member, "LEAD");
   415→  };
   416→
   417→  return (
   418→    <div className="space-y-6">
   419→      {/* Header */}
   420→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   421→        <div className="flex items-center gap-3">
   422→          <Button variant="ghost" size="icon" onClick={() => router.push("/dashboard/projects")}>
   423→            <ArrowLeft className="h-4 w-4" />
   424→          </Button>
   425→          <div className="flex items-center gap-2">
   426→            <div
   427→              className="h-3 w-3 rounded-full shrink-0"
   428→              style={{ backgroundColor: project.color || "#6366f1" }}
   429→            />
   430→            <h1 className="text-2xl font-bold tracking-tight">{project.name}</h1>
   431→          </div>
   432→        </div>
   433→        {isAdmin && (
   434→          <div className="flex items-center gap-2">
   435→            <Button variant="outline" size="sm" onClick={openEditDialog}>
   436→              <Pencil className="mr-2 h-4 w-4" />
   437→              Edit
   438→            </Button>
   439→            {isSuperAdmin && (
   440→              <Button variant="outline" size="sm" className="text-destructive" onClick={handleArchive}>
   441→                <Trash2 className="mr-2 h-4 w-4" />
   442→                Archive
   443→              </Button>
   444→            )}
   445→          </div>
   446→        )}
   447→      </div>
   448→
   449→      {/* Project Info Cards */}
   450→      <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
   451→        <Card>
   452→          <CardContent className="p-4">
   453→            <p className="text-xs text-muted-foreground mb-1">Status</p>
   454→            <Badge variant="secondary" className={status.className}>
   455→              {status.label}
   456→            </Badge>
   457→          </CardContent>
   458→        </Card>
   459→        <Card>
   460→          <CardContent className="p-4">
   461→            <p className="text-xs text-muted-foreground mb-1">Priority</p>
   462→            <Badge variant="secondary" className={priority.className}>
   463→              {priority.label}
   464→            </Badge>
   465→          </CardContent>
   466→        </Card>
   467→        <Card>
   468→          <CardContent className="p-4">
   469→            <p className="text-xs text-muted-foreground mb-1">Deadline</p>
   470→            <p className={`text-sm font-medium ${overdue ? "text-destructive" : ""}`}>
   471→              {formatDate(project.deadline)}
   472→            </p>
   473→          </CardContent>
   474→        </Card>
   475→        <Card>
   476→          <CardContent className="p-4">
   477→            <p className="text-xs text-muted-foreground mb-1">Team Size</p>
   478→            <p className="text-sm font-medium">{project.memberCount} member{project.memberCount !== 1 ? "s" : ""}</p>
   479→          </CardContent>
   480→        </Card>
   481→      </div>
   482→
   483→      {/* Description + Client Info */}
   484→      <Card>
   485→        <CardContent className="p-5">
   486→          <div className="grid gap-4 sm:grid-cols-2">
   487→            <div>
   488→              <p className="text-xs text-muted-foreground mb-1 font-medium uppercase tracking-wide">Description</p>
   489→              <p className="text-sm">{project.description || "No description provided"}</p>
   490→            </div>
   491→            <div className="space-y-3">
   492→              {project.clientName && (
   493→                <div>
   494→                  <p className="text-xs text-muted-foreground mb-1 font-medium uppercase tracking-wide">Client</p>
   495→                  <p className="text-sm">{project.clientName}</p>
   496→                </div>
   497→              )}
   498→              <div>
   499→                <p className="text-xs text-muted-foreground mb-1 font-medium uppercase tracking-wide">Created</p>
   500→                <p className="text-sm">{formatDate(project.createdAt)}</p>
   501→              </div>
   502→              <div>
   503→                <p className="text-xs text-muted-foreground mb-1 font-medium uppercase tracking-wide">Last Updated</p>
   504→                <p className="text-sm">{formatDate(project.updatedAt)}</p>
   505→              </div>
   506→            </div>
   507→          </div>
   508→        </CardContent>
   509→      </Card>
   510→
   511→      {/* Team Management */}
   512→      <div>
   513→        <div className="flex items-center justify-between mb-4">
   514→          <h2 className="text-lg font-semibold flex items-center gap-2">
   515→            <Users className="h-5 w-5" />
   516→            Project Team
   517→          </h2>
   518→          {isAdmin && (
   519→            <Button size="sm" onClick={openAddMemberDialog}>
   520→              <UserPlus className="mr-2 h-4 w-4" />
   521→              Add Member
   522→            </Button>
   523→          )}
   524→          {isSuperAdmin && (
   525→            <p className="text-xs text-muted-foreground">
   526→              Super admin is auto-assigned to all projects.
   527→            </p>
   528→          )}
   529→        </div>
   530→
   531→        {/* Team Leads */}
   532→        <Card className="mb-4 border-purple-200 dark:border-purple-800/40">
   533→          <CardHeader className="pb-3">
   534→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   535→              <Crown className="h-4 w-4 text-purple-600" />
   536→              Team Leads
   537→            </CardTitle>
   538→          </CardHeader>
   539→          <CardContent className="pt-0">
   540→            {leads.length > 0 ? (
   541→              <div className="space-y-3">
   542→                {leads.map((leadMember) => (
   543→                  <div key={leadMember.id} className="flex items-center justify-between">
   544→                    <div className="flex items-center gap-3">
   545→                      <Avatar className="h-9 w-9">
   546→                        <AvatarFallback className="text-xs bg-purple-100 text-purple-700 dark:bg-purple-900/30 dark:text-purple-400">
   547→                          {leadMember.user.name
   548→                            .split(" ")
   549→                            .map((n) => n[0])
   550→                            .join("")
   551→                            .toUpperCase()
   552→                            .slice(0, 2)}
   553→                        </AvatarFallback>
   554→                      </Avatar>
   555→                      <div>
   556→                        <p className="text-sm font-medium">{leadMember.user.name}</p>
   557→                        <p className="text-xs text-muted-foreground">{leadMember.user.email}</p>
   558→                      </div>
   559→                    </div>
   560→                    {isSuperAdmin && (
   561→                      <DropdownMenu>
   562→                        <DropdownMenuTrigger asChild>
   563→                          <Button variant="ghost" size="icon" className="h-8 w-8">
   564→                            <MoreHorizontal className="h-4 w-4" />
   565→                          </Button>
   566→                        </DropdownMenuTrigger>
   567→                        <DropdownMenuContent align="end">
   568→                          <DropdownMenuItem
   569→                            onClick={() => handleChangeRole(leadMember, "MEMBER")}
   570→                          >
   571→                            <Shield className="mr-2 h-4 w-4" />
   572→                            Demote to Member
   573→                          </DropdownMenuItem>
   574→                          <DropdownMenuItem
   575→                            onClick={() => handleChangeRole(leadMember, "DEVELOPER")}
   576→                          >
   577→                            Change to Developer

   606→
   607→        {/* Team Members */}
   608→        {otherMembers.length > 0 ? (

   693→                          } else {

   698→                        <SelectTrigger className="h-6 w-auto min-w-[100px] text-[10px] border-dashed">
   699→                          <SelectValue />
   700→                        </SelectTrigger>
   701→                        <SelectContent>
   702→                          <SelectItem value="LEAD">Team Lead</SelectItem>
   703→                          <SelectItem value="DEVELOPER">Developer</SelectItem>
   704→                          <SelectItem value="MARKETER">Marketer</SelectItem>
   705→                          <SelectItem value="VIEWER">Viewer</SelectItem>
   706→                          <SelectItem value="MEMBER">Member</SelectItem>
   707→                        </SelectContent>
   708→                      </Select>

   711→                </CardContent>

   719→              <Users className="h-8 w-8 text-muted-foreground mb-2" />

   721→                {leads.length > 0 ? "No other team members" : "No team members yet"}

   781→                  <SelectTrigger id="edit-status">
   782→                    <SelectValue />
   783→                  </SelectTrigger>
   784→                  <SelectContent>
   785→                    <SelectItem value="ACTIVE">Active</SelectItem>
   786→                    <SelectItem value="ON_HOLD">On Hold</SelectItem>
   787→                    <SelectItem value="COMPLETED">Completed</SelectItem>
   788→                    <SelectItem value="ARCHIVED">Archived</SelectItem>
   789→                  </SelectContent>
   790→                </Select>
   791→              </div>
   792→            </div>
   793→            <div className="grid grid-cols-2 gap-4">

   795→                <Label htmlFor="edit-client">Client Name</Label>

   797→                  id="edit-client"
   798→                  value={editForm.clientName}
   799→                  onChange={(e) => setEditForm({ ...editForm, clientName: e.target.value })}

   809→                />

   811→            </div>
   812→            <div className="grid gap-2">
   813→              <Label htmlFor="edit-color">Color</Label>
   814→              <div className="flex items-center gap-2">
   815→                <Input
   816→                  id="edit-color"
   817→                  type="color"
   818→                  value={editForm.color}

   820→                  className="w-10 h-9 p-1 cursor-pointer"
   821→                />
   822→                <Input

   834→            <Button onClick={handleEditSave} disabled={editLoading || !editForm.name.trim()}>

   848→      {/* Add Member Dialog */}
   849→      <Dialog open={addMemberOpen} onOpenChange={setAddMemberOpen}>

   856→              <Label htmlFor="select-member">Select Member *</Label>

   900→                  <SelectValue />
   901→                </SelectTrigger>
   902→                <SelectContent>
   903→                  {/* LEAD only available for SUPERADMIN, and only if selected user is ADMIN/SUPERADMIN */}
   904→                  {(() => {
   905→                    const selected = availableMembers.find((m) => m.id === selectedUserId);
   906→                    const canBeLead = isSuperAdmin && selected && (selected.role === "ADMIN" || selected.role === "SUPERADMIN");

   915→                        <SelectItem value="DEVELOPER">{projectRoleLabels.DEVELOPER}</SelectItem>
   916→                        <SelectItem value="MARKETER">{projectRoleLabels.MARKETER}</SelectItem>
   917→                        <SelectItem value="VIEWER">{projectRoleLabels.VIEWER}</SelectItem>
   918→                        <SelectItem value="MEMBER">{projectRoleLabels.MEMBER}</SelectItem>
   919→                      </>

   921→                  })()}

   924→              <p className="text-xs text-muted-foreground">
   925→                Multiple team leads are allowed. Leads must be admins.

   928→          </div>
   929→          <DialogFooter>
   930→            <Button variant="outline" onClick={() => setAddMemberOpen(false)}>

   934→              onClick={handleAddMember}
   935→              disabled={addMemberLoading || !selectedUserId}
   936→            >

   938→                <>
   939→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   940→                  Adding...
   941→                </>
   942→              ) : (
   943→                "Add to Team"
   944→              )}
   945→            </Button>
