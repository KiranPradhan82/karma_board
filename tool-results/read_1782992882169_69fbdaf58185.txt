   306→  // Logo upload handler
   307→  async function handleLogoUpload(e: React.ChangeEvent<HTMLInputElement>) {
   308→    const file = e.target.files?.[0];
   309→    if (!file) return;
   310→    if (file.size > 500 * 1024) {
   311→      errorToast({ error: "File too large. Maximum size is 500 KB.", title: "Upload failed" });
   312→      return;
   313→    }
   314→    setLogoSaving(true);
   315→    try {
   316→      const dataUrl = await new Promise<string>((resolve, reject) => {
   317→        const reader = new FileReader();
   318→        reader.onload = () => resolve(reader.result as string);
   319→        reader.onerror = reject;
   320→        reader.readAsDataURL(file);
   321→      });
   322→      const res = await fetch("/api/settings", {
   323→        method: "PUT",
   324→        headers: { "Content-Type": "application/json" },
   325→        body: JSON.stringify({ settings: { BRANDING_LOGO: dataUrl } }),
   326→      });
   327→      const json = await res.json();
   328→      if (json.success) {
   329→        setBrandingLogo(dataUrl);
   330→        toast.success("Logo updated successfully");
   331→      } else {
   332→        errorToast({ error: json.error, title: "Failed to save logo" });
   333→      }
   334→    } catch (err) {
   335→      errorToast({ error: err, title: "Logo upload failed" });
   336→    } finally {
   337→      setLogoSaving(false);
   338→      // Reset input so same file can be re-selected
   339→      if (logoInputRef.current) logoInputRef.current.value = "";
   340→    }
   341→  }

   487→  // Save settings
   488→  async function handleSave() {
   489→    setSaving(true);
   490→    try {
   491→      const updateSettings: Record<string, string> = {};
   492→
   493→      if (emailProvider !== (settings.EMAIL_PROVIDER?.value || "")) {
   494→        updateSettings.EMAIL_PROVIDER = emailProvider;
   495→      }
   496→      if (resendKeyChanged && resendApiKey) {
   497→        updateSettings.RESEND_API_KEY = resendApiKey;
   498→      }
   499→      if (fromEmail !== (settings.RESEND_FROM_EMAIL?.value || "")) {
   500→        updateSettings.RESEND_FROM_EMAIL = fromEmail;
   501→      }
   502→      if (fromName !== (settings.RESEND_FROM_NAME?.value || "")) {
   503→        updateSettings.RESEND_FROM_NAME = fromName;
   504→      }
   505→      if (smtpUser !== (settings.SMTP_USER?.value || "")) {
   506→        updateSettings.SMTP_USER = smtpUser;
   507→      }
   508→      if (smtpPassChanged && smtpPassword) {
   509→        updateSettings.SMTP_PASSWORD = smtpPassword;
   510→      }
   511→
   512→      if (Object.keys(updateSettings).length === 0) {
   513→        toast.info("No changes to save");
   514→        setSaving(false);
   515→        return;
   516→      }
   517→
   518→      const res = await fetch("/api/settings", {
