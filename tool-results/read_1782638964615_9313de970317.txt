     1→     1→import type { Resend as ResendType } from "resend";
     2→     2→import { decrypt } from "@/lib/encryption";
     3→     3→import { getTursoClient } from "@/lib/api-auth";
     4→     4→
     5→     5→type EmailProvider = "resend" | "gmail-smtp";
     6→     6→
     7→     7→interface EmailConfig {
     8→     8→  provider: EmailProvider;
     9→     9→  // Resend
    10→    10→  resendApiKey: string | null;
    11→    11→  resendFromEmail: string;
    12→    12→  // Gmail SMTP
    13→    13→  smtpUser: string | null; // your@gmail.com
    14→    14→  smtpPass: string | null; // app password
    15→    15→  // Shared
    16→    16→  fromName: string;
    17→    17→}
    18→    18→
    19→    19→/**
    20→    20→ * Get email configuration — reads from Settings DB first, falls back to env vars.
    21→    21→ * Provider is determined by EMAIL_PROVIDER env var (or DB setting), defaults to "gmail-smtp".
    22→    22→ */
    23→    23→async function getEmailConfig(): Promise<EmailConfig> {
    24→    24→  const defaultProvider: EmailProvider =
    25→    25→    (process.env.EMAIL_PROVIDER as EmailProvider) || "gmail-smtp";
    26→    26→
    27→    27→  try {
    28→    28→    const tursoUrl = process.env.TURSO_DATABASE_URL;
    29→    29→    const tursoToken = process.env.TURSO_AUTH_TOKEN;
    30→    30→
    31→    31→    if (tursoUrl && tursoToken) {
    32→    32→      const client = getTursoClient();
    33→    33→
    34→    34→      // Fetch all email-related settings from DB (now includes SMTP_USER)
    35→    35→      const result = await client.execute({
    36→    36→        sql: `SELECT key, value FROM "Settings" WHERE key IN (?, ?, ?, ?, ?, ?)`,
    37→    37→        args: [
    38→    38→          "EMAIL_PROVIDER",
    39→    39→          "RESEND_API_KEY",
    40→    40→          "RESEND_FROM_EMAIL",
    41→    41→          "EMAIL_FROM_NAME",
    42→    42→          "SMTP_USER",
    43→    43→          "SMTP_PASSWORD",
    44→    44→        ],
    45→    45→      });
    46→    46→
    47→    47→      let provider = defaultProvider;
    48→    48→      let resendApiKey: string | null = null;
    49→    49→      let resendFromEmail = "";
    50→    50→      let fromName = process.env.EMAIL_FROM_NAME || "KarmaBoard";
    51→    51→      let smtpUser: string | null = null;
    52→    52→      let smtpPass: string | null = null;
    53→    53→
    54→    54→      for (const row of result.rows) {
    55→    55→        const key = row.key as string;
    56→    56→        const rawValue = row.value as string;
    57→    57→
    58→    58→        switch (key) {
    59→    59→          case "EMAIL_PROVIDER":
    60→    60→            if (rawValue === "resend" || rawValue === "gmail-smtp") {
    61→    61→              provider = rawValue;
    62→    62→            }
    63→    63→            break;
    64→    64→          case "RESEND_API_KEY":
    65→    65→            try {
    66→    66→              resendApiKey = decrypt(rawValue);
    67→    67→            } catch {
    68→    68→              resendApiKey = null;
    69→    69→            }
    70→    70→            break;
    71→    71→          case "RESEND_FROM_EMAIL":
    72→    72→            resendFromEmail = rawValue;
    73→    73→            break;
    74→    74→          case "EMAIL_FROM_NAME":
    75→    75→            fromName = rawValue || fromName;
    76→    76→            break;
    77→    77→          case "RESEND_FROM_NAME": // legacy key name compatibility
    78→    78→            fromName = rawValue || fromName;
    79→    79→            break;
    80→    80→          case "SMTP_USER":
    81→    81→            smtpUser = rawValue;
    82→    82→            break;
    83→    83→          case "SMTP_PASSWORD":
    84→    84→            try {
    85→    85→              smtpPass = decrypt(rawValue);
    86→    86→            } catch {
    87→    87→              smtpPass = null;
    88→    88→            }
    89→    89→            break;
    90→    90→        }
    91→    91→      }
    92→    92→
    93→    93→      // Fall back to env vars if DB values are empty
    94→    94→      if (!resendApiKey && process.env.RESEND_API_KEY) {
    95→    95→        resendApiKey = process.env.RESEND_API_KEY;
    96→    96→      }
    97→    97→      if (!resendFromEmail && process.env.RESEND_FROM_EMAIL) {
    98→    98→        resendFromEmail = process.env.RESEND_FROM_EMAIL;
    99→    99→      }
   100→   100→      if (!smtpUser && process.env.SMTP_USER) {
   101→   101→        smtpUser = process.env.SMTP_USER;
   102→   102→      }
   103→   103→      if (!smtpPass && process.env.SMTP_PASSWORD) {
   104→   104→        smtpPass = process.env.SMTP_PASSWORD;
   105→   105→      }
   106→   106→
   107→   107→      console.log(`[email] Config loaded from DB+env. Provider: ${provider}, SMTP user: ${smtpUser ? "SET" : "NOT SET"}, SMTP pass: ${smtpPass ? "SET" : "NOT SET"}`);
   108→   108→
   109→   109→      return {
   110→   110→        provider,
   111→   111→        resendApiKey,
   112→   112→        resendFromEmail,
   113→   113→        smtpUser,
   114→   114→        smtpPass,
   115→   115→        fromName,
   116→   116→      };
   117→   117→    }
   118→   118→  } catch (error) {
   119→   119→    console.error("[email] Error reading settings from DB:", error);
   120→   120→  }
   121→   121→
   122→   122→  // Final fallback to env vars only
   123→   123→  const smtpUser = process.env.SMTP_USER || null;
   124→   124→  const smtpPass = process.env.SMTP_PASSWORD || null;
   125→   125→
   126→   126→  console.log(`[email] Config from env vars only. Provider: ${defaultProvider}, SMTP user: ${smtpUser ? "SET" : "NOT SET"}, SMTP pass: ${smtpPass ? "SET" : "NOT SET"}`);
   127→   127→
   128→   128→  return {
   129→   129→    provider: defaultProvider,
   130→   130→    resendApiKey: process.env.RESEND_API_KEY || null,
   131→   131→    resendFromEmail: process.env.RESEND_FROM_EMAIL || "",
   132→   132→    smtpUser,
   133→   133→    smtpPass,
   134→   134→    fromName: process.env.EMAIL_FROM_NAME || "KarmaBoard",
   135→   135→  };
   136→   136→}

   140→   140→async function getResendClient(apiKey: string): Promise<ResendType> {
   141→   141→  const { Resend } = await import("resend");
   142→   142→  return new Resend(apiKey);

   144→   144→
   145→   145→async function sendViaResend(
   146→   146→  config: EmailConfig,
   147→   147→  to: string,
   148→   148→  subject: string,
   149→   149→  html: string
   150→   150→): Promise<{ success: boolean; error?: string }> {
   151→   151→  if (!config.resendApiKey) {
   152→   152→    return { success: false, error: "Resend API key not configured" };
   153→   153→  }
   154→   154→  if (!config.resendFromEmail) {
   155→   155→    return { success: false, error: "Resend from email not configured" };
   156→   156→  }
   157→   157→
   158→   158→  const resend = await getResendClient(config.resendApiKey);
   159→   159→
   160→   160→  const { data, error } = await resend.emails.send({
   161→   161→    from: `${config.fromName} <${config.resendFromEmail}>`,
   162→   162→    to: [to],
   163→   163→    subject,
   164→   164→    html,
   165→   165→  });
   166→   166→
   167→   167→  if (error) {
   168→   168→    return { success: false, error: error.message };
   169→   169→  }
   170→   170→
   171→   171→  console.log(`[email] Resend email sent to ${to}, emailId: ${data?.id}`);
   172→   172→  return { success: true };
   173→   173→}

   175→   175→// ===== Gmail SMTP Transport =====
   176→   176→
   177→   177→async function sendViaGmailSmtp(
   178→   178→  config: EmailConfig,
   179→   179→  to: string,
   180→   180→  subject: string,
   181→   181→  html: string
   182→   182→): Promise<{ success: boolean; error?: string }> {
   183→   183→  if (!config.smtpUser || !config.smtpPass) {
   184→   184→    return {
   185→   185→      success: false,
   186→   186→      error: `Gmail SMTP not configured (SMTP_USER: ${config.smtpUser ? "SET" : "NOT SET"}, SMTP_PASSWORD: ${config.smtpPass ? "SET" : "NOT SET"})`,
   187→   187→    };
   188→   188→  }
   189→   189→
   190→   190→  const nodemailer = await import("nodemailer");
   191→   191→
   192→   192→  const transporter = nodemailer.createTransport({
   193→   193→    service: "gmail",
   194→   194→    auth: {
   195→   195→      user: config.smtpUser,
   196→   196→      pass: config.smtpPass,
   197→   197→    },
   198→   198→  });
   199→   199→
   200→   200→  try {
   201→   201→    const info = await transporter.sendMail({
   202→   202→      from: `"${config.fromName}" <${config.smtpUser}>`,
   203→   203→      to,
   204→   204→      subject,
   205→   205→      html,
   206→   206→    });
   207→   207→
   208→   208→    console.log(`[email] Gmail SMTP email sent to ${to}, messageId: ${info.messageId}`);
   209→   209→    return { success: true };
   210→   210→  } catch (error) {
   211→   211→    const msg = error instanceof Error ? error.message : String(error);
   212→   212→    console.error("[email] Gmail SMTP error:", msg);
   213→   213→    return { success: false, error: msg };
   214→   214→  }
   215→   215→}
