   522→   522→  firstName: string,
   523→   523→  isLastDoc: boolean,
   524→   524→): string {
   525→   525→  // Build progress table with correct statuses based on current doc index
   526→   526→  const docs: [string, string, string][] = [

   535→   535→  // Determine statuses: docs before current are "Done", current is "Current", next is "Up Next", rest are "Pending"

   543→   543→    } else {

   545→   545→    }
   546→   546→  }

   548→   548→  const progressTable = docs.map(([num, name, status]) => `| ${num} | ${name} | ${status} |`).join("\n");
