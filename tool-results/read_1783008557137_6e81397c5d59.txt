    17→// ===== Document type mapping =====
    18→const DOC_TYPE_MAP: Record<string, string> = {
    19→  "/prd": "prd",
    20→  "/trd": "trd",
    21→  "/flow": "flow",
    22→  "/ux": "ux",
    23→  "/schema": "schema",
    24→  "/plan": "plan",

    27→// Document keyword signatures for detecting update-type messages
    28→const DOC_SIGNATURES = [
    29→  { label: "prd", keywords: ["Product Requirements Document", "Executive Summary", "Feature Requirements", "User Stories", "Target Audience", "Product Vision", "Non-Functional Requirements"] },
    30→  { label: "trd", keywords: ["Technical Requirements Document", "Architecture Overview", "Technology Stack", "API Specification", "Security Requirements", "Performance Requirements"] },
    31→  { label: "flow", keywords: ["Application Flow Document", "User Journey", "Screen Flow", "Navigation Architecture", "State Management", "Interaction Patterns"] },
    32→  { label: "ux", keywords: ["UI/UX Design Brief", "Design Principles", "Design System", "Color Palette", "Typography", "Component Guidelines", "Accessibility"] },
    33→  { label: "schema", keywords: ["Backend Schema Document", "Entity Relationship", "Schema Definitions", "Enum Types", "Data Integrity Rules", "Seed Data"] },
    34→  { label: "plan", keywords: ["Implementation Plan", "Phase Breakdown", "Task Breakdown", "Sprint Planning", "Resource Requirements", "Risk Register", "Quality Gates"] },
    35→];
    36→
    37→function detectDocTypeFromContent(content: string): string | null {
    38→  if (content.length < 1500) return null;
    39→  const lower = content.toLowerCase();
    40→  let bestMatch = "";
    41→  let bestCount = 0;
    42→  for (const sig of DOC_SIGNATURES) {
    43→    const count = sig.keywords.filter(kw => lower.includes(kw.toLowerCase())).length;
    44→    if (count >= 3 && count > bestCount) {
    45→      bestCount = count;
    46→      bestMatch = sig.label;
    47→    }
    48→  }
    49→  return bestMatch || null;
    50→}

   176→// GET /api/ai/chat — Paginated chat history
   177→export async function GET(request: NextRequest) {
   178→  try {
   179→    const user = await getAuthUser(request);
   180→    if (!user) {
   181→      return NextResponse.json({ success: false, error: "Authentication required" }, { status: 401 });
   182→    }
   183→
   184→    const { searchParams } = new URL(request.url);
   185→    const projectId = searchParams.get("projectId");
   186→    if (!projectId) {
   187→      return NextResponse.json({ success: false, error: "projectId is required" }, { status: 400 });
   188→    }
   189→
   190→    // Check project access
   191→    const canAccess = await hasProjectAccess(user.id, user.role, projectId);
   192→    if (!canAccess) {
   193→      return NextResponse.json({ success: false, error: "You don't have access to this project" }, { status: 403 });
   194→    }
   195→
   196→    const page = Math.max(1, parseInt(searchParams.get("page") || "1"));
   197→    const limit = Math.min(100, Math.max(1, parseInt(searchParams.get("limit") || "50")));
   198→
   199→    const client = getTursoClient();
   200→    const offset = (page - 1) * limit;
   201→
   202→    let messagesResult;
   203→
   204→    if (user.role === "SUPERADMIN") {
   205→      // SUPERADMIN sees all messages — newest first (client will reverse for display)
   206→      messagesResult = await client.execute({
   207→        sql: `SELECT m.*, u.name as userName
   208→              FROM "AiChat" m
   209→              LEFT JOIN "User" u ON m."userId" = u.id
   210→              WHERE m."projectId" = ?
   211→              ORDER BY m."timestamp" DESC
   212→              LIMIT ? OFFSET ?`,
   213→        args: [projectId, limit, offset],
   214→      });
   215→    } else {
   216→      // Others see only own messages + assistant messages — newest first
   217→      messagesResult = await client.execute({
   218→        sql: `SELECT m.*, u.name as userName
   219→              FROM "AiChat" m
   220→              LEFT JOIN "User" u ON m."userId" = u.id
   221→              WHERE m."projectId" = ? AND (m."userId" = ? OR m."role" = 'assistant')
   222→              ORDER BY m."timestamp" DESC
   223→              LIMIT ? OFFSET ?`,
   224→        args: [projectId, user.id, limit, offset],
   225→      });
   226→    }
   227→
   228→    const countResult = await client.execute({
   229→      sql: `SELECT COUNT(*) as total FROM "AiChat" WHERE "projectId" = ?`,
   230→      args: [projectId],
   231→    });
   232→    const total = Number(countResult.rows[0].total);
   233→
   234→    const messages = messagesResult.rows.map((row) => ({
   235→      id: row.id,
   236→      userId: row.userId,
   237→      projectId: row.projectId,
   238→      role: row.role,
   239→      content: row.content,
   240→      timestamp: row.timestamp,
   241→      userName: row.userName,
   242→    }));
   243→
   244→    return NextResponse.json({
   245→      success: true,
   246→      data: {
   247→        messages,
   248→        pagination: {
   249→          page,
   250→          limit,
   251→          total,
   252→          totalPages: Math.ceil(total / limit),
   253→        },
   254→      },
   255→    });
   256→  } catch (error) {
   257→    console.error("[GET /api/ai/chat] Error:", error);
   258→    return NextResponse.json({ success: false, error: "Internal server error" }, { status: 500 });
   259→  }
   260→}

   266→/**
   267→ * Parse tasks from a markdown document. Uses doc-type-specific patterns
   268→ * to extract actionable items from PRD, TRD, Flow, UX, Schema, and Plan docs.
   269→ */
   270→async function autoGenerateTodosFromDoc(
   271→  dbClient: ReturnType<typeof getTursoClient>,
   272→  projectId: string,
   273→  docContent: string,
   274→  docType: string,
   275→  userId: string,
   276→): Promise<number> {
   277→  // Get current max sortOrder for the project
   278→  const maxOrder = await dbClient.execute({
   279→    sql: `SELECT COALESCE(MAX("sortOrder"), -1) as maxOrder FROM "ProjectTodo" WHERE "projectId" = ?`,
   280→    args: [projectId],
   281→  });
   282→  let sortOrder = Number(maxOrder.rows[0].maxOrder) + 1;
   283→
   284→  const tasks: { title: string; description: string; priority: string }[] = [];
   285→  const lines = docContent.split('\n');
   286→  let currentSection = '';
   287→  let currentPhase = '';
   288→
   289→  // Doc-type-specific section prefixes for richer descriptions
   290→  const DOC_LABELS: Record<string, string> = {
   291→    prd: "Product Requirements",
   292→    trd: "Technical Requirements",
   293→    flow: "Application Flow",
   294→    ux: "UI/UX Design",
   295→    schema: "Database Schema",
   296→    plan: "Implementation Plan",
   297→  };
   298→  const docLabel = DOC_LABELS[docType] || docType.toUpperCase();
   299→
   300→  for (let i = 0; i < lines.length; i++) {
   301→    const line = lines[i].trim();
   302→
   303→    // Track current section/phase for context
   304→    if (line.match(/^#{1,3}\s/)) {
   305→      currentSection = line.replace(/^#+\s*/, '').replace(/\*\*/g, '').trim();
   306→      if (/phase/i.test(currentSection)) {
   307→        currentPhase = currentSection;
   308→      }
   309→      continue;
   310→    }
   311→
   312→    // Match markdown list items: "- [ ] task", "- task", "* task", numbered lists
   313→    const taskMatch = line.match(/^[-*]\s+\[?\s*\]?\s+(.+)/) || line.match(/^\d+\.\s+(.+)/);
   314→    if (taskMatch) {
   315→      let taskText = taskMatch[1].trim();
   316→      // Skip if it's a sub-heading or too short
   317→      if (taskText.length < 5) continue;
   318→      // Skip if it's just a bold title (like a section header in a list)
   319→      if (taskText.match(/^\*\*[^*]+\*\*$/) && taskText.length < 80) continue;
   320→
   321→      // Determine priority from context
   322→      let priority = 'MEDIUM';
   323→      if (/\b(critical|urgent|high.?priority|must.?have|blocker|p0)\b/i.test(taskText) ||
   324→          /\b(critical|urgent|high.?priority|must.?have|blocker|p0)\b/i.test(currentSection)) {
   325→        priority = 'HIGH';
   326→      } else if (/\b(low.?priority|nice.?to.?have|optional|future|p[2-3])\b/i.test(taskText) ||
   327→                 /\b(low.?priority|nice.?to.?have|optional|future|p[2-3])\b/i.test(currentSection)) {
   328→        priority = 'LOW';
   329→      }
   330→
   331→      // Build description from context
   332→      let description = `Source: ${docLabel}`;
   333→      if (currentPhase) {
   334→        description += `\nPhase: ${currentPhase}`;
   335→      }
   336→      if (currentSection && currentSection !== currentPhase) {
   337→        description += `\nSection: ${currentSection}`;
   338→      }
   339→
   340→      // Try to grab a sub-description line (next non-empty, non-list line)
   341→      for (let j = i + 1; j < Math.min(i + 3, lines.length); j++) {
   342→        const nextLine = lines[j].trim();
   343→        if (nextLine && !nextLine.match(/^[-*#]/) && !nextLine.match(/^\d+\./)) {
   344→          description += `\n${nextLine}`;
   345→          break;
   346→        }
   347→      }
   348→
   349→      tasks.push({
   350→        title: taskText,
   351→        description: description.trim(),
   352→        priority,
   353→      });
   354→    }
   355→  }
   356→
   357→  if (tasks.length === 0) return 0;
   358→
   359→  // Check for existing todos to avoid duplicates
   360→  const existingTodos = await dbClient.execute({
   361→    sql: `SELECT title FROM "ProjectTodo" WHERE "projectId" = ?`,
   362→    args: [projectId],
   363→  });
   364→  const existingTitles = new Set(existingTodos.rows.map(r => (r.title as string).toLowerCase()));
   365→
   366→  const newTasks = tasks.filter(t => !existingTitles.has(t.title.toLowerCase()));
   367→  if (newTasks.length === 0) {
   368→    console.log(`[autoGenerateTodos] All ${tasks.length} tasks from ${docType} already exist, skipping.`);
   369→    return 0;
   370→  }
   371→
   372→  // Batch insert all tasks
   373→  for (const task of newTasks) {
   374→    const id = crypto.randomUUID();
   375→    await dbClient.execute({
   376→      sql: `INSERT INTO "ProjectTodo" (id, "projectId", title, description, status, priority, "sortOrder", "createdBy", "createdAt", "updatedAt")
   377→            VALUES (?, ?, ?, ?, 'PENDING', ?, ?, ?, datetime('now'), datetime('now'))`,
   378→      args: [id, projectId, task.title, task.description || null, task.priority, sortOrder++, userId],
   379→    });
   380→  }
   381→
   382→  console.log(`[autoGenerateTodos] Created ${newTasks.length} tasks from ${docType} (out of ${tasks.length} parsed, ${existingTitles.size} existing)`);
   383→  return newTasks.length;
   384→}

   386→// POST /api/ai/chat — Send message and get AI response (with agentic tool-calling loop)
   387→export async function POST(request: NextRequest) {
   388→  try {
   389→    const user = await getAuthUser(request);
   390→    if (!user) {
   391→      return NextResponse.json({ success: false, error: "Authentication required" }, { status: 401 });
   392→    }
   393→
   394→    const rawBody = await request.json();
   395→    const { projectId, content, files, fileData, fileName, fileType } = rawBody as { projectId: string; content: string; files?: unknown[]; fileData?: string; fileName?: string; fileType?: string };
   396→
   397→    if (!projectId || !content) {
   398→      return NextResponse.json({ success: false, error: "projectId and content are required" }, { status: 400 });
   399→    }
   400→
   401→    const attachedImages: { data: string; type: string; name: string }[] = [];
   402→    if (files && Array.isArray(files) && files.length > 0) {
   403→      // New format: array of { data, type, name }
   404→      for (const f of files) {
   405→        if (f.data && f.type && f.type.startsWith("image/")) {
   406→          attachedImages.push({ data: f.data, type: f.type, name: f.name || "image" });
   407→        }
   408→      }
   409→    } else if (fileData && fileType && fileType.startsWith("image/")) {
   410→      // Legacy single-file format
   411→      attachedImages.push({ data: fileData, type: fileType, name: fileName || "image" });
   412→    }
   413→    const hasImages = attachedImages.length > 0;
   414→
   415→    // Check project access
   416→    const canAccess = await hasProjectAccess(user.id, user.role, projectId);
   417→    if (!canAccess) {
   418→      return NextResponse.json({ success: false, error: "You don't have access to this project" }, { status: 403 });
   419→    }
   420→
   421→    const client = getTursoClient();
   422→    const ip = getClientIp(request);
   423→
   424→    // ===== Inject Settings-managed API keys into process.env for this request =====
   425→    // This ensures ALL downstream AI calls (chatCompletion, visionCompletion, fallbacks)
   426→    // use keys from the Settings UI rather than requiring env vars.
   427→    try {
   428→      const { getAllProviderConfigs } = await import("@/lib/settings-resolver");
   429→      const configs = await getAllProviderConfigs();
   430→      for (const [key, value] of Object.entries(configs)) {
   431→        if (value && !process.env[key]) {
   432→          process.env[key] = value;
   433→        }
   434→      }
   435→      console.log("[POST /api/ai/chat] Settings-managed keys injected into process.env");
   436→    } catch (injectErr) {
   437→      console.error("[POST /api/ai/chat] Settings key injection failed (non-fatal):", injectErr);
   438→    }
   439→
   440→    // Auto-migrate AI tables if they don't exist
   441→    await ensureAiTables(client);
   442→
   443→    // Save user message
   444→    const userMsgId = crypto.randomUUID();
   445→    await client.execute({
   446→      sql: `INSERT INTO "AiChat" (id, "userId", "projectId", "role", "content", "timestamp")
   447→            VALUES (?, ?, ?, 'user', ?, datetime('now'))`,
   448→      args: [userMsgId, user.id, projectId, content],
   449→    });
   450→
   451→    // Detect command
   452→    const trimmed = content.trim();
   453→    const command = trimmed.startsWith("/") ? trimmed.split(/[\s\n]/)[0] : undefined;
   454→
   455→    // Detect command type (needed before loading history to determine history limit)
   456→    const isDocCommand = !!command && ["/docs", "/prd", "/trd", "/flow", "/ux", "/schema", "/plan", "/init"].includes(command);
   457→    const isSkillCommand = !!command && ["/standup", "/risks", "/summarize", "/code-review", "/knowledge"].includes(command);
   458→
   459→    // Load recent chat messages for context
   460→    // For doc commands: only last 6 messages (saves tokens, avoids bloated context)
   461→    // For skill commands & regular chat: last 20 messages for context-aware analysis
   462→    const historyLimit = isDocCommand ? 6 : 20;
   463→    const historyResult = await client.execute({
   464→      sql: `SELECT role, content FROM "AiChat"
   465→            WHERE "projectId" = ?
   466→            ORDER BY "timestamp" DESC
   467→            LIMIT ?`,
   468→      args: [projectId, historyLimit],
   469→    });
   470→    const chatHistory = historyResult.rows.reverse().map((row) => ({
   471→      role: row.role,
   472→      content: row.content,
   473→    }));
   474→
   475→    // Load user name from DB
   476→    let userName = user.name;
   477→    try {
   478→      const userResult = await client.execute({
   479→        sql: `SELECT name FROM "User" WHERE id = ?`,
   480→        args: [user.id],
   481→      });
   482→      if (userResult.rows.length > 0) {

   658→      label: string;

   666→      userId: user.id,

   674→      { role: "system", content: enrichedSystemPrompt },

   677→    for (const msg of chatHistory) {

   717→          aiText = result.content;
   718→        }
   719→      } else {
   720→        // ===== Agentic Tool-Calling Loop with Provider Fallback =====
   721→        // Helper: call AI with automatic fallback on 429/413/401 errors
   722→        const callWithFallback = async (
   723→          messages: AiMessage[],
   724→          model: string,
   725→          maxTok: number,
   726→          tools?: Parameters<typeof chatCompletion>[0]["tools"],
   727→          toolChoice?: Parameters<typeof chatCompletion>[0]["tool_choice"],

   730→          let result = await chatCompletion({
   731→            messages,
   732→            model,
   733→            maxTokens: maxTok,
   734→            tools,
   735→            tool_choice: toolChoice,

   742→              vision: false,
   743→            });

   750→                messages,
   751→                model: fallbackModel,
   752→                maxTokens: maxTok,
   753→                tools,
   754→                tool_choice: toolChoice,
   755→              });

" (auto-routed)" : ""} command=${command || "none"} has_tools=${!!result.tool_calls} content_len=${result.content?.length || 0} success=${result.success} error=${result.error || "none"}`);

   830→

   833→            role: "assistant",

   848→              label: toolLabel,
   849→              icon: toolIcon,
   850→              status: "running",

   858→            toolExecutions[execIndex].status = toolResult.success ? "success" : "error";

   863→              role: "tool",
   864→              tool_call_id: toolCall.id,
   865→              content: toolResult.result,
   866→              name: toolName,
   867→            });
   868→          }
   869→
   870→          // Loop continues — AI will see tool results and decide next action
   871→        }
   872→

   874→        if (!finalContent && round >= MAX_TOOL_ROUNDS) {

   890→                { role: "system", content: enrichedSystemPrompt },

   915→        aiText = typeof finalContent === "string" ? finalContent : JSON.stringify(finalContent);

   928→        { pattern: /(?:I[''](ve| have| will| am| just))?\s*(?:pushed|pushing|committed|deployed|deploying|created|creating|built|building|cloned|cloning|merged|merging)\s+(?:the\s+)?(?:code|changes|files|project|application|app|repo|repository|to\s+GitHub|to\s+the\s+repo)/gi,

   931→          replacement: "" },
   932→        { pattern: /(?:Build\s+passed|Deployment\s+successful|Successfully\s+deployed|CI\/CD\s+passed)/gi,
   933→          replacement: "[Note: This was not actually executed.]" },
   934→      ];

   936→      // Only apply filter if NO github/file tools were used
   937→      const usedGitTools = toolNames.has("github_pull") || toolNames.has("github_push_code") || toolNames.has("create_or_update_file");
   938→      if (!usedGitTools) {
   939→        for (const { pattern, replacement } of fakeActionPatterns) {
   940→          aiText = aiText.replace(pattern, replacement);
   941→        }
   942→        // Remove references to other projects/applications
   943→        const otherProjectPattern = /(?:(?:the|a|an)\s+)?(?:Share\s+Sathi|[A-Z][a-z]+(?:\s+[A-Z][a-z]+)+\s+application)(?!\s+(?:project|in\s+KarmaBoard))/g;
   944→        aiText = aiText.replace(otherProjectPattern, "[irrelevant project reference removed]");
   945→      }
   946→    }
   947→
   948→    // Save AI response
   949→    const aiMsgId = crypto.randomUUID();
   950→    await client.execute({
   951→      sql: `INSERT INTO "AiChat" (id, "userId", "projectId", "role", "content", "timestamp")
   952→            VALUES (?, ?, ?, 'assistant', ?, datetime('now'))`,
   953→      args: [aiMsgId, user.id, projectId, aiText],
   954→    });
   955→
   956→    // ===== Auto-save document to ProjectDocument =====
   957→    let documentInfo: { id: string; docType: string; title: string; version: number } | undefined;
   958→    if (!aiError && aiText.length > 500) {
   959→      try {
   960→        let docType: string | undefined;
   961→
   962→        // Check if this is a direct doc command
   963→        if (isDocCommand && command && DOC_TYPE_MAP[command]) {
   964→          docType = DOC_TYPE_MAP[command];
   965→        }
   966→        // Or if the AI response matches a document signature (update flow)
   967→        if (!docType) {
   968→          const detected = detectDocTypeFromContent(aiText);
   969→          docType = detected || undefined;
   970→        }
   971→
   972→        if (docType) {
   973→          // Extract title from markdown heading
   974→          const title = aiText.split("\n").find(l => l.trim().startsWith("#"))
   975→            ?.replace(/^#+\s*/, "").replace(/\*\*/g, "").trim().slice(0, 100)
   976→            || docType.toUpperCase() + " Document";

  1015→      }
  1016→    }
  1017→
  1018→    // ===== Auto-generate todos from pre-coding documents =====

  1031→      }
  1032→    }
  1033→
  1034→    // Log activity
  1035→    const toolSummary = toolExecutions.length > 0
  1036→      ? ` (tools used: ${toolExecutions.map((t) => t.toolName).join(", ")})`
  1037→      : "";
  1038→    await logActivity({
  1039→      userId: user.id,
  1040→      action: "AI_CHAT_MESSAGE",
  1041→      details: `Sent AI chat message in project${project?.name ? ` "${project.name}"` : ""}${command ? ` (command: ${command})` : ""}${toolSummary}`,
  1042→      entity: "ai_chat",
  1043→      entityId: projectId,
  1044→      ipAddress: ip,
  1045→      tursoClient: client,
  1046→    });
  1047→
  1048→    // ===== Extract TRD theme colors (runs after doc save if TRD was generated) =====

  1050→      try {
  1051→        const settingsKey = "PROJECT_THEME:" + projectId;

  1059→            const themeColors = { colors: [...new Set(colorMatches)] };
  1060→            await client.execute({
  1061→              sql: `INSERT OR REPLACE INTO "Settings" (key, value, "updatedAt") VALUES (?, ?, datetime('now'))`,
  1062→              args: [settingsKey, JSON.stringify(themeColors)],
  1063→            });
  1064→            console.log("[Chat] Extracted theme colors from TRD:", themeColors.colors.length);
  1065→          }
  1066→        }
  1067→      } catch (themeErr) {
  1068→        console.error("[Chat] Theme extraction error (non-fatal):", themeErr);
  1069→      }
  1070→    }
  1071→
  1072→    // ===== Auto-push document to GitHub (fire-and-forget, non-blocking) =====
  1073→    if (documentInfo && !aiError) {
  1074→      // Fire-and-forget: push to GitHub in background without blocking the response
  1075→      (async () => {
  1076→        try {
  1077→          const repoResult = await client.execute({
  1078→            sql: `SELECT value FROM "Settings" WHERE key = 'GITHUB_REPO_URL'`,
  1079→            args: [],
  1080→          });
  1081→
  1082→          if (repoResult.rows.length === 0) {
  1083→            console.log("[Chat] No GitHub config, skipping auto-push for", documentInfo.docType);
  1084→            return;
  1085→          }
  1086→
  1087→          const repoUrl = repoResult.rows[0].value as string;
  1088→          // Use settings-resolver to get decrypted PAT from Settings DB
  1089→          const token = await getGitHubPat();
  1090→          if (!token) {
  1091→            console.log("[Chat] No GitHub PAT in settings, skipping auto-push for", documentInfo.docType);
  1092→            return;
  1093→          }
  1094→
  1095→          const githubConfig = { repoUrl, token };
  1096→          const docLabel = documentInfo.docType.toUpperCase();
  1097→          const commitMsg = `docs: update ${docLabel} document (v${documentInfo.version})`;
  1098→
  1099→          // Push markdown file
  1100→          const mdPath = "docs/pre-coding/" + documentInfo.docType + ".md";
  1101→          await pushFile(githubConfig, mdPath, aiText, commitMsg);
  1102→          console.log("[Chat] GitHub push: markdown", mdPath);
  1103→
  1104→          // Push PDF if available
  1105→          if (documentInfo) {
  1106→            const pdfResult = await client.execute({
  1107→              sql: `SELECT "pdfData" FROM "ProjectDocument" WHERE id = ? AND "pdfData" IS NOT NULL AND "pdfData" != ''`,
  1108→              args: [documentInfo.id],
  1109→            });

  1111→              const pdfPath = "docs/pre-coding/" + documentInfo.docType + ".pdf";
  1112→              await pushBinaryFile(githubConfig, pdfPath, pdfResult.rows[0].pdfData as string, commitMsg);
  1113→              console.log("[Chat] GitHub push: PDF", pdfPath);
  1114→            }
  1115→          }
  1116→
  1117→          console.log("[Chat] GitHub auto-push complete for", documentInfo.docType, "v" + documentInfo.version);
  1118→        } catch (ghErr) {
  1119→          console.error("[Chat] GitHub auto-push failed (non-fatal):", ghErr);
  1120→        }
  1121→      })();
  1122→    }
  1123→
  1124→    // ===== z.ai Bridge: Check if /init should include bridge info =====
  1125→    // If command is /init and project has documents, check z.ai config and add bridge data
  1126→    let zaiBridge: {
  1127→      chatId: string;
  1128→      chatUrl: string;
  1129→      context: string;
  1130→      modelName: string;
  1131→      documentsFound: number;
  1132→      isNewChat: boolean;
  1133→      aiResponse?: string;
  1134→      apiError?: string;
  1135→      chatMessagesFound?: number;
  1136→      docsSource?: string;
  1137→    } | undefined;
  1138→
  1139→    if (command === "/init") {
  1140→      try {
  1141→        // Count project documents (no minimum required — send whatever exists)
  1142→        const docsCountResult = await client.execute({
  1143→          sql: `SELECT COUNT(*) as count FROM "ProjectDocument" WHERE "projectId" = ? AND "docType" IN ('prd', 'trd', 'flow', 'ux', 'schema', 'plan')`,
  1144→          args: [projectId],

  1146→        const docsCount = Number(docsCountResult.rows[0]?.count || 0);
  1147→

  1279→          zaiBridge = {

  1294→
  1295→          console.log(`[Chat] z.ai Bridge: ${docsCount} docs + ${chatMessagesFound} chat msgs → ${chunkResult.totalChunks} chunks, ${chunkResult.chunksSent} sent, aiResponse=${!!chunkResult.aiResponse}, apiError=${!!chunkResult.apiError}`);
  1296→        }
  1297→      } catch (bridgeErr) {

  1300→    }
  1301→
  1302→    return NextResponse.json({

  1316→  } catch (error) {

  1319→  }
