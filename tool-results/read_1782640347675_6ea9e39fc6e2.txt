    77→
    78→  const [settings, setSettings] = useState<Record<string, SettingItem>>({});
    79→  const [loading, setLoading] = useState(true);
    80→  const [saving, setSaving] = useState(false);
    81→
    82→  // Form state

    84→  const [resendApiKey, setResendApiKey] = useState("");
    85→  const [fromEmail, setFromEmail] = useState("");
    86→  const [fromName, setFromName] = useState("");
    87→  const [smtpUser, setSmtpUser] = useState("");
    88→  const [smtpPassword, setSmtpPassword] = useState("");
    89→
    90→  // Track changes to sensitive fields
    91→  const [resendKeyChanged, setResendKeyChanged] = useState(false);
    92→  const [smtpPassChanged, setSmtpPassChanged] = useState(false);
    93→
    94→  // PDF Theme state
    95→  const [pdfThemePrimary, setPdfThemePrimary] = useState("#1E40AF");

    97→  const [pdfThemePrimaryBg, setPdfThemePrimaryBg] = useState("#EFF6FF");
    98→  const [pdfThemeAltRowBg, setPdfThemeAltRowBg] = useState("#F9FAFB");
    99→  const [pdfThemeCoverTop, setPdfThemeCoverTop] = useState("#1E3A8A");
   100→  const [pdfThemeAccent, setPdfThemeAccent] = useState("#059669");
   101→  const [pdfThemeWarning, setPdfThemeWarning] = useState("#D97706");
   102→  const [pdfThemeDanger, setPdfThemeDanger] = useState("#DC2626");
   103→  const [pdfThemeHasChanges, setPdfThemeHasChanges] = useState(false);
   104→  const [pdfThemeSaving, setPdfThemeSaving] = useState(false);
   105→
   106→  // z.ai Bridge state
   107→  const [zaiApiKey, setZaiApiKey] = useState("");
   108→  const [zaiApiKeyChanged, setZaiApiKeyChanged] = useState(false);
   109→  const [zaiEmail, setZaiEmail] = useState("");
   110→  const [zaiEmailChanged, setZaiEmailChanged] = useState(false);
   111→  const [zaiPassword, setZaiPassword] = useState("");
   112→  const [zaiPasswordChanged, setZaiPasswordChanged] = useState(false);
   113→  const [zaiBaseUrl, setZaiBaseUrl] = useState("https://api.z.ai/api/paas/v4");
   114→  const [zaiModel, setZaiModel] = useState("glm-4.7-flash");
   115→  const [zaiTestStatus, setZaiTestStatus] = useState<null | 'success' | 'error'>(null);
   116→  const [zaiTesting, setZaiTesting] = useState(false);
   117→  const [zaiSaving, setZaiSaving] = useState(false);

   119→  // Fetch settings
   120→  useEffect(() => {
   121→    async function fetchSettings() {
   122→      try {
   123→        const res = await fetch("/api/settings");
   124→        const json = await res.json();
   125→        if (json.success) {
   126→          setSettings(json.data);
   127→          const data = json.data as Record<string, SettingItem>;
   128→
   129→          if (data.EMAIL_PROVIDER) setEmailProvider(data.EMAIL_PROVIDER.value);
   130→          if (data.RESEND_API_KEY) setResendApiKey(data.RESEND_API_KEY.value);
   131→          if (data.RESEND_FROM_EMAIL) setFromEmail(data.RESEND_FROM_EMAIL.value);
   132→          if (data.RESEND_FROM_NAME) setFromName(data.RESEND_FROM_NAME.value);
   133→          if (data.SMTP_USER) setSmtpUser(data.SMTP_USER.value);
   134→          if (data.SMTP_PASSWORD) setSmtpPassword(data.SMTP_PASSWORD.value);
   135→
   136→          // Load z.ai Bridge settings
   137→          if (data.ZAI_BRIDGE_API_KEY) setZaiApiKey(data.ZAI_BRIDGE_API_KEY.value);
   138→          if (data.ZAI_BRIDGE_EMAIL) setZaiEmail(data.ZAI_BRIDGE_EMAIL.value);
   139→          if (data.ZAI_BRIDGE_PASSWORD) setZaiPassword(data.ZAI_BRIDGE_PASSWORD.value);
   140→          if (data.ZAI_BRIDGE_BASE_URL) setZaiBaseUrl(data.ZAI_BRIDGE_BASE_URL.value);
   141→          if (data.ZAI_BRIDGE_MODEL) setZaiModel(data.ZAI_BRIDGE_MODEL.value);
   142→
   143→          // Load PDF theme
   144→          if (data.PDF_THEME) {
   145→            try {
   146→              const theme = JSON.parse(data.PDF_THEME.value);
   147→              if (theme.primary) setPdfThemePrimary(theme.primary);
   148→              if (theme.primaryLight) setPdfThemePrimaryLight(theme.primaryLight);
   149→              if (theme.primaryBg) setPdfThemePrimaryBg(theme.primaryBg);
   150→              if (theme.altRowBg) setPdfThemeAltRowBg(theme.altRowBg);
   151→              if (theme.coverGradientTop) setPdfThemeCoverTop(theme.coverGradientTop);
   152→              if (theme.accent) setPdfThemeAccent(theme.accent);
   153→              if (theme.warning) setPdfThemeWarning(theme.warning);
   154→              if (theme.danger) setPdfThemeDanger(theme.danger);
   155→            } catch { /* ignore invalid JSON */ }
   156→          }
   157→        } else {
   158→          errorToast({ error: json.error, title: "Failed to load settings" });
   159→        }
   160→      } catch {
   161→        errorToast({ error: "Failed to load settings" });
   162→      } finally {
   163→        setLoading(false);
   164→      }
   165→    }
   166→    if (userRole === "SUPERADMIN") fetchSettings();
   167→  }, [userRole]);

   170→  async function handleTestZai() {

   172→    setZaiTestStatus(null);
   173→    try {
   174→      const res = await fetch("/api/settings/test-zai", {

   180→      setZaiTestStatus(json.success ? "success" : "error");

   182→        toast.success("z.ai connection successful! Model: " + (json.model || "OK"));
   183→      } else {
   184→        errorToast({ error: json.error, title: "Connection failed" });
   185→      }
   186→    } catch (err) {
   187→      setZaiTestStatus("error");
   188→      errorToast({ error: err, title: "Connection test failed" });
   189→    } finally {
   190→      setZaiTesting(false);
   191→    }
   192→  }
   193→
   194→  // Save z.ai bridge settings
   195→  async function handleSaveZai() {
   196→    setZaiSaving(true);
   197→    try {
   198→      const updateSettings: Record<string, string> = {};
   199→      if (zaiApiKeyChanged && zaiApiKey) updateSettings.ZAI_BRIDGE_API_KEY = zaiApiKey;
   200→      if (zaiEmailChanged && zaiEmail) updateSettings.ZAI_BRIDGE_EMAIL = zaiEmail;
   201→      if (zaiPasswordChanged && zaiPassword) updateSettings.ZAI_BRIDGE_PASSWORD = zaiPassword;
   202→      if (zaiBaseUrl !== (settings.ZAI_BRIDGE_BASE_URL?.value || "https://api.z.ai/api/paas/v4")) updateSettings.ZAI_BRIDGE_BASE_URL = zaiBaseUrl;
   203→      if (zaiModel !== (settings.ZAI_BRIDGE_MODEL?.value || "glm-4.7-flash")) updateSettings.ZAI_BRIDGE_MODEL = zaiModel;
   204→
   205→      const res = await fetch("/api/settings", {
   206→        method: "PUT",
   207→        headers: { "Content-Type": "application/json" },
   208→        body: JSON.stringify({ settings: updateSettings }),
   209→      });
   210→      const json = await res.json();
   211→      if (json.success) {
   212→        toast.success("z.ai Bridge settings saved successfully");
   213→        setZaiApiKeyChanged(false);
   214→        setZaiEmailChanged(false);
   215→        setZaiPasswordChanged(false);
   216→        // Refresh settings
   217→        const refreshRes = await fetch("/api/settings");
   218→        const refreshJson = await refreshRes.json();
   219→        if (refreshJson.success) {
   220→          setSettings(refreshJson.data);
   221→          const data = refreshJson.data as Record<string, SettingItem>;
   222→          if (data.ZAI_BRIDGE_API_KEY) setZaiApiKey(data.ZAI_BRIDGE_API_KEY.value);
   223→          if (data.ZAI_BRIDGE_EMAIL) setZaiEmail(data.ZAI_BRIDGE_EMAIL.value);
   224→          if (data.ZAI_BRIDGE_PASSWORD) setZaiPassword(data.ZAI_BRIDGE_PASSWORD.value);
   225→          if (data.ZAI_BRIDGE_BASE_URL) setZaiBaseUrl(data.ZAI_BRIDGE_BASE_URL.value);
   226→          if (data.ZAI_BRIDGE_MODEL) setZaiModel(data.ZAI_BRIDGE_MODEL.value);
   227→        }
   228→      } else {
   229→        errorToast({ error: json.error, title: "Failed to save z.ai settings" });
   230→      }
   231→    } catch (err) {
   232→      errorToast({ error: err, title: "Save z.ai settings failed" });
   233→    } finally {
   234→      setZaiSaving(false);
   235→    }
   236→  }

   238→  // Save settings
   239→  async function handleSave() {
   240→    setSaving(true);
   241→    try {
   242→      const updateSettings: Record<string, string> = {};
   243→
   244→      if (emailProvider !== (settings.EMAIL_PROVIDER?.value || "")) {
   245→        updateSettings.EMAIL_PROVIDER = emailProvider;
   246→      }
   247→      if (resendKeyChanged && resendApiKey) {
   248→        updateSettings.RESEND_API_KEY = resendApiKey;
   249→      }
   250→      if (fromEmail !== (settings.RESEND_FROM_EMAIL?.value || "")) {
   251→        updateSettings.RESEND_FROM_EMAIL = fromEmail;
   252→      }
   253→      if (fromName !== (settings.RESEND_FROM_NAME?.value || "")) {
   254→        updateSettings.RESEND_FROM_NAME = fromName;
   255→      }
   256→      if (smtpUser !== (settings.SMTP_USER?.value || "")) {
   257→        updateSettings.SMTP_USER = smtpUser;
   258→      }
   259→      if (smtpPassChanged && smtpPassword) {
   260→        updateSettings.SMTP_PASSWORD = smtpPassword;
   261→      }
   262→
   263→      if (Object.keys(updateSettings).length === 0) {
   264→        toast.info("No changes to save");
   265→        setSaving(false);
   266→        return;
   267→      }
   268→
   269→      const res = await fetch("/api/settings", {
   270→        method: "PUT",
   271→        headers: { "Content-Type": "application/json" },
   272→        body: JSON.stringify({ settings: updateSettings }),
   273→      });
   274→      const json = await res.json();
   275→
   276→      if (json.success) {
   277→        toast.success(json.message || "Settings saved successfully");
   278→        setResendKeyChanged(false);
   279→        setSmtpPassChanged(false);
   280→
   281→        // Refresh settings
   282→        const refreshRes = await fetch("/api/settings");
   283→        const refreshJson = await refreshRes.json();
   284→        if (refreshJson.success) {
   285→          setSettings(refreshJson.data);
   286→          const data = refreshJson.data as Record<string, SettingItem>;
   287→          if (data.RESEND_API_KEY && !resendKeyChanged) setResendApiKey(data.RESEND_API_KEY.value);
   288→          if (data.RESEND_FROM_EMAIL) setFromEmail(data.RESEND_FROM_EMAIL.value);
   289→          if (data.RESEND_FROM_NAME) setFromName(data.RESEND_FROM_NAME.value);
   290→          if (data.EMAIL_PROVIDER) setEmailProvider(data.EMAIL_PROVIDER.value);
   291→          if (data.SMTP_USER) setSmtpUser(data.SMTP_USER.value);
   292→          if (data.SMTP_PASSWORD && !smtpPassChanged) setSmtpPassword(data.SMTP_PASSWORD.value);
   293→          if (data.ZAI_BRIDGE_API_KEY && !zaiApiKeyChanged) setZaiApiKey(data.ZAI_BRIDGE_API_KEY.value);
   294→          if (data.ZAI_BRIDGE_BASE_URL) setZaiBaseUrl(data.ZAI_BRIDGE_BASE_URL.value);
   295→          if (data.ZAI_BRIDGE_MODEL) setZaiModel(data.ZAI_BRIDGE_MODEL.value);
   296→        }
   297→      } else {
   298→        errorToast({ error: json.error, title: "Failed to save settings" });
   299→      }
   300→    } catch (err) {
   301→      errorToast({ error: err, title: "Save settings failed" });
   302→    } finally {
   303→      setSaving(false);
   304→    }
   305→  }
   306→
   307→  function hasChanges(): boolean {
   308→    return (
   309→      emailProvider !== (settings.EMAIL_PROVIDER?.value || "") ||
   310→      resendKeyChanged ||
   311→      fromEmail !== (settings.RESEND_FROM_EMAIL?.value || "") ||
   312→      fromName !== (settings.RESEND_FROM_NAME?.value || "") ||
   313→      smtpUser !== (settings.SMTP_USER?.value || "") ||
   314→      smtpPassChanged
   315→    );
   316→  }

   318→  // Save PDF theme separately
   319→  async function handleSavePdfTheme() {
   320→    setPdfThemeSaving(true);
   321→    try {
   322→      const themeObj = {
   323→        primary: pdfThemePrimary,
   324→        primaryLight: pdfThemePrimaryLight,
   325→        primaryBg: pdfThemePrimaryBg,
   326→        altRowBg: pdfThemeAltRowBg,
   327→        coverGradientTop: pdfThemeCoverTop,
   328→        coverGradientBottom: pdfThemePrimary,
   329→        accent: pdfThemeAccent,
   330→        warning: pdfThemeWarning,
   331→        danger: pdfThemeDanger,
   332→      };
   333→
   334→      const res = await fetch("/api/settings", {
   335→        method: "PUT",
   336→        headers: { "Content-Type": "application/json" },
   337→        body: JSON.stringify({ settings: { PDF_THEME: JSON.stringify(themeObj) } }),
   338→      });
   339→      const json = await res.json();
   340→
   341→      if (json.success) {
   342→        toast.success("PDF theme saved successfully");
   343→        setPdfThemeHasChanges(false);
   344→        // Refresh settings
   345→        const refreshRes = await fetch("/api/settings");
   346→        const refreshJson = await refreshRes.json();
   347→        if (refreshJson.success) {
   348→          setSettings(refreshJson.data);
   349→          const data = refreshJson.data as Record<string, SettingItem>;
   350→          if (data.PDF_THEME) {
   351→            const theme = JSON.parse(data.PDF_THEME.value);
   352→            if (theme.primary) setPdfThemePrimary(theme.primary);
   353→            if (theme.primaryLight) setPdfThemePrimaryLight(theme.primaryLight);
   354→            if (theme.primaryBg) setPdfThemePrimaryBg(theme.primaryBg);
   355→            if (theme.altRowBg) setPdfThemeAltRowBg(theme.altRowBg);
   356→            if (theme.coverGradientTop) setPdfThemeCoverTop(theme.coverGradientTop);
   357→            if (theme.accent) setPdfThemeAccent(theme.accent);
   358→            if (theme.warning) setPdfThemeWarning(theme.warning);
   359→            if (theme.danger) setPdfThemeDanger(theme.danger);
   360→          }
   361→        }
   362→      } else {
   363→        errorToast({ error: json.error, title: "Failed to save PDF theme" });
   364→      }
   365→    } catch (err) {
   366→      errorToast({ error: err, title: "Save PDF theme failed" });
   367→    } finally {
   368→      setPdfThemeSaving(false);
   369→    }
   370→  }

   373→  useEffect(() => {

   382→        pdfThemeAltRowBg !== (theme.altRowBg || "#F9FAFB") ||

   390→  }, [pdfThemePrimary, pdfThemePrimaryLight, pdfThemePrimaryBg, pdfThemeAltRowBg, pdfThemeCoverTop, pdfThemeAccent, pdfThemeWarning, pdfThemeDanger, settings.PDF_THEME]);
   391→
   392→  if (userRole !== "SUPERADMIN") {
   393→    return (
   394→      <div className="flex flex-col items-center justify-center py-16 text-center">

   396→        <h2 className="text-lg font-semibold">Access Denied</h2>
   397→        <p className="text-sm text-muted-foreground mt-1">
   398→          Only superadmins can access application settings.
   399→        </p>
   400→      </div>
   401→    );
   402→  }
   403→
   404→  if (loading) return <SettingsSkeleton />;
   405→
   406→  const isGmail = emailProvider === "gmail-smtp";
   407→
   408→  return (
   409→    <div className="space-y-6 max-w-3xl">
   410→      {/* Header */}
   411→      <div>
   412→        <h1 className="text-2xl font-bold tracking-tight">Settings</h1>
   413→        <p className="text-muted-foreground text-sm mt-1">
   414→          Manage application configuration and email service settings.
   415→        </p>
   416→      </div>
   417→
   418→      {/* Email Configuration */}
   419→      <Card>
   420→        <CardHeader>
   421→          <CardTitle className="text-lg flex items-center gap-2">
   422→            <Mail className="h-5 w-5" />
   423→            Email Service
   424→          </CardTitle>
   425→          <CardDescription>
   426→            Configure how KarmaBoard sends emails (welcome emails, notifications).
   427→            Settings are stored encrypted in the database and take effect immediately.
   428→          </CardDescription>
   429→        </CardHeader>
   430→        <CardContent className="space-y-5">
   431→          {/* Info alert */}
   432→          <Alert>
   433→            <AlertTriangle className="h-4 w-4" />
   434→            <AlertDescription>
   435→              Changes take effect immediately for the next member creation. No redeployment needed.
   436→            </AlertDescription>
   437→          </Alert>
   438→
   439→          <Separator />
   440→
   441→          {/* Provider Selector */}
   442→          <div className="space-y-1.5">
   443→            <Label>Email Provider</Label>
   444→            <Select value={emailProvider} onValueChange={setEmailProvider}>
   445→              <SelectTrigger className="w-full">
   446→                <SelectValue />
   447→              </SelectTrigger>
   448→              <SelectContent>
   449→                <SelectItem value="gmail-smtp">
   450→                  Gmail SMTP (testing — no domain needed)

   452→                <SelectItem value="resend">
   453→                  Resend (production — requires a domain)
   454→                </SelectItem>
   455→              </SelectContent>
   456→            </Select>
   457→            <p className="text-xs text-muted-foreground">
   458→              Use <strong>Gmail SMTP</strong> for testing with your Gmail account.
   459→              Switch to <strong>Resend</strong> when you have a custom domain.

   463→          <Separator />
   464→
   465→          {/* From Name — shared by both providers */}
   466→          <div className="space-y-1.5">
   467→            <Label htmlFor="from-name">
   468→              <User className="h-3.5 w-3.5 inline mr-1" />
   469→              From Name
   470→            </Label>
   471→            <Input
   472→              id="from-name"
   473→              placeholder="KarmaBoard"
   474→              value={fromName}
   475→              onChange={(e) => setFromName(e.target.value)}

   477→            <p className="text-xs text-muted-foreground">

   480→          </div>
   481→
   482→          {isGmail ? (
   483→            /* ========== Gmail SMTP Fields ========== */
   484→            <>
   485→              {/* Gmail Address */}
   486→              <div className="space-y-1.5">
   487→                <Label htmlFor="smtp-user" className="flex items-center gap-1.5">
   488→                  <Mail className="h-3.5 w-3.5" />
