   106→   106→const ROLE_STYLES: Record<string, string> = {
   107→   107→  SUPERADMIN: 'bg-red-100 text-red-700 border-red-200',
   108→   108→  ADMIN: 'bg-amber-100 text-amber-700 border-amber-200',

   110→   110→};
   111→   111→
   112→   112→const STATUS_STYLES: Record<string, string> = {
   113→   113→  ACTIVE: 'bg-emerald-100 text-emerald-700 border-emerald-200',
   114→   114→  INACTIVE: 'bg-gray-100 text-gray-600 border-gray-200',
   115→   115→  ON_LEAVE: 'bg-sky-100 text-sky-700 border-sky-200',
   116→   116→};
   117→   117→
   118→   118→function getInitials(name: string) {
   119→   119→  return name
   120→   120→    .split(' ')
   121→   121→    .map((n) => n[0])
   122→   122→    .join('')
   123→   123→    .toUpperCase()
   124→   124→    .slice(0, 2);
   125→   125→}
   126→   126→
   127→   127→function parseSkills(skills: string | null | undefined): string[] {
   128→   128→  if (!skills) return [];
   129→   129→  try {
   130→   130→    return JSON.parse(skills);
   131→   131→  } catch {
   132→   132→    return skills.split(',').map((s) => s.trim()).filter(Boolean);
   133→   133→  }
   134→   134→}

   191→   191→
   192→   192→function MemberDialogInner({
   193→   193→  mode,
   194→   194→  member,
   195→   195→  onSubmit,
   196→   196→  loading,
   197→   197→  onClose,
   198→   198→}: {
   199→   199→  open: boolean;
   200→   200→  onOpenChange: (open: boolean) => void;
   201→   201→  mode: 'add' | 'edit';
   202→   202→  member: Member | null;
   203→   203→  onSubmit: (data: MemberFormData) => void;
   204→   204→  loading: boolean;
   205→   205→}) {
   206→   206→  const [form, setForm] = useState<MemberFormData>(() => {
   207→   207→    if (mode === 'edit' && member) {
   208→   208→      return {
   209→   209→        name: member.name,
   210→   210→        email: member.email,
   211→   211→        password: '',
   212→   212→        jobTitle: member.jobTitle || '',
   213→   213→        phone: member.phone || '',
   214→   214→        skills: parseSkills(member.skills).join(', '),
   215→   215→        role: member.role,
   216→   216→        status: member.status,
   217→   217→      };
   218→   218→    }
   219→   219→    return EMPTY_FORM;
   220→   220→  });
   221→   221→  const [errors, setErrors] = useState<Record<string, string>>({});
   222→   222→
   223→   223→
   224→   224→  function validate(): boolean {
   225→   225→    const errs: Record<string, string> = {};
   226→   226→    if (!form.name || form.name.length < 2) errs.name = 'Name must be at least 2 characters';
   227→   227→    if (!form.email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) errs.email = 'Valid email is required';
   228→   228→    // Password is auto-generated for new members — no validation needed
   229→   229→    setErrors(errs);
   230→   230→    return Object.keys(errs).length === 0;
   231→   231→  }
   232→   232→
   233→   233→  function handleSubmit(e: React.FormEvent) {
   234→   234→    e.preventDefault();
   235→   235→    if (validate()) onSubmit(form);
   236→   236→  }
   237→   237→
   238→   238→  return (
   239→   239→    <Dialog open onOpenChange={(v) => { if (!v) onClose(); }}>
   240→   240→      <DialogContent className="sm:max-w-md max-h-[90vh] overflow-y-auto">
   241→   241→        <DialogHeader>
   242→   242→          <DialogTitle>{mode === 'add' ? 'Add New Member' : 'Edit Member'}</DialogTitle>
   243→   243→          <DialogDescription>
   244→   244→            {mode === 'add'
   245→   245→              ? 'Create a new team member account.'
   246→   246→              : `Edit ${member?.name || 'member'} details.`}
   247→   247→          </DialogDescription>
   248→   248→        </DialogHeader>
   249→   249→
   250→   250→        <form onSubmit={handleSubmit} className="space-y-4">
   251→   251→          {/* Name */}
   252→   252→          <div className="space-y-1.5">
   253→   253→            <Label htmlFor="name">Name *</Label>
   254→   254→            <Input
   255→   255→              id="name"
   256→   256→              placeholder="John Doe"
   257→   257→              value={form.name}
   258→   258→              onChange={(e) => setForm({ ...form, name: e.target.value })}
   259→   259→            />
   260→   260→            {errors.name && <p className="text-sm text-destructive">{errors.name}</p>}
   261→   261→          </div>
   262→   262→
   263→   263→          {/* Email */}
   264→   264→          <div className="space-y-1.5">
   265→   265→            <Label htmlFor="email">Email *</Label>
   266→   266→            <Input
   267→   267→              id="email"
   268→   268→              type="email"
   269→   269→              placeholder="john@example.com"
   270→   270→              value={form.email}
   271→   271→              onChange={(e) => setForm({ ...form, email: e.target.value })}
   272→   272→              disabled={mode === 'edit'}
   273→   273→            />
   274→   274→            {errors.email && <p className="text-sm text-destructive">{errors.email}</p>}
   275→   275→          </div>
   276→   276→
   277→   277→          {/* Password is auto-generated — no field shown */}
   278→   278→          {mode === 'add' && (
   279→   279→            <div className="rounded-lg border bg-muted/50 px-3 py-2">
   280→   280→              <p className="text-xs text-muted-foreground">
   281→   281→                A temporary password will be automatically generated and sent to the member&apos;s email. They will be required to change it on first login.
   282→   282→              </p>
   283→   283→            </div>
   284→   284→          )}
   285→   285→
   286→   286→          {/* Job Title */}
   287→   287→          <div className="space-y-1.5">
   288→   288→            <Label htmlFor="jobTitle">Job Title</Label>
   289→   289→            <Input
   290→   290→              id="jobTitle"
   291→   291→              placeholder="Software Engineer"
   292→   292→              value={form.jobTitle}
   293→   293→              onChange={(e) => setForm({ ...form, jobTitle: e.target.value })}
   294→   294→            />
   295→   295→          </div>
   296→   296→
   297→   297→          {/* Phone */}
   298→   298→          <div className="space-y-1.5">
   299→   299→            <Label htmlFor="phone">Phone</Label>
   300→   300→            <Input
   301→   301→              id="phone"
   302→   302→              placeholder="+1 (555) 123-4567"
   303→   303→              value={form.phone}
   304→   304→              onChange={(e) => setForm({ ...form, phone: e.target.value })}
   305→   305→            />
   306→   306→          </div>
   307→   307→
   308→   308→          {/* Skills */}
   309→   309→          <div className="space-y-1.5">
   310→   310→            <Label htmlFor="skills">Skills</Label>
   311→   311→            <Input
   312→   312→              id="skills"
   313→   313→              placeholder="React, Node.js, TypeScript"
   314→   314→              value={form.skills}
   315→   315→              onChange={(e) => setForm({ ...form, skills: e.target.value })}
   316→   316→            />
   317→   317→            <p className="text-xs text-muted-foreground">Separate skills with commas</p>
   318→   318→          </div>
   319→   319→
   320→   320→          {/* Role + Status */}
   321→   321→          <div className="grid grid-cols-2 gap-3">
   322→   322→            <div className="space-y-1.5">
   323→   323→              <Label>Role</Label>
   324→   324→              <Select
   325→   325→                value={form.role}
   326→   326→                onValueChange={(v) => setForm({ ...form, role: v })}
   327→   327→              >
   328→   328→                <SelectTrigger className="w-full">
   329→   329→                  <SelectValue />
   330→   330→                </SelectTrigger>
   331→   331→                <SelectContent>
   332→   332→                  <SelectItem value="MEMBER">Member</SelectItem>
   333→   333→                  <SelectItem value="ADMIN">Admin</SelectItem>
   334→   334→                  <SelectItem value="SUPERADMIN">Superadmin</SelectItem>
   335→   335→                </SelectContent>
   336→   336→              </Select>
   337→   337→            </div>
   338→   338→
   339→   339→            {mode === 'edit' && (
   340→   340→              <div className="space-y-1.5">
   341→   341→                <Label>Status</Label>
   342→   342→                <Select
   343→   343→                  value={form.status}
   344→   344→                  onValueChange={(v) => setForm({ ...form, status: v })}
   345→   345→                >
   346→   346→                  <SelectTrigger className="w-full">
   347→   347→                    <SelectValue />
   348→   348→                  </SelectTrigger>
   349→   349→                  <SelectContent>
   350→   350→                    <SelectItem value="ACTIVE">Active</SelectItem>
   351→   351→                    <SelectItem value="INACTIVE">Inactive</SelectItem>
   352→   352→                    <SelectItem value="ON_LEAVE">On Leave</SelectItem>
   353→   353→                  </SelectContent>
   354→   354→                </Select>
   355→   355→              </div>
   356→   356→            )}
   357→   357→          </div>
   358→   358→
   359→   359→          <DialogFooter className="pt-2">
   360→   360→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   361→   361→              Cancel
   362→   362→            </Button>
   363→   363→            <Button type="submit" disabled={loading}>
   364→   364→              {loading && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   365→   365→              {mode === 'add' ? 'Create Member' : 'Save Changes'}
   366→   366→            </Button>
   367→   367→          </DialogFooter>
   368→   368→        </form>
   369→   369→      </DialogContent>
   370→   370→    </Dialog>
   371→   371→  );
   372→   372→}
