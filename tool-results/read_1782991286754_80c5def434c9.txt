   110→interface ClientFormData {
   111→  name: string;
   112→  email: string;
   113→  company: string;
   114→  address: string;
   115→  phone: string;
   116→  notes: string;
   117→}
   118→
   119→const EMPTY_FORM: ClientFormData = {
   120→  name: '',
   121→  email: '',
   122→  company: '',
   123→  address: '',
   124→  phone: '',
   125→  notes: '',
   126→};
   127→
   128→function ClientFormDialog({
   129→  open,
   130→  onOpenChange,
   131→  mode,
   132→  client,
   133→  onSubmit,
   134→  loading,
   135→}: {
   136→  open: boolean;
   137→  onOpenChange: (open: boolean) => void;
   138→  mode: 'add' | 'edit';
   139→  client: Client | null;
   140→  onSubmit: (data: ClientFormData) => void;
   141→  loading: boolean;
   142→}) {
   143→  const [form, setForm] = useState<ClientFormData>(EMPTY_FORM);
   144→  const [errors, setErrors] = useState<Record<string, string>>({});
   145→
   146→  useEffect(() => {
   147→    if (mode === 'edit' && client) {
   148→      setForm({
   149→        name: client.name,
   150→        email: client.email,
   151→        company: client.company || '',
   152→        address: client.address || '',
   153→        phone: client.phone || '',
   154→        notes: client.notes || '',
   155→      });
   156→    } else {
   157→      setForm(EMPTY_FORM);
   158→    }
   159→    setErrors({});
   160→  }, [mode, client, open]);
   161→
   162→  function validate(): boolean {
   163→    const errs: Record<string, string> = {};
   164→    if (!form.name || form.name.length < 2) errs.name = 'Name must be at least 2 characters';
   165→    if (!form.email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) errs.email = 'Valid email is required';
   166→    setErrors(errs);
   167→    return Object.keys(errs).length === 0;
   168→  }
   169→
   170→  function handleSubmit(e: React.FormEvent) {
   171→    e.preventDefault();
   172→    if (validate()) onSubmit(form);
   173→  }
   174→
   175→  return (
   176→    <Dialog open={open} onOpenChange={(v) => { if (!v) { setErrors({}); } onOpenChange(v); }}>
   177→      <DialogContent className="sm:max-w-md max-h-[90vh] overflow-y-auto">
   178→        <DialogHeader>
   179→          <DialogTitle>{mode === 'add' ? 'Add New Client' : 'Edit Client'}</DialogTitle>
   180→          <DialogDescription>
   181→            {mode === 'add'
   182→              ? 'Create a new client account. A welcome email with temporary password will be sent.'
   183→              : `Edit ${client?.name || 'client'} details.`}
   184→          </DialogDescription>
   185→        </DialogHeader>
   186→
   187→        <form onSubmit={handleSubmit} className="space-y-4">
   188→          <div className="space-y-1.5">
   189→            <Label htmlFor="client-name">Name *</Label>
   190→            <Input
   191→              id="client-name"
   192→              placeholder="John Doe"
   193→              value={form.name}
   194→              onChange={(e) => setForm({ ...form, name: e.target.value })}
   195→            />
   196→            {errors.name && <p className="text-sm text-destructive">{errors.name}</p>}
   197→          </div>
   198→
   199→          <div className="space-y-1.5">
   200→            <Label htmlFor="client-email">Email *</Label>
   201→            <Input
   202→              id="client-email"
   203→              type="email"
   204→              placeholder="client@example.com"
   205→              value={form.email}
   206→              onChange={(e) => setForm({ ...form, email: e.target.value })}
   207→              disabled={mode === 'edit'}
   208→            />
   209→            {errors.email && <p className="text-sm text-destructive">{errors.email}</p>}
   210→          </div>
   211→
   212→          {mode === 'add' && (
   213→            <div className="rounded-lg border bg-muted/50 px-3 py-2">
   214→              <p className="text-xs text-muted-foreground">
   215→                A temporary password will be automatically generated and sent to the client&apos;s email. They will be required to change it on first login.
   216→              </p>
   217→            </div>
   218→          )}
   219→
   220→          <div className="space-y-1.5">
   221→            <Label htmlFor="client-company">Company</Label>
   222→            <Input
   223→              id="client-company"
   224→              placeholder="Acme Corp"
   225→              value={form.company}
   226→              onChange={(e) => setForm({ ...form, company: e.target.value })}
   227→            />
   228→          </div>
   229→
   230→          <div className="space-y-1.5">
   231→            <Label htmlFor="client-address">Address</Label>
   232→            <Input
   233→              id="client-address"
   234→              placeholder="123 Main St, City, State"
   235→              value={form.address}
   236→              onChange={(e) => setForm({ ...form, address: e.target.value })}
   237→            />
   238→          </div>
   239→
   240→          <div className="space-y-1.5">
   241→            <Label htmlFor="client-phone">Phone</Label>
   242→            <Input
   243→              id="client-phone"
   244→              placeholder="+1 (555) 123-4567"
   245→              value={form.phone}
   246→              onChange={(e) => setForm({ ...form, phone: e.target.value })}
   247→            />
   248→          </div>
   249→
   250→          <div className="space-y-1.5">
   251→            <Label htmlFor="client-notes">Notes</Label>
   252→            <Textarea
   253→              id="client-notes"
   254→              placeholder="Internal notes about this client..."
   255→              value={form.notes}
   256→              onChange={(e) => setForm({ ...form, notes: e.target.value })}
   257→              rows={2}
   258→            />
   259→          </div>
   260→
   261→          <DialogFooter className="pt-2">
   262→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   263→              Cancel
   264→            </Button>
   265→            <Button type="submit" disabled={loading}>
   266→              {loading && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   267→              {mode === 'add' ? 'Create Client' : 'Save Changes'}
   268→            </Button>
   269→          </DialogFooter>
   270→        </form>
   271→      </DialogContent>
   272→    </Dialog>
   273→  );
   274→}

   412→/* ------------------------------------------------------------------ */
   413→/*  Main Clients Page                                                  */
   414→/* ------------------------------------------------------------------ */
   415→
   416→export default function ClientsPage() {
   417→  const { data: session } = useSession();
   418→  const userRole = (session?.user as { role?: string })?.role || 'MEMBER';
   419→  const { showError, ErrorDetailDialog } = useApiError();
   420→
   421→  const [clients, setClients] = useState<Client[]>([]);
   422→  const [pagination, setPagination] = useState<Pagination>({ page: 1, limit: 20, total: 0, totalPages: 0 });
   423→  const [loading, setLoading] = useState(true);
   424→
   425→  // Filters
   426→  const [search, setSearch] = useState('');
   427→  const [statusFilter, setStatusFilter] = useState('');
   428→  const [sortBy, setSortBy] = useState('createdAt');
   429→  const [sortOrder, setSortOrder] = useState('desc');
   430→
   431→  // Dialogs
   432→  const [clientDialogOpen, setClientDialogOpen] = useState(false);
   433→  const [clientDialogMode, setClientDialogMode] = useState<'add' | 'edit'>('add');
   434→  const [editingClient, setEditingClient] = useState<Client | null>(null);
   435→  const [dialogLoading, setDialogLoading] = useState(false);
   436→
   437→  const [detailDialogOpen, setDetailDialogOpen] = useState(false);
   438→  const [detailClient, setDetailClient] = useState<Client | null>(null);
   439→
   440→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false);
   441→  const [deletingClient, setDeletingClient] = useState<Client | null>(null);
   442→  const [deleteLoading, setDeleteLoading] = useState(false);
   443→
   444→  // Fetch clients
   445→  const fetchClients = useCallback(async () => {
   446→    setLoading(true);
   447→    try {
   448→      const params = new URLSearchParams();
   449→      if (search) params.set('search', search);
   450→      if (statusFilter) params.set('status', statusFilter);
   451→      params.set('page', pagination.page.toString());
   452→      params.set('limit', pagination.limit.toString());
   453→      params.set('sortBy', sortBy);
   454→      params.set('sortOrder', sortOrder);
   455→
   456→      const res = await fetch(`/api/clients?${params.toString()}`);
   457→      const json = await res.json();
   458→
   459→      if (json.success) {
   460→        setClients(json.data.clients);
   461→        setPagination(json.data.pagination);
   462→      } else {
   463→        showError('Failed to load clients', json.error || 'Unknown error', `URL: /api/clients?${params.toString()}`);
   464→        errorToast({ error: json.error, title: 'Failed to load clients' });
   465→      }
   466→    } catch (err) {
   467→      const errMsg = err instanceof Error ? err.message : String(err);
   468→      showError('Failed to load clients', errMsg, `URL: /api/clients?${params.toString()}`);
   469→      errorToast({ error: 'Failed to load clients' });
   470→    } finally {
   471→      setLoading(false);
   472→    }
   473→  }, [search, statusFilter, pagination.page, pagination.limit, sortBy, sortOrder]);
   474→
   475→  useEffect(() => {
   476→    fetchClients();
   477→  }, [fetchClients]);
   478→
   479→  useEffect(() => {
   480→    setPagination((prev) => ({ ...prev, page: 1 }));
   481→  }, [search, statusFilter, sortBy, sortOrder]);
   482→
   483→  function clearFilters() {
   484→    setSearch('');
   485→    setStatusFilter('');
   486→    setSortBy('createdAt');
   487→    setSortOrder('desc');
   488→  }
   489→
   490→  const hasActiveFilters = !!(search || statusFilter);
   491→
   492→  // Create / Update
   493→  async function handleFormSubmit(data: ClientFormData) {
   494→    setDialogLoading(true);
   495→    try {
   496→      if (clientDialogMode === 'add') {
   497→        const res = await fetch('/api/clients', {
   498→          method: 'POST',
   499→          headers: { 'Content-Type': 'application/json' },
   500→          body: JSON.stringify(data),
   501→        });
   502→        const json = await res.json();
   503→        if (json.success) {
   504→          if (json.data.emailSent) {
   505→            toast.success('Client created! A welcome email with temporary password has been sent.');
   506→          } else {
   507→            toast.success(`Client created! Email not sent: ${json.data.emailError || 'Unknown error'}`);
   508→          }
   509→          setClientDialogOpen(false);
   510→          fetchClients();
   511→        } else {
   512→          showError('Failed to create client', json.error || 'Unknown error');
   513→          errorToast({ error: json.error, title: 'Failed to create client' });
   514→        }
   515→      } else if (editingClient) {
   516→        const updateData: Record<string, unknown> = {};
   517→        if (data.name !== editingClient.name) updateData.name = data.name;
   518→        if (data.company !== (editingClient.company || '')) updateData.company = data.company || null;
   519→        if (data.address !== (editingClient.address || '')) updateData.address = data.address || null;
   520→        if (data.phone !== (editingClient.phone || '')) updateData.phone = data.phone || null;
   521→        if (data.notes !== (editingClient.notes || '')) updateData.notes = data.notes || null;
   522→
   523→        if (Object.keys(updateData).length === 0) {
   524→          toast.info('No changes to save');
   525→          setClientDialogOpen(false);
   526→          return;
   527→        }
   528→
   529→        const res = await fetch(`/api/clients/${editingClient.id}`, {
   530→          method: 'PUT',
   531→          headers: { 'Content-Type': 'application/json' },
   532→          body: JSON.stringify(updateData),
   533→        });
   534→        const json = await res.json();
   535→        if (json.success) {
   536→          toast.success('Client updated successfully');
   537→          setClientDialogOpen(false);
   538→          fetchClients();
   539→        } else {
   540→          showError('Failed to update client', json.error || 'Unknown error', `URL: /api/clients/${editingClient.id}`);
   541→          errorToast({ error: json.error, title: 'Failed to update client' });
   542→        }
   543→      }
   544→    } catch (formError) {
   545→      const errMsg = formError instanceof Error ? formError.message : String(formError);
   546→      showError('Failed to save client', errMsg);
   547→      errorToast({ error: errMsg, title: 'Request failed' });
   548→    } finally {
   549→      setDialogLoading(false);
   550→    }
   551→  }
   552→
   553→  // Delete
   554→  async function handleDelete() {
   555→    if (!deletingClient) return;
   556→    setDeleteLoading(true);
   557→    try {
   558→      const res = await fetch(`/api/clients/${deletingClient.id}`, { method: 'DELETE' });
   559→      const json = await res.json();
   560→      if (json.success) {
   561→        toast.success('Client deleted successfully');
   562→        setDeleteDialogOpen(false);
   563→        setDeletingClient(null);
   564→        fetchClients();
   565→      } else {
   566→        showError('Failed to delete client', json.error || 'Unknown error', `URL: /api/clients/${deletingClient.id}`);
   567→        errorToast({ error: json.error, title: 'Failed to delete client' });
   568→      }
   569→    } catch (err) {
   570→      const errMsg = err instanceof Error ? err.message : String(err);
   571→      showError('Failed to delete client', errMsg, `URL: /api/clients/${deletingClient?.id}`);
   572→      errorToast({ error: errMsg, title: 'Delete client failed' });
   573→    } finally {
   574→      setDeleteLoading(false);
   575→    }
   576→  }
   577→
   578→  // Page navigation
   579→  function goToPage(page: number) {
   580→    setPagination((prev) => ({ ...prev, page }));
   581→  }
   582→
   583→  return (
   584→    <div className="space-y-6">
   585→      {/* Header */}
   586→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   587→        <div>
   588→          <h1 className="text-2xl font-bold tracking-tight">Clients</h1>
   589→          <p className="text-muted-foreground text-sm mt-1">
   590→            Manage your clients and their project access.
   591→          </p>
   592→        </div>
   593→        {userRole === 'SUPERADMIN' && (
   594→          <Button
   595→            onClick={() => {
   596→              setClientDialogMode('add');
   597→              setEditingClient(null);
   598→              setClientDialogOpen(true);
   599→            }}
   600→          >
   601→            <Plus className="mr-2 h-4 w-4" />
