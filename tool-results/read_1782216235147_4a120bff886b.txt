    14→// ===== Document type mapping =====
    15→const DOC_TYPE_MAP: Record<string, string> = {
    16→  "/prd": "prd",
    17→  "/trd": "trd",
    18→  "/flow": "flow",
    19→  "/ux": "ux",
    20→  "/schema": "schema",
    21→  "/plan": "plan",
    22→};
    23→
    24→// Document keyword signatures for detecting update-type messages
    25→const DOC_SIGNATURES = [
    26→  { label: "prd", keywords: ["Product Requirements Document", "Executive Summary", "Feature Requirements", "User Stories", "Target Audience", "Product Vision", "Non-Functional Requirements"] },
    27→  { label: "trd", keywords: ["Technical Requirements Document", "Architecture Overview", "Technology Stack", "API Specification", "Security Requirements", "Performance Requirements"] },
    28→  { label: "flow", keywords: ["Application Flow Document", "User Journey", "Screen Flow", "Navigation Architecture", "State Management", "Interaction Patterns"] },
    29→  { label: "ux", keywords: ["UI/UX Design Brief", "Design Principles", "Design System", "Color Palette", "Typography", "Component Guidelines", "Accessibility"] },
    30→  { label: "schema", keywords: ["Backend Schema Document", "Entity Relationship", "Schema Definitions", "Enum Types", "Data Integrity Rules", "Seed Data"] },
    31→  { label: "plan", keywords: ["Implementation Plan", "Phase Breakdown", "Task Breakdown", "Sprint Planning", "Resource Requirements", "Risk Register", "Quality Gates"] },
    32→];
    33→
    34→function detectDocTypeFromContent(content: string): string | null {
    35→  if (content.length < 1500) return null;
    36→  const lower = content.toLowerCase();
    37→  let bestMatch = "";
    38→  let bestCount = 0;
    39→  for (const sig of DOC_SIGNATURES) {
    40→    const count = sig.keywords.filter(kw => lower.includes(kw.toLowerCase())).length;
    41→    if (count >= 3 && count > bestCount) {
    42→      bestCount = count;
    43→      bestMatch = sig.label;
    44→    }
    45→  }
    46→  return bestMatch || null;
    47→}

    57→async function hasProjectAccess(userId: string, userRole: string, projectId: string): Promise<boolean> {
    58→  if (userRole === "SUPERADMIN") return true;

    61→    sql: `SELECT id FROM "ProjectMember" WHERE "projectId" = ? AND "userId" = ? AND "removedAt" IS NULL`,
    62→    args: [projectId, userId],
    63→  });
    64→  return result.rows.length > 0;
    65→}

   173→// GET /api/ai/chat — Paginated chat history
   174→export async function GET(request: NextRequest) {
   175→  try {
   176→    const user = await getAuthUser(request);
   177→    if (!user) {
   178→      return NextResponse.json({ success: false, error: "Authentication required" }, { status: 401 });
   179→    }
   180→
   181→    const { searchParams } = new URL(request.url);
   182→    const projectId = searchParams.get("projectId");
   183→    if (!projectId) {
   184→      return NextResponse.json({ success: false, error: "projectId is required" }, { status: 400 });
   185→    }
   186→
   187→    // Check project access
   188→    const canAccess = await hasProjectAccess(user.id, user.role, projectId);
   189→    if (!canAccess) {
   190→      return NextResponse.json({ success: false, error: "You don't have access to this project" }, { status: 403 });
   191→    }
   192→
   193→    const page = Math.max(1, parseInt(searchParams.get("page") || "1"));
   194→    const limit = Math.min(100, Math.max(1, parseInt(searchParams.get("limit") || "50")));
   195→
   196→    const client = getTursoClient();
   197→    const offset = (page - 1) * limit;
   198→
   199→    let messagesResult;
   200→
   201→    if (user.role === "SUPERADMIN") {
   202→      // SUPERADMIN sees all messages — newest first (client will reverse for display)
   203→      messagesResult = await client.execute({
   204→        sql: `SELECT m.*, u.name as userName
   205→              FROM "AiChat" m
   206→              LEFT JOIN "User" u ON m."userId" = u.id
   207→              WHERE m."projectId" = ?
   208→              ORDER BY m."timestamp" DESC
   209→              LIMIT ? OFFSET ?`,
   210→        args: [projectId, limit, offset],
   211→      });
   212→    } else {
   213→      // Others see only own messages + assistant messages — newest first
   214→      messagesResult = await client.execute({
   215→        sql: `SELECT m.*, u.name as userName
   216→              FROM "AiChat" m
   217→              LEFT JOIN "User" u ON m."userId" = u.id
   218→              WHERE m."projectId" = ? AND (m."userId" = ? OR m."role" = 'assistant')
   219→              ORDER BY m."timestamp" DESC
   220→              LIMIT ? OFFSET ?`,
   221→        args: [projectId, user.id, limit, offset],
   222→      });
   223→    }
   224→
   225→    const countResult = await client.execute({
   226→      sql: `SELECT COUNT(*) as total FROM "AiChat" WHERE "projectId" = ?`,
   227→      args: [projectId],
   228→    });
   229→    const total = Number(countResult.rows[0].total);
   230→
   231→    const messages = messagesResult.rows.map((row) => ({
   232→      id: row.id,
   233→      userId: row.userId,
   234→      projectId: row.projectId,
   235→      role: row.role,
   236→      content: row.content,
   237→      timestamp: row.timestamp,
   238→      userName: row.userName,
   239→    }));
   240→
   241→    return NextResponse.json({
   242→      success: true,
   243→      data: {
   244→        messages,
   245→        pagination: {
   246→          page,
   247→          limit,
   248→          total,
   249→          totalPages: Math.ceil(total / limit),
   250→        },
   251→      },
   252→    });
   253→  } catch (error) {
   254→    console.error("[GET /api/ai/chat] Error:", error);
   255→    return NextResponse.json({ success: false, error: "Internal server error" }, { status: 500 });
   256→  }
   257→}

   263→// POST /api/ai/chat — Send message and get AI response (with agentic tool-calling loop)
   264→export async function POST(request: NextRequest) {
   265→  try {
   266→    const user = await getAuthUser(request);
   267→    if (!user) {
   268→      return NextResponse.json({ success: false, error: "Authentication required" }, { status: 401 });
   269→    }
   270→
   271→    const rawBody = await request.json();
   272→    const { projectId, content, files, fileData, fileName, fileType } = rawBody as { projectId: string; content: string; files?: unknown[]; fileData?: string; fileName?: string; fileType?: string };
   273→
   274→    if (!projectId || !content) {
   275→      return NextResponse.json({ success: false, error: "projectId and content are required" }, { status: 400 });
   276→    }
   277→
   278→    const attachedImages: { data: string; type: string; name: string }[] = [];
   279→    if (files && Array.isArray(files) && files.length > 0) {
   280→      // New format: array of { data, type, name }
   281→      for (const f of files) {
   282→        if (f.data && f.type && f.type.startsWith("image/")) {
   283→          attachedImages.push({ data: f.data, type: f.type, name: f.name || "image" });
   284→        }
   285→      }
   286→    } else if (fileData && fileType && fileType.startsWith("image/")) {
   287→      // Legacy single-file format
   288→      attachedImages.push({ data: fileData, type: fileType, name: fileName || "image" });
   289→    }
   290→    const hasImages = attachedImages.length > 0;
   291→
   292→    // Check project access
   293→    const canAccess = await hasProjectAccess(user.id, user.role, projectId);
   294→    if (!canAccess) {
   295→      return NextResponse.json({ success: false, error: "You don't have access to this project" }, { status: 403 });
   296→    }
   297→
   298→    const client = getTursoClient();
   299→    const ip = getClientIp(request);
   300→
   301→    // Auto-migrate AI tables if they don't exist
   302→    await ensureAiTables(client);
   303→
   304→    // Save user message
   305→    const userMsgId = crypto.randomUUID();
   306→    await client.execute({
   307→      sql: `INSERT INTO "AiChat" (id, "userId", "projectId", "role", "content", "timestamp")
   308→            VALUES (?, ?, ?, 'user', ?, datetime('now'))`,
   309→      args: [userMsgId, user.id, projectId, content],
   310→    });
   311→
   312→    // Detect command
   313→    const trimmed = content.trim();
   314→    const command = trimmed.startsWith("/") ? trimmed.split(/[\s\n]/)[0] : undefined;
   315→
   316→    // Detect command type (needed before loading history to determine history limit)
   317→    const isDocCommand = !!command && ["/docs", "/prd", "/trd", "/flow", "/ux", "/schema", "/plan", "/init"].includes(command);
   318→
   319→    // Load recent chat messages for context
   320→    // For doc commands: only last 6 messages (saves tokens, avoids bloated context)
   321→    // For regular chat: last 20 messages for natural conversation flow
   322→    const historyLimit = isDocCommand ? 6 : 20;
   323→    const historyResult = await client.execute({
   324→      sql: `SELECT role, content FROM "AiChat"
   325→            WHERE "projectId" = ?
   326→            ORDER BY "timestamp" DESC
   327→            LIMIT ?`,
   328→      args: [projectId, historyLimit],
   329→    });
   330→    const chatHistory = historyResult.rows.reverse().map((row) => ({
   331→      role: row.role,
   332→      content: row.content,
   333→    }));
   334→
   335→    // Load user name from DB
   336→    let userName = user.name;
   337→    try {
   338→      const userResult = await client.execute({
   339→        sql: `SELECT name FROM "User" WHERE id = ?`,
   340→        args: [user.id],
   341→      });
   342→      if (userResult.rows.length > 0) {
   343→        userName = userResult.rows[0].name as string;
   344→      }
   345→    } catch {
   346→      // Use token name as fallback
   347→    }
   348→
   349→    // Load project info (with status, priority, deadline)
   350→    const projectResult = await client.execute({
   351→      sql: `SELECT name, description, "clientName", status, priority, deadline FROM "Project" WHERE id = ?`,
   352→      args: [projectId],
   353→    });
   354→    const project = projectResult.rows[0];
   355→
   356→    // Load team member count for this project
   357→    let teamCount: number | undefined;
   358→    try {
   359→      const teamResult = await client.execute({
   360→        sql: `SELECT COUNT(*) as count FROM "ProjectMember" WHERE "projectId" = ? AND "removedAt" IS NULL`,
   361→        args: [projectId],
   362→      });
   363→      teamCount = Number(teamResult.rows[0].count);
   364→    } catch {
   365→      // Non-critical
   366→    }
   367→
   368→    // Load protocol steps for /docs command
   369→    let protocolSteps: { title: string; description?: string; commandTag?: string }[] = [];
   370→    if (command === "/docs") {
   371→      try {
   372→        const stepsResult = await client.execute({
   373→          sql: `SELECT ps.title, ps.description, ps."commandTag"
   374→                FROM "AiProtocolStep" ps
   375→                JOIN "AiProtocol" p ON ps."protocolId" = p.id
   376→                WHERE (p."isGlobal" = 1 OR p."projectId" = ?)
   377→                ORDER BY ps."stepOrder" ASC`,
   378→          args: [projectId],
   379→        });
   380→        protocolSteps = stepsResult.rows.map((row) => ({
   381→          title: row.title as string,
   382→          description: (row.description as string) || undefined,
   383→          commandTag: (row.commandTag as string) || undefined,
   384→        }));
   385→      } catch (err) {
   386→        console.error("[POST /api/ai/chat] Error loading protocol steps:", err);
   387→      }
   388→    }
   389→
   390→    // Resolve model: per-project setting > auto-selection > env var > global default
   391→    let projectModel: string | null = null;
   392→    try {
   393→      const modelSetting = await client.execute({
   394→        sql: `SELECT value FROM "Settings" WHERE key = ?`,
   395→        args: [`ai_model:${projectId}`],
   396→      });
   397→      if (modelSetting.rows.length > 0 && modelSetting.rows[0].value) {
   398→        projectModel = modelSetting.rows[0].value as string;
   399→      }
   400→    } catch {
   401→      // Settings table might not exist yet, fall back to default
   402→    }
   403→
   404→    // Auto-select model for doc commands if user hasn't explicitly chosen one
   405→    // GLM-4-Flash: FREE permanently, 128K context, 16K output, function calling support
   406→    // Perfect for doc generation — large output window + free + reliable
   407→    const DOC_AUTO_MODEL = "glm-4-flash";
   408→    let activeModel: string;
   409→    let modelAutoSelected = false;
   410→    if (projectModel) {
   411→      // User explicitly chose a model — respect it
   412→      activeModel = projectModel;
   413→    } else if (isDocCommand) {
   414→      // Auto-select the best model for doc generation
   415→      activeModel = DOC_AUTO_MODEL;
   416→      modelAutoSelected = true;
   417→    } else {
   418→      activeModel = getGlobalDefaultModel();
   419→    }
   420→
   421→    // Build system prompt with rich context
   422→    const systemPrompt = buildSystemPrompt({
   423→      userName,
   424→      userRole: user.role,
   425→      projectName: project?.name as string | undefined,
   426→      projectDescription: project?.description as string | undefined,
   427→      projectClient: project?.clientName as string | undefined,
   428→      projectStatus: project?.status as string | undefined,
   429→      projectDeadline: project?.deadline as string | null | undefined,
   430→      projectPriority: project?.priority as string | undefined,
   431→      teamCount,
   432→      protocolSteps: protocolSteps.length > 0 ? protocolSteps : undefined,
   433→      command,
   434→    });
   435→
   436→    // ===== Determine max_tokens based on model capability =====
   437→    // Use the model's actual max_output_tokens — NO artificial caps.
   438→    // For /docs: use the full model output (important: GLM-4-Flash has 16K, which is enough for one doc)
   439→    // For individual doc commands: use 90% of max to leave room for formatting
   440→    // For regular chat: use 4096 (plenty for conversational responses)
   441→    const modelCap = getModelCapability(activeModel);
   442→    const modelMaxTokens = modelCap?.maxOutputTokens || 4096;
   443→    let maxTokens: number;
   444→    if (command === "/docs" || command === "/init") {
   445→      maxTokens = modelMaxTokens; // Use full output for /docs and /init
   446→    } else if (isDocCommand) {
   447→      maxTokens = Math.floor(modelMaxTokens * 0.9); // 90% for individual docs
   448→    } else {
   449→      maxTokens = 4096; // Regular chat doesn't need huge output
   450→    }
   451→
   452→    // Get tools for this user's role
   453→    // For doc commands: ONLY pass list_projects and get_project_info (read-only data tools)
   454→    // knowledge_research is REMOVED from doc commands — it wastes agentic loop rounds
   455→    // create/update/add_member are irrelevant for doc generation
   456→    // For /init: add save_github_config tool so AI can save credentials
   457→    let availableTools = getToolsForRole(user.role);
   458→    if (isDocCommand && command !== "/init") {
   459→      availableTools = availableTools.filter((tool) =>
   460→        ["list_projects", "get_project_info"].includes(tool.function.name)
   461→      );
   462→    } else if (command === "/init") {
   463→      availableTools = availableTools.filter((tool) =>
   464→        ["save_github_config", "save_database_config"].includes(tool.function.name)
   465→      );
   466→    }
   467→
   468→    // For vision, use the appropriate vision-capable model
   469→    const visionModel = hasImages ? getVisionModel(activeModel) : activeModel;
   470→
   471→    // ===== Auto-Route: Check if prompt fits the selected model =====
   472→    let modelAutoRouted = false;
   473→    let modelRouteReason = "";
   474→    try {
   475→      const estimatedTokens = estimatePromptTokens(
   476→        [{ role: "system", content: systemPrompt }, ...chatHistory.map((m) => ({ role: String(m.role), content: String(m.content) }))],
   477→        availableTools.length > 0 ? availableTools : undefined
   478→      );
   479→      const route = findBestModelForPrompt(estimatedTokens, activeModel, {
   480→        tools: !hasImages && availableTools.length > 0,
   481→        vision: hasImages,
   482→      });
   483→      if (route.autoRouted) {
   484→        console.log(`[AI Auto-Route] ${route.reason}`);
   485→        console.log(`[AI Auto-Route] Switching: ${activeModel} -> ${route.model}`);
   486→        activeModel = route.model;
   487→        modelAutoRouted = true;
   488→        modelRouteReason = route.reason;
   489→      } else {
   490→        console.log(`[AI Model] Using ${activeModel} (no routing needed, ~${estimatedTokens.toLocaleString()} tokens estimated)`);
   491→      }
   492→    } catch (routeErr) {
   493→      console.error("[AI Auto-Route] Routing check failed, using selected model:", routeErr);
   494→    }
   495→
   496→    // ===== Agentic Loop Setup =====
   497→    let aiText: string;
   498→    let aiError = false;
   499→    const toolExecutions: {
   500→      toolName: string;
   501→      label: string;
   502→      icon: string;
   503→      status: "success" | "error" | "running";
   504→      displayMessage: string;
   505→    }[] = [];

   507→    // Tool executor context
   508→    const executorCtx = {
   509→      userId: user.id,
   510→      userRole: user.role,
   511→      userName: userName || user.name,
   512→      tursoClient: client,
   513→    };
   514→

   516→    const aiMessages: AiMessage[] = [
   517→      { role: "system", content: systemPrompt },
   518→    ];
   519→
   520→    for (const msg of chatHistory) {
   521→      aiMessages.push({
   522→        role: msg.role as "user" | "assistant",
   523→        content: msg.content,
   524→      });

   526→
   527→    const MAX_TOOL_ROUNDS = isDocCommand ? 4 : 5; // Reduced from 8 — /docs should WRITE, not loop on tools

   563→        // ===== Agentic Tool-Calling Loop with Provider Fallback =====

   593→                messages,
   594→                model: fallbackModel,
   595→                maxTokens: maxTok,

" (auto-routed)" : ""} command=${command || "none"} has_tools=${!!result.tool_calls} content_len=${result.content?.length || 0} success=${result.success} error=${result.error || "none"}`);
   653→

   676→            role: "assistant",
   677→            content: result.content || "",

   706→              role: "tool",
   707→              tool_call_id: toolCall.id,
   708→              content: toolResult.result,
   709→              name: toolName,
   710→            });
   711→          }
   712→
   713→          // Loop continues — AI will see tool results and decide next action

   724→        }

   732→              const retryMessages: AiMessage[] = [

   743→                finalContent = retryResult.content;
   744→                console.log(`[AI Fallback] Retry succeeded, content_len=${retryResult.content.length}`);
   745→              } else {
   746→                console.error(`[AI Fallback] Retry also failed: ${retryResult.error}`);
   747→                finalContent = `I received your \`${command}\` command but all AI models returned an error after trying multiple providers.\n\n**Error:** ${retryResult.error || "All providers failed"}\n\n**Suggestions:**\n- Wait a moment and try again (rate limits reset periodically)\n- Try a simpler command like \`/help\`\n- Check that your AI provider API keys are valid in Vercel settings`;
   748→              }
   749→            } catch (retryErr) {
   750→              console.error("[AI Fallback] Retry error:", retryErr);
   751→              finalContent = `I received your \`${command}\` command but encountered an error. Please try again.`;
   752→            }
   753→          } else {
   754→            finalContent = "I'm here to help! What would you like me to do?";
   755→          }
   756→        }

   758→        aiText = typeof finalContent === "string" ? finalContent : JSON.stringify(finalContent);

   761→      console.error("[POST /api/ai/chat] AI call error:", err);
   762→      aiText = "I encountered an unexpected error with the AI service. Please try again in a moment.";
   763→      aiError = true;

   771→        { pattern: /(?:I[''](ve| have| will| am| just))?\s*(?:pushed|pushing|committed|deployed|deploying|created|creating|built|building|cloned|cloning|merged|merging)\s+(?:the\s+)?(?:code|changes|files|project|application|app|repo|repository|to\s+GitHub|to\s+the\s+repo)/gi,

   776→          replacement: "[Note: This was not actually executed.]" },

   778→
   779→      // Only apply filter if NO github/file tools were used

   781→      if (!usedGitTools) {

   799→    // ===== Auto-save document to ProjectDocument =====

   866→      userId: user.id,
   867→      action: "AI_CHAT_MESSAGE",
   868→      details: `Sent AI chat message in project${project?.name ? ` "${project.name}"` : ""}${command ? ` (command: ${command})` : ""}${toolSummary}`,

   955→    // ===== z.ai Bridge: Check if /init should include bridge info =====

   957→    let zaiBridge: {

   959→      chatUrl: string;
   960→      context: string;
   961→      modelName: string;
   962→      documentsFound: number;
   963→      isNewChat: boolean;

   967→      docsSource?: string;

   969→
   970→    if (command === "/init") {
   971→      try {
   972→        // Count project documents (no minimum required — send whatever exists)

   974→          sql: `SELECT COUNT(*) as count FROM "ProjectDocument" WHERE "projectId" = ? AND "docType" IN ('prd', 'trd', 'flow', 'ux', 'schema', 'plan')`,
