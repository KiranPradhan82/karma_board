    98→  userId: string;
    99→  role: string;
   100→  joinedAt: string;

   112→interface AvailableMember {

   119→
   120→export default function ProjectDetailPage() {
   121→  const { data: session } = useSession();
   122→  const router = useRouter();
   123→  const params = useParams();
   124→  const projectId = params.id as string;
   125→  const user = session?.user as { role?: string; id?: string } | undefined;
   126→  const isAdmin = user?.role === "ADMIN" || user?.role === "SUPERADMIN";
   127→  const isSuperAdmin = user?.role === "SUPERADMIN";
   128→
   129→  const [project, setProject] = useState<ProjectDetail | null>(null);
   130→  const [team, setTeam] = useState<TeamMember[]>([]);
   131→  const [loading, setLoading] = useState(true);
   132→
   133→  // Edit project dialog
   134→  const [editOpen, setEditOpen] = useState(false);
   135→  const [editLoading, setEditLoading] = useState(false);
   136→  const [editForm, setEditForm] = useState({
   137→    name: "",
   138→    description: "",
   139→    priority: "MEDIUM",
   140→    status: "ACTIVE",
   141→    clientName: "",
   142→    color: "#6366f1",
   143→    deadline: "",
   144→  });
   145→
   146→  // Add member dialog
   147→  const [addMemberOpen, setAddMemberOpen] = useState(false);
   148→  const [addMemberLoading, setAddMemberLoading] = useState(false);
   149→  const [availableMembers, setAvailableMembers] = useState<AvailableMember[]>([]);
   150→  const [selectedUserId, setSelectedUserId] = useState("");
   151→  const [selectedRole, setSelectedRole] = useState("MEMBER");
   152→  const [loadingMembers, setLoadingMembers] = useState(false);
   153→
   154→  const fetchProject = useCallback(async () => {
   155→    try {
   156→      const res = await fetch(`/api/projects/${projectId}`);
   157→      const data = await res.json();
   158→      if (data.success) {
   159→        setProject(data.data.project);
   160→      } else {
   161→        errorToast({ error: data.error, title: "Failed to load project" });
   162→        router.push("/dashboard/projects");
   163→      }
   164→    } catch {
   165→      errorToast({ error: "Failed to load project" });
   166→      router.push("/dashboard/projects");
   167→    } finally {
   168→      setLoading(false);
   169→    }
   170→  }, [projectId, router]);
   171→
   172→  const fetchTeam = useCallback(async () => {
   173→    try {
   174→      const res = await fetch(`/api/projects/${projectId}/team`);
   175→      const data = await res.json();
   176→      if (data.success) {
   177→        setTeam(data.data.members);
   178→      }
   179→    } catch {
   180→      // Silently fail — team data is secondary
   181→    }
   182→  }, [projectId]);
   183→
   184→  useEffect(() => {
   185→    setLoading(true);
   186→    fetchProject();
   187→    fetchTeam();
   188→  }, [fetchProject, fetchTeam]);
   189→
   190→  const handleEditSave = async () => {
   191→    if (!editForm.name.trim()) {
   192→      toast.error("Project name is required");
   193→      return;
   194→    }
   195→    setEditLoading(true);
   196→    try {
   197→      const body: Record<string, unknown> = {
   198→        name: editForm.name,
   199→        description: editForm.description || null,
   200→        priority: editForm.priority,
   201→        status: editForm.status,
   202→        clientName: editForm.clientName || null,
   203→        color: editForm.color,
   204→        deadline: editForm.deadline ? new Date(editForm.deadline).toISOString() : null,
   205→      };
   206→
   207→      const res = await fetch(`/api/projects/${projectId}`, {
   208→        method: "PATCH",
   209→        headers: { "Content-Type": "application/json" },
   210→        body: JSON.stringify(body),
   211→      });
   212→      const data = await res.json();
   213→      if (data.success) {
   214→        toast.success("Project updated");
   215→        setEditOpen(false);
   216→        fetchProject();
   217→      } else {
   218→        errorToast({ error: data.error, title: "Failed to update project" });
   219→      }
   220→    } catch {
   221→      errorToast({ error: "Failed to update project" });
   222→    } finally {
   223→      setEditLoading(false);
   224→    }
   225→  };
   226→
   227→  const openEditDialog = () => {
   228→    if (!project) return;
   229→    setEditForm({
   230→      name: project.name,
   231→      description: project.description || "",
   232→      priority: project.priority,
   233→      status: project.status,
   234→      clientName: project.clientName || "",
   235→      color: project.color || "#6366f1",
   236→      deadline: project.deadline ? project.deadline.split("T")[0] : "",
   237→    });
   238→    setEditOpen(true);
   239→  };
   240→
   241→  const openAddMemberDialog = async () => {
   242→    setAddMemberOpen(true);
   243→    setSelectedUserId("");
   244→    setSelectedRole("MEMBER");
   245→    setLoadingMembers(true);
   246→    try {
   247→      // Fetch all active members (not deleted)
   248→      const res = await fetch("/api/members?limit=200&status=ACTIVE");
   249→      const data = await res.json();
   250→      if (data.success) {
   251→        // Filter out members already in this project
   252→        const existingIds = new Set(team.map((m) => m.userId));
   253→        const available = data.data.members.filter(
   254→          (m: AvailableMember) => !existingIds.has(m.id)
   255→        );
   256→        setAvailableMembers(available);
   257→      }
   258→    } catch {
   259→      errorToast({ error: "Failed to load available members" });
   260→    } finally {
   261→      setLoadingMembers(false);
   262→    }
   263→  };
   264→
   265→  const handleAddMember = async () => {
   266→    if (!selectedUserId) {
   267→      toast.error("Please select a team member");
   268→      return;
   269→    }
   270→    setAddMemberLoading(true);
   271→    try {
   272→      const res = await fetch(`/api/projects/${projectId}/team`, {
   273→        method: "POST",
   274→        headers: { "Content-Type": "application/json" },
   275→        body: JSON.stringify({ userId: selectedUserId, role: selectedRole }),
   276→      });
   277→      const data = await res.json();
   278→      if (data.success) {
   279→        toast.success("Team member added");
   280→        setAddMemberOpen(false);
   281→        fetchTeam();
   282→        fetchProject(); // Update member count
   283→      } else {
   284→        errorToast({ error: data.error, title: "Failed to add member" });
   285→      }
   286→    } catch {
   287→      errorToast({ error: "Failed to add member" });
   288→    } finally {
   289→      setAddMemberLoading(false);
   290→    }
   291→  };
   292→
   293→  const handleRemoveMember = async (member: TeamMember) => {
   294→    if (member.role === "LEAD" && !isSuperAdmin) {
   295→      toast.error("Only super admin can remove a team lead");
   296→      return;
   297→    }
   298→    if (!confirm(`Remove ${member.user.name} from this project?`)) return;
   299→    try {
   300→      const res = await fetch(`/api/projects/${projectId}/team`, {
   301→        method: "DELETE",
   302→        headers: { "Content-Type": "application/json" },
   303→        body: JSON.stringify({ userId: member.userId }),
   304→      });
   305→      const data = await res.json();
   306→      if (data.success) {
   307→        toast.success("Member removed");
   308→        fetchTeam();
   309→        fetchProject();
   310→      } else {
   311→        errorToast({ error: data.error, title: "Failed to remove member" });
   312→      }
   313→    } catch {
   314→      errorToast({ error: "Failed to remove member" });
   315→    }
   316→  };
   317→
   318→  const handleChangeRole = async (member: TeamMember, newRole: string) => {
   319→    // Multiple LEADs allowed — no need to check for existing lead
   320→    // Only ADMIN+ global role can be LEAD in a project
   321→    try {
   322→      const memberRes = await fetch(`/api/members/${member.userId}`);
   323→      const memberData = await memberRes.json();
   324→      if (memberData.success && newRole === "LEAD") {
   325→        const globalRole = memberData.data.role;
   326→        if (globalRole !== "ADMIN" && globalRole !== "SUPERADMIN") {
   327→          toast.error("Only admins can be assigned as Team Lead");
   328→          return;
   329→        }
   330→      }
   331→    } catch {
   332→      // If fetch fails, let the backend handle the validation
   333→    }
   334→
   335→    try {
   336→      const res = await fetch(`/api/projects/${projectId}/team/${member.userId}/role`, {
   337→        method: "PATCH",
   338→        headers: { "Content-Type": "application/json" },
   339→        body: JSON.stringify({ role: newRole }),
   340→      });
   341→      const data = await res.json();
   342→      if (data.success) {
   343→        toast.success("Role updated");
   344→        fetchTeam();
   345→        fetchProject(); // Refresh member count in header
   346→      } else {
   347→        errorToast({ error: data.error, title: "Failed to change role" });
   348→      }
   349→    } catch {
   350→      errorToast({ error: "Failed to change role" });
   351→    }
   352→  };
   353→
   354→  const handleArchive = async () => {
   355→    if (!project || !confirm(`Archive "${project.name}"?`)) return;
   356→    try {
   357→      const res = await fetch(`/api/projects/${projectId}`, { method: "DELETE" });
   358→      const data = await res.json();
   359→      if (data.success) {
   360→        toast.success("Project archived");
   361→        router.push("/dashboard/projects");
   362→      } else {
   363→        errorToast({ error: data.error, title: "Failed to archive project" });
   364→      }
   365→    } catch {
   366→      errorToast({ error: "Failed to archive project" });
   367→    }
   368→  };
   369→
   370→  function formatDate(dateStr: string | null) {
   371→    if (!dateStr) return "—";
   372→    try {
   373→      return new Date(dateStr).toLocaleDateString("en-GB", {
   374→        day: "numeric",
   375→        month: "short",
   376→        year: "numeric",
   377→      });
   378→    } catch {
   379→      return dateStr;
   380→    }
   381→  }
   382→
   383→  function isOverdue() {
   384→    if (!project?.deadline || project.status === "COMPLETED" || project.status === "ARCHIVED") return false;
   385→    return new Date(project.deadline) < new Date();
   386→  }
   387→
   388→  if (loading) {
   389→    return (
   390→      <div className="flex items-center justify-center py-20">
   391→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   392→      </div>
   393→    );
   394→  }
   395→
   396→  if (!project) {
   397→    return (
   398→      <div className="flex flex-col items-center justify-center py-20">
   399→        <p className="text-muted-foreground">Project not found</p>
   400→        <Button variant="link" onClick={() => router.push("/dashboard/projects")}>
   401→          Back to Projects
   402→        </Button>
   403→      </div>
   404→    );
   405→  }
   406→
   407→  const overdue = isOverdue();
   408→  const priority = priorityConfig[project.priority] || priorityConfig.MEDIUM;
   409→  const status = statusConfig[project.status] || statusConfig.ACTIVE;
   410→  const leads = team.filter((m) => m.role === "LEAD");
   411→  const otherMembers = team.filter((m) => m.role !== "LEAD");
   412→
   413→  const handlePromoteToLead = async (member: TeamMember) => {
   414→    // Multiple leads allowed — just promote directly
   415→    await handleChangeRole(member, "LEAD");
   416→  };
   417→
   418→  return (
   419→    <div className="space-y-6">
   420→      {/* Header */}
   421→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   422→        <div className="flex items-center gap-3">
   423→          <Button variant="ghost" size="icon" onClick={() => router.push("/dashboard/projects")}>
   424→            <ArrowLeft className="h-4 w-4" />
   425→          </Button>
   426→          <div className="flex items-center gap-2">
   427→            <div
   428→              className="h-3 w-3 rounded-full shrink-0"
   429→              style={{ backgroundColor: project.color || "#6366f1" }}
   430→            />
   431→            <h1 className="text-2xl font-bold tracking-tight">{project.name}</h1>
   432→          </div>
   433→        </div>
   434→        {isAdmin && (
   435→          <div className="flex items-center gap-2">
   436→            <Button variant="outline" size="sm" onClick={openEditDialog}>
   437→              <Pencil className="mr-2 h-4 w-4" />
   438→              Edit
   439→            </Button>
   440→            {isSuperAdmin && (
   441→              <Button variant="outline" size="sm" className="text-destructive" onClick={handleArchive}>
   442→                <Trash2 className="mr-2 h-4 w-4" />
   443→                Archive
   444→              </Button>
   445→            )}
   446→          </div>
   447→        )}
   448→      </div>
   449→
   450→      {/* Project Info Cards */}
   451→      <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
   452→        <Card>
   453→          <CardContent className="p-4">
   454→            <p className="text-xs text-muted-foreground mb-1">Status</p>
   455→            <Badge variant="secondary" className={status.className}>
   456→              {status.label}
   457→            </Badge>
   458→          </CardContent>
   459→        </Card>
   460→        <Card>
   461→          <CardContent className="p-4">
   462→            <p className="text-xs text-muted-foreground mb-1">Priority</p>
   463→            <Badge variant="secondary" className={priority.className}>
   464→              {priority.label}
   465→            </Badge>
   466→          </CardContent>
   467→        </Card>
   468→        <Card>
   469→          <CardContent className="p-4">
   470→            <p className="text-xs text-muted-foreground mb-1">Deadline</p>
   471→            <p className={`text-sm font-medium ${overdue ? "text-destructive" : ""}`}>
   472→              {formatDate(project.deadline)}
   473→            </p>
   474→          </CardContent>
   475→        </Card>
   476→        <Card>
   477→          <CardContent className="p-4">
   478→            <p className="text-xs text-muted-foreground mb-1">Team Size</p>
   479→            <p className="text-sm font-medium">{project.memberCount} member{project.memberCount !== 1 ? "s" : ""}</p>
   480→          </CardContent>
   481→        </Card>
   482→      </div>
   483→
   484→      {/* Description + Client Info */}
   485→      <Card>
   486→        <CardContent className="p-5">
   487→          <div className="grid gap-4 sm:grid-cols-2">
   488→            <div>
   489→              <p className="text-xs text-muted-foreground mb-1 font-medium uppercase tracking-wide">Description</p>
   490→              <p className="text-sm">{project.description || "No description provided"}</p>
   491→            </div>
   492→            <div className="space-y-3">
   493→              {project.clientName && (
   494→                <div>
   495→                  <p className="text-xs text-muted-foreground mb-1 font-medium uppercase tracking-wide">Client</p>
   496→                  <p className="text-sm">{project.clientName}</p>
   497→                </div>
   498→              )}
   499→              <div>
   500→                <p className="text-xs text-muted-foreground mb-1 font-medium uppercase tracking-wide">Created</p>
   501→                <p className="text-sm">{formatDate(project.createdAt)}</p>
   502→              </div>
   503→              <div>
   504→                <p className="text-xs text-muted-foreground mb-1 font-medium uppercase tracking-wide">Last Updated</p>
   505→                <p className="text-sm">{formatDate(project.updatedAt)}</p>
   506→              </div>
   507→            </div>
   508→          </div>
   509→        </CardContent>
   510→      </Card>
   511→
   512→      {/* Team Management */}
   513→      <div>
   514→        <div className="flex items-center justify-between mb-4">
   515→          <h2 className="text-lg font-semibold flex items-center gap-2">
   516→            <Users className="h-5 w-5" />
   517→            Project Team
   518→          </h2>
   519→          {isAdmin && (
   520→            <Button size="sm" onClick={openAddMemberDialog}>
   521→              <UserPlus className="mr-2 h-4 w-4" />
   522→              Add Member
   523→            </Button>
   524→          )}
   525→          {isSuperAdmin && (
   526→            <p className="text-xs text-muted-foreground">
   527→              Super admin is auto-assigned to all projects.
   528→            </p>
   529→          )}
   530→        </div>
   531→
   532→        {/* Team Leads */}
   533→        <Card className="mb-4 border-purple-200 dark:border-purple-800/40">
   534→          <CardHeader className="pb-3">
   535→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   536→              <Crown className="h-4 w-4 text-purple-600" />
   537→              Team Leads
   538→            </CardTitle>
   539→          </CardHeader>
   540→          <CardContent className="pt-0">
   541→            {leads.length > 0 ? (
   542→              <div className="space-y-3">
   543→                {leads.map((leadMember) => (
   544→                  <div key={leadMember.id} className="flex items-center justify-between">
   545→                    <div className="flex items-center gap-3">
   546→                      <Avatar className="h-9 w-9">
   547→                        <AvatarFallback className="text-xs bg-purple-100 text-purple-700 dark:bg-purple-900/30 dark:text-purple-400">
   548→                          {leadMember.user.name
   549→                            .split(" ")
   550→                            .map((n) => n[0])
   551→                            .join("")
   552→                            .toUpperCase()
   553→                            .slice(0, 2)}
   554→                        </AvatarFallback>
   555→                      </Avatar>
   556→                      <div>
   557→                        <p className="text-sm font-medium">{leadMember.user.name}</p>
   558→                        <p className="text-xs text-muted-foreground">{leadMember.user.email}</p>
   559→                      </div>
   560→                    </div>
   561→                    {isSuperAdmin && (
   562→                      <DropdownMenu>
   563→                        <DropdownMenuTrigger asChild>
   564→                          <Button variant="ghost" size="icon" className="h-8 w-8">
   565→                            <MoreHorizontal className="h-4 w-4" />
   566→                          </Button>
   567→                        </DropdownMenuTrigger>
   568→                        <DropdownMenuContent align="end">
   569→                          <DropdownMenuItem
   570→                            onClick={() => handleChangeRole(leadMember, "MEMBER")}
   571→                          >
   572→                            <Shield className="mr-2 h-4 w-4" />
   573→                            Demote to Member
   574→                          </DropdownMenuItem>
   575→                          <DropdownMenuItem

   608→        {/* Team Members */}
