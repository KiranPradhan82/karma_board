   135→   135→
   136→   136→  // Add todo dialog
   137→   137→  const [addOpen, setAddOpen] = useState(false);
   138→   138→  const [addLoading, setAddLoading] = useState(false);
   139→   139→  const [addForm, setAddForm] = useState({
   140→   140→    title: "",
   141→   141→    description: "",
   142→   142→    assigneeId: "",
   143→   143→    priority: "MEDIUM",
   144→   144→    dueDate: "",
   145→   145→  });
   146→   146→
   147→   147→  // Edit todo dialog
   148→   148→  const [editOpen, setEditOpen] = useState(false);
   149→   149→  const [editLoading, setEditLoading] = useState(false);
   150→   150→  const [editTodoId, setEditTodoId] = useState<string | null>(null);

   153→   153→    description: "",
   154→   154→    assigneeId: "",
   155→   155→    priority: "MEDIUM",

   164→   164→      if (assigneeFilter !== "ALL") params.set("assigneeId", assigneeFilter);

   169→   169→        setTodos(data.data.todos);
   170→   170→        // Always fetch unfiltered summary
   171→   171→        const summaryRes = await fetch(`/api/projects/${projectId}/todos`);
   172→   172→        const summaryData = await summaryRes.json();
   173→   173→        if (summaryData.success) {
   174→   174→          setSummary(summaryData.data.summary);
   175→   175→        }

   178→   178→      // Silently fail
   179→   179→    } finally {
   180→   180→      setLoading(false);
   181→   181→    }

   185→   185→    fetchTodos();

   189→   189→

   191→   191→    if (isSuperAdmin) {
   192→   192→      // SA can move anything to COMPLETED directly
   193→   193→      if (todo.status !== "COMPLETED") return "COMPLETED";
   194→   194→      return null;
   195→   195→    }
   196→   196→
   197→   197→    // Members: PENDING → IN_PROGRESS → PENDING_REVIEW
   198→   198→    switch (todo.status) {
   199→   199→      case "PENDING": return "IN_PROGRESS";
   200→   200→      case "IN_PROGRESS": return "PENDING_REVIEW";
   201→   201→      default: return null;
   202→   202→    }
   203→   203→  };

   205→   205→  const handleToggleStatus = async (todo: TodoItem) => {
   206→   206→    const nextStatus = getNextStatus(todo);
   207→   207→    if (!nextStatus) return;
   208→   208→
   209→   209→    // Optimistic update
   210→   210→    setTodos((prev) =>
   211→   211→      prev.map((t) => (t.id === todo.id ? { ...t, status: nextStatus } : t))
   212→   212→    );
   213→   213→
   214→   214→    try {
   215→   215→      const res = await fetch(`/api/projects/${projectId}/todos/${todo.id}`, {
   216→   216→        method: "PATCH",
   217→   217→        headers: { "Content-Type": "application/json" },
   218→   218→        body: JSON.stringify({ status: nextStatus }),
   219→   219→      });
   220→   220→      const data = await res.json();
   221→   221→      if (!data.success) {
   222→   222→        // Revert optimistic update
   223→   223→        setTodos((prev) =>
   224→   224→          prev.map((t) => (t.id === todo.id ? { ...t, status: todo.status } : t))
   225→   225→        );
   226→   226→        errorToast({ error: data.error, title: "Failed to update task" });
   227→   227→      } else {
   228→   228→        fetchTodos();
   229→   229→        if (nextStatus === "COMPLETED") {
   230→   230→          toast.success("Task completed — client notified");
   231→   231→        } else if (nextStatus === "PENDING_REVIEW") {
   232→   232→          toast.success("Task submitted for review");
   233→   233→        } else {
   234→   234→          toast.success("Task status updated");
   235→   235→        }
   236→   236→      }
   237→   237→    } catch {
   238→   238→      setTodos((prev) =>
   239→   239→        prev.map((t) => (t.id === todo.id ? { ...t, status: todo.status } : t))
   240→   240→      );
   241→   241→      errorToast({ error: "Failed to update task" });
   242→   242→    }
   243→   243→  };
   244→   244→
   245→   245→  // Super admin: reject a PENDING_REVIEW task back to IN_PROGRESS
   246→   246→  const handleRejectReview = async (todo: TodoItem) => {
   247→   247→    setTodos((prev) =>
   248→   248→      prev.map((t) => (t.id === todo.id ? { ...t, status: "IN_PROGRESS" } : t))
   249→   249→    );
   250→   250→    try {
   251→   251→      const res = await fetch(`/api/projects/${projectId}/todos/${todo.id}`, {
   252→   252→        method: "PATCH",
   253→   253→        headers: { "Content-Type": "application/json" },
   254→   254→        body: JSON.stringify({ status: "IN_PROGRESS" }),
   255→   255→      });
   256→   256→      const data = await res.json();
   257→   257→      if (!data.success) {
   258→   258→        setTodos((prev) =>
   259→   259→          prev.map((t) => (t.id === todo.id ? { ...t, status: todo.status } : t))
   260→   260→        );
   261→   261→        errorToast({ error: data.error, title: "Failed to reject task" });
   262→   262→      } else {
   263→   263→        fetchTodos();
   264→   264→        toast.success("Task sent back for revisions");
   265→   265→      }
   266→   266→    } catch {
   267→   267→      setTodos((prev) =>
   268→   268→        prev.map((t) => (t.id === todo.id ? { ...t, status: todo.status } : t))
   269→   269→      );
   270→   270→      errorToast({ error: "Failed to reject task" });
   271→   271→    }
   272→   272→  };
   273→   273→
   274→   274→  // ── Handlers ──
   275→   275→
   276→   276→  const handleAddTodo = async () => {

   278→   278→      toast.error("Task title is required");
   279→   279→      return;
   280→   280→    }
   281→   281→    setAddLoading(true);
   282→   282→    try {
   283→   283→      const res = await fetch(`/api/projects/${projectId}/todos`, {
   284→   284→        method: "POST",
   285→   285→        headers: { "Content-Type": "application/json" },
   286→   286→        body: JSON.stringify({
   287→   287→          title: addForm.title.trim(),
   288→   288→          description: addForm.description.trim() || null,
   289→   289→          assigneeId: addForm.assigneeId || null,
   290→   290→          priority: addForm.priority,
   291→   291→          dueDate: addForm.dueDate ? new Date(addForm.dueDate).toISOString() : null,
   292→   292→        }),
   293→   293→      });
   294→   294→      const data = await res.json();
   295→   295→      if (data.success) {
   296→   296→        toast.success("Task added");
   297→   297→        setAddOpen(false);
   298→   298→        setAddForm({ title: "", description: "", assigneeId: "", priority: "MEDIUM", dueDate: "" });
   299→   299→        fetchTodos();
   300→   300→      } else {
   301→   301→        errorToast({ error: data.error, title: "Failed to add task" });
   302→   302→      }
   303→   303→    } catch {
   304→   304→      errorToast({ error: "Failed to add task" });
   305→   305→    } finally {
   306→   306→      setAddLoading(false);
   307→   307→    }
   308→   308→  };

   374→   374→    const sortedTodos = [...todos].sort((a, b) => a.sortOrder - b.sortOrder);
   375→   375→    const idx = sortedTodos.findIndex((t) => t.id === todo.id);
   376→   376→    if (direction === "up" && idx <= 0) return;
   377→   377→    if (direction === "down" && idx >= sortedTodos.length - 1) return;
   378→   378→
   379→   379→    const swapWith = direction === "up" ? sortedTodos[idx - 1] : sortedTodos[idx + 1];
   380→   380→
   381→   381→    // Optimistic swap
   382→   382→    setTodos((prev) =>
   383→   383→      prev.map((t) => {
   384→   384→        if (t.id === todo.id) return { ...t, sortOrder: swapWith.sortOrder };
   385→   385→        if (t.id === swapWith.id) return { ...t, sortOrder: todo.sortOrder };
   386→   386→        return t;
   387→   387→      })
   388→   388→    );
   389→   389→
   390→   390→    try {
   391→   391→      await fetch(`/api/projects/${projectId}/todos/${todo.id}`, {
   392→   392→        method: "PATCH",
   393→   393→        headers: { "Content-Type": "application/json" },
   394→   394→        body: JSON.stringify({ sortOrder: swapWith.sortOrder }),
   395→   395→      });
   396→   396→      await fetch(`/api/projects/${projectId}/todos/${swapWith.id}`, {
   397→   397→        method: "PATCH",
   398→   398→        headers: { "Content-Type": "application/json" },
   399→   399→        body: JSON.stringify({ sortOrder: todo.sortOrder }),
   400→   400→      });
   401→   401→    } catch {
   402→   402→      // Revert
   403→   403→      fetchTodos();
   404→   404→    }
   405→   405→  };
   406→   406→

   411→   411→    } catch {
   412→   412→      return null;
   413→   413→    }
   414→   414→  }

   418→   418→    return new Date(dateStr) < new Date();
   419→   419→  }

   426→   426→    if (isSuperAdmin) return true;
   427→   427→    if (isAdmin) return true;
   428→   428→    // Regular members can toggle if it's not in PENDING_REVIEW or COMPLETED
   429→   429→    if (todo.status === "PENDING_REVIEW" || todo.status === "COMPLETED") return false;
   430→   430→    return canManage;
   431→   431→  };
   432→   432→

   444→   444→            Add Task

   446→   446→        )}
   447→   447→      </div>
   448→   448→
   449→   449→      {/* Summary Progress */}
   450→   450→      {summary && summary.total > 0 && (
   451→   451→        <Card className="mb-4">
   452→   452→          <CardContent className="p-4">
   453→   453→            <div className="flex items-center justify-between mb-2">
   454→   454→              <div className="flex items-center gap-3">
   455→   455→                <span className="text-sm font-medium">{summary.completionPercent}% complete</span>
   456→   456→                <div className="flex items-center gap-2 text-xs text-muted-foreground">
   457→   457→                  <span className="flex items-center gap-1">
   458→   458→                    <span className="h-2 w-2 rounded-full bg-emerald-500" />
   459→   459→                    {summary.done} done
   460→   460→                  </span>
   461→   461→                  <span className="flex items-center gap-1">
   462→   462→                    <span className="h-2 w-2 rounded-full bg-orange-500" />
   463→   463→                    {summary.pendingReview} review
   464→   464→                  </span>
