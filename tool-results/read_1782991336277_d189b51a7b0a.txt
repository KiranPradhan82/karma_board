    18→// ===== Tool: create_project =====
    19→
    20→async function createProject(
    21→  args: { name: string; description?: string; priority?: string; deadline?: string; color?: string; clientName?: string },
    22→  ctx: ExecutorContext
    23→): Promise<AiToolResult> {
    24→  const { tursoClient, userId, userRole } = ctx;
    25→
    26→  // RBAC: ADMIN+ or SUPERADMIN only
    27→  if (userRole !== "SUPERADMIN" && userRole !== "ADMIN") {
    28→    return {
    29→      toolCallId: "",
    30→      toolName: "create_project",
    31→      success: false,
    32→      result: "Permission denied: Only ADMIN and SUPERADMIN can create projects.",
    33→      displayMessage: "Permission denied — only admins can create projects.",
    34→    };
    35→  }
    36→
    37→  // Validate
    38→  if (!args.name || args.name.trim().length === 0) {
    39→    return {
    40→      toolCallId: "",
    41→      toolName: "create_project",
    42→      success: false,
    43→      result: "Project name is required.",
    44→      displayMessage: "Missing project name.",
    45→    };
    46→  }
    47→
    48→  if (args.name.length > 100) {
    49→    return {
    50→      toolCallId: "",
    51→      toolName: "create_project",
    52→      success: false,
    53→      result: "Project name must be 100 characters or less.",
    54→      displayMessage: "Project name is too long (max 100 chars).",
    55→    };
    56→  }
    57→
    58→  const priority = args.priority || "MEDIUM";
    59→  const deadline = args.deadline || null;
    60→
    61→  try {
    62→    const projectId = crypto.randomUUID();
    63→    await tursoClient.execute({
    64→      sql: `INSERT INTO "Project" (id, name, description, status, priority, "clientName", color, deadline, "createdAt", "updatedAt")
    65→            VALUES (?, ?, ?, 'ACTIVE', ?, ?, ?, ?, datetime('now'), datetime('now'))`,
    66→      args: [
    67→        projectId,
    68→        args.name.trim(),
    69→        args.description?.trim() || null,
    70→        priority,
    71→        args.clientName?.trim() || null,
    72→        args.color || null,
    73→        deadline,
    74→      ],
    75→    });
    76→
    77→    // Auto-add the creator as LEAD
    78→    await tursoClient.execute({
    79→      sql: `INSERT INTO "ProjectMember" (id, "projectId", "userId", role, "joinedAt")
    80→            VALUES (?, ?, ?, 'LEAD', datetime('now'))`,
    81→      args: [crypto.randomUUID(), projectId, userId],
    82→    });
    83→
    84→    return {
    85→      toolCallId: "",
    86→      toolName: "create_project",
    87→      success: true,
    88→      result: JSON.stringify({ projectId, name: args.name.trim(), priority, deadline, clientName: args.clientName?.trim() || null }),
    89→      displayMessage: `Created project "${args.name.trim()}" with ${priority} priority.${deadline ? ` Deadline: ${deadline}.` : ""}`,
    90→    };
    91→  } catch (error) {
    92→    console.error("[createProject tool] DB error:", error);
    93→    return {
    94→      toolCallId: "",
    95→      toolName: "create_project",
    96→      success: false,
    97→      result: `Database error: ${error instanceof Error ? error.message : "Unknown error"}`,
    98→      displayMessage: "Failed to create project due to a database error.",
    99→    };
   100→  }
   101→}

   103→// ===== Tool: list_projects =====
   104→
   105→async function listProjects(
   106→  args: { status?: string },
   107→  ctx: ExecutorContext
   108→): Promise<AiToolResult> {
   109→  const { tursoClient, userId, userRole } = ctx;
   110→
   111→  try {
   112→    let rows;
   113→    if (userRole === "SUPERADMIN") {
   114→      if (args.status) {
   115→        rows = (await tursoClient.execute({
   116→          sql: `SELECT id, name, description, status, priority, deadline, "clientName", color
   117→                FROM "Project"
   118→                WHERE status = ? AND status != 'ARCHIVED'
   119→                ORDER BY "createdAt" DESC`,
   120→          args: [args.status],
   121→        })).rows;
   122→      } else {
   123→        rows = (await tursoClient.execute({
   124→          sql: `SELECT id, name, description, status, priority, deadline, "clientName", color
   125→                FROM "Project"
   126→                WHERE status != 'ARCHIVED'
   127→                ORDER BY "createdAt" DESC`,
   128→          args: [],
   129→        })).rows;
   130→      }
   131→    } else {
   132→      if (args.status) {
   133→        rows = (await tursoClient.execute({
   134→          sql: `SELECT p.id, p.name, p.description, p.status, p.priority, p.deadline, p."clientName", p.color
   135→                FROM "Project" p
   136→                JOIN "ProjectMember" pm ON p.id = pm."projectId"
   137→                WHERE pm."userId" = ? AND pm."removedAt" IS NULL AND p.status = ? AND p.status != 'ARCHIVED'
   138→                ORDER BY p."createdAt" DESC`,
   139→          args: [userId, args.status],
   140→        })).rows;
   141→      } else {
   142→        rows = (await tursoClient.execute({
   143→          sql: `SELECT p.id, p.name, p.description, p.status, p.priority, p.deadline, p."clientName", p.color
   144→                FROM "Project" p
   145→                JOIN "ProjectMember" pm ON p.id = pm."projectId"
   146→                WHERE pm."userId" = ? AND pm."removedAt" IS NULL AND p.status != 'ARCHIVED'
   147→                ORDER BY p."createdAt" DESC`,
   148→          args: [userId],
   149→        })).rows;
   150→      }
   151→    }
   152→
   153→    const projects = rows.map((row) => ({
   154→      id: row.id,
   155→      name: row.name,
   156→      description: row.description || null,
   157→      status: row.status,
   158→      priority: row.priority,
   159→      deadline: row.deadline || null,
   160→      clientName: row.clientName || null,
   161→      color: row.color || null,
   162→    }));
   163→
   164→    return {
   165→      toolCallId: "",
   166→      toolName: "list_projects",
   167→      success: true,
   168→      result: JSON.stringify({ count: projects.length, projects }),
   169→      displayMessage: `Found ${projects.length} project${projects.length !== 1 ? "s" : ""}.`,
   170→    };
   171→  } catch (error) {
   172→    console.error("[listProjects tool] DB error:", error);
   173→    return {
   174→      toolCallId: "",
   175→      toolName: "list_projects",
   176→      success: false,
   177→      result: `Database error: ${error instanceof Error ? error.message : "Unknown error"}`,
   178→      displayMessage: "Failed to list projects.",
   179→    };
   180→  }
   181→}

   183→// ===== Tool: get_project_info =====
   184→
   185→async function getProjectInfo(
   186→  args: { projectId: string },
   187→  ctx: ExecutorContext
   188→): Promise<AiToolResult> {
   189→  const { tursoClient, userId, userRole } = ctx;
   190→
   191→  // Check access
   192→  if (userRole !== "SUPERADMIN") {
   193→    const access = await tursoClient.execute({
   194→      sql: `SELECT id FROM "ProjectMember" WHERE "projectId" = ? AND "userId" = ? AND "removedAt" IS NULL`,
   195→      args: [args.projectId, userId],
   196→    });
   197→    if (access.rows.length === 0) {
   198→      return {
   199→        toolCallId: "",
   200→        toolName: "get_project_info",
   201→        success: false,
   202→        result: "You don't have access to this project.",
   203→        displayMessage: "You don't have access to this project.",
   204→      };
   205→    }
   206→  }
   207→
   208→  try {
   209→    const projectRows = await tursoClient.execute({
   210→      sql: `SELECT id, name, description, status, priority, deadline, "clientName", color, "createdAt", "updatedAt"
   211→            FROM "Project" WHERE id = ?`,
   212→      args: [args.projectId],
   213→    });
   214→
   215→    if (projectRows.rows.length === 0) {
   216→      return {
   217→        toolCallId: "",
   218→        toolName: "get_project_info",
   219→        success: false,
   220→        result: "Project not found.",
   221→        displayMessage: "Project not found.",
   222→      };
   223→    }
   224→
   225→    const row = projectRows.rows[0];
   226→
   227→    // Get team count
   228→    const teamResult = await tursoClient.execute({
   229→      sql: `SELECT COUNT(*) as count FROM "ProjectMember" WHERE "projectId" = ? AND "removedAt" IS NULL`,
   230→      args: [args.projectId],
   231→    });
   232→    const teamCount = Number(teamResult.rows[0].count);
   233→
   234→    const project = {
   235→      id: row.id,
   236→      name: row.name,
   237→      description: row.description || null,
   238→      status: row.status,
   239→      priority: row.priority,
   240→      deadline: row.deadline || null,
   241→      clientName: row.clientName || null,
   242→      color: row.color || null,
   243→      teamMembers: teamCount,
   244→      createdAt: row.createdAt,
   245→      updatedAt: row.updatedAt,
   246→    };
   247→
   248→    return {
   249→      toolCallId: "",
   250→      toolName: "get_project_info",
   251→      success: true,
   252→      result: JSON.stringify(project),
   253→      displayMessage: `Retrieved details for "${project.name}" (Status: ${project.status}, Team: ${teamCount}).`,
   254→    };
   255→  } catch (error) {
   256→    console.error("[getProjectInfo tool] DB error:", error);
   257→    return {
   258→      toolCallId: "",
   259→      toolName: "get_project_info",
   260→      success: false,
   261→      result: `Database error: ${error instanceof Error ? error.message : "Unknown error"}`,
   262→      displayMessage: "Failed to get project info.",
   263→    };
   264→  }
   265→}

   268→
   269→async function updateProject(
   270→  args: {
   271→    projectId: string;
   272→    name?: string;
   273→    description?: string;
   274→    status?: string;
   275→    priority?: string;
   276→    deadline?: string;
   277→    color?: string;
   278→  },
   279→  ctx: ExecutorContext
   280→): Promise<AiToolResult> {
   281→  const { tursoClient, userRole } = ctx;
   282→
   283→  // RBAC: ADMIN+ or SUPERADMIN only
   284→  if (userRole !== "SUPERADMIN" && userRole !== "ADMIN") {
   285→    return {
   286→      toolCallId: "",
   287→      toolName: "update_project",
   288→      success: false,
   289→      result: "Permission denied: Only ADMIN and SUPERADMIN can update projects.",
   290→      displayMessage: "Permission denied — only admins can update projects.",
   291→    };
   292→  }
   293→
   294→  // Build dynamic UPDATE query
   295→  const updates: string[] = [];
   296→  const values: unknown[] = [];
   297→
   298→  if (args.name !== undefined) {
   299→    updates.push(`name = ?`);
   300→    values.push(args.name.trim());
   301→  }
   302→  if (args.description !== undefined) {
   303→    updates.push(`description = ?`);
   304→    values.push(args.description?.trim() || null);
   305→  }
   306→  if (args.status !== undefined) {
   307→    updates.push(`status = ?`);
   308→    values.push(args.status);
   309→  }
   310→  if (args.priority !== undefined) {
   311→    updates.push(`priority = ?`);
   312→    values.push(args.priority);
   313→  }
   314→  if (args.deadline !== undefined) {
   315→    updates.push(`deadline = ?`);
   316→    values.push(args.deadline || null);
   317→  }
   318→  if (args.color !== undefined) {
   319→    updates.push(`color = ?`);
   320→    values.push(args.color || null);
   321→  }
   322→
   323→  if (updates.length === 0) {
   324→    return {
   325→      toolCallId: "",
   326→      toolName: "update_project",
   327→      success: false,
   328→      result: "No fields to update.",
   329→      displayMessage: "No update fields provided.",
   330→    };
   331→  }
   332→
   333→  updates.push(`"updatedAt" = datetime('now')`);
   334→  values.push(args.projectId);
   335→
   336→  try {
   337→    await tursoClient.execute({
   338→      sql: `UPDATE "Project" SET ${updates.join(", ")} WHERE id = ?`,
   339→      args: values,
   340→    });
   341→
   342→    // Get updated project name for display
   343→    const nameResult = await tursoClient.execute({
   344→      sql: `SELECT name FROM "Project" WHERE id = ?`,
   345→      args: [args.projectId],
   346→    });
   347→    const projectName = nameResult.rows[0]?.name || "Project";
   348→
   349→    return {
   350→      toolCallId: "",
   351→      toolName: "update_project",
   352→      success: true,
   353→      result: JSON.stringify({ projectId: args.projectId, updatedFields: args }),
   354→      displayMessage: `Updated "${projectName}" successfully.`,
   355→    };
   356→  } catch (error) {
   357→    console.error("[updateProject tool] DB error:", error);
   358→    return {
   359→      toolCallId: "",
   360→      toolName: "update_project",
   361→      success: false,
   362→      result: `Database error: ${error instanceof Error ? error.message : "Unknown error"}`,
   363→      displayMessage: "Failed to update project.",
   364→    };
   365→  }
   366→}

   368→// ===== Tool: add_project_member =====
   369→
   370→async function addProjectMember(
   371→  args: { projectId: string; userId: string; role?: string },
   372→  ctx: ExecutorContext
   373→): Promise<AiToolResult> {
   374→  const { tursoClient, userRole, userId } = ctx;
   375→
   376→  // RBAC: ADMIN, SUPERADMIN, or LEAD of this project
   377→  if (userRole !== "SUPERADMIN" && userRole !== "ADMIN") {
   378→    // Check if user is LEAD of this project
   379→    const leadCheck = await tursoClient.execute({
   380→      sql: `SELECT role FROM "ProjectMember" WHERE "projectId" = ? AND "userId" = ? AND "removedAt" IS NULL AND role = 'LEAD'`,
   381→      args: [args.projectId, userId],
   382→    });
   383→    if (leadCheck.rows.length === 0) {
   384→      return {
   385→        toolCallId: "",
   386→        toolName: "add_project_member",
   387→        success: false,
   388→        result: "Permission denied: Only ADMIN, SUPERADMIN, or project LEAD can add members.",
   389→        displayMessage: "Permission denied — only admins and project leads can add members.",
   390→      };
   391→    }
   392→  }
   393→
   394→  const memberRole = args.role || "MEMBER";
   395→
   396→  try {
   397→    // Check if user exists
   398→    const userExists = await tursoClient.execute({
   399→      sql: `SELECT id, name FROM "User" WHERE id = ?`,
   400→      args: [args.userId],
   401→    });
   402→    if (userExists.rows.length === 0) {
   403→      return {
   404→        toolCallId: "",
   405→        toolName: "add_project_member",
   406→        success: false,
   407→        result: "User not found.",
   408→        displayMessage: "User not found in the system.",
   409→      };
   410→    }
   411→
   412→    const memberName = userExists.rows[0].name as string;
   413→
   414→    // Check if already a member
   415→    const existing = await tursoClient.execute({
   416→      sql: `SELECT id FROM "ProjectMember" WHERE "projectId" = ? AND "userId" = ? AND "removedAt" IS NULL`,
   417→      args: [args.projectId, args.userId],
   418→    });
   419→    if (existing.rows.length > 0) {
   420→      return {
   421→        toolCallId: "",
   422→        toolName: "add_project_member",
   423→        success: false,
   424→        result: "User is already a member of this project.",
   425→        displayMessage: `${memberName} is already a member of this project.`,
   426→      };
   427→    }
   428→
   429→    // Enforce single LEAD rule
   430→    if (memberRole === "LEAD") {
   431→      const existingLead = await tursoClient.execute({
   432→        sql: `SELECT id FROM "ProjectMember" WHERE "projectId" = ? AND role = 'LEAD' AND "removedAt" IS NULL`,
   433→        args: [args.projectId],
   434→      });
   435→      if (existingLead.rows.length > 0) {
   436→        return {
   437→          toolCallId: "",
   438→          toolName: "add_project_member",
   439→          success: false,
   440→          result: "This project already has a LEAD. Remove the existing LEAD first or assign a different role.",
   441→          displayMessage: "This project already has a Lead. Only one Lead per project is allowed.",
   442→        };
   443→      }
   444→    }
   445→
   446→    await tursoClient.execute({
   447→      sql: `INSERT INTO "ProjectMember" (id, "projectId", "userId", role, "joinedAt")
   448→            VALUES (?, ?, ?, ?, datetime('now'))`,
   449→      args: [crypto.randomUUID(), args.projectId, args.userId, memberRole],
   450→    });
   451→
   452→    return {
   453→      toolCallId: "",
   454→      toolName: "add_project_member",
   455→      success: true,
   456→      result: JSON.stringify({ projectId: args.projectId, userId: args.userId, role: memberRole }),
   457→      displayMessage: `Added ${memberName} to the project as ${memberRole}.`,
   458→    };
   459→  } catch (error) {
   460→    console.error("[addProjectMember tool] DB error:", error);
   461→    return {
   462→      toolCallId: "",
   463→      toolName: "add_project_member",
   464→      success: false,
   465→      result: `Database error: ${error instanceof Error ? error.message : "Unknown error"}`,
   466→      displayMessage: "Failed to add project member.",
   467→    };
   468→  }
   469→}

   472→
   473→async function knowledgeResearch(
   474→  args: { query: string; category?: string },
   475→  _ctx: ExecutorContext
   476→): Promise<AiToolResult> {
   477→  if (!args.query || args.query.trim().length === 0) {
   478→    return {
   479→      toolCallId: "",
   480→      toolName: "knowledge_research",
   481→      success: false,
   482→      result: "Research query is required.",
   483→      displayMessage: "Missing research query.",
   484→    };
   485→  }
   486→
   487→  const query = args.query.trim();
   488→  const category = args.category || "general";
   489→
   490→  // Knowledge-based research: return a structured prompt that guides the AI
   491→  // to use its extensive training knowledge for the topic. This avoids fake HTTP
   492→  // calls to non-existent endpoints while still providing valuable research context.
   493→  const categoryGuidance: Record<string, string> = {
   494→    competitors: `Provide a competitive analysis including: top 3-5 competitors/products in this space, their key features, pricing models, strengths/weaknesses, and what differentiates them. Include market positioning insights.`,
   495→    technology: `Provide a technology analysis including: current best practices, recommended tech stack options, version considerations, framework comparisons, performance implications, and community/ecosystem support.`,

   498→    market_trends: `Provide market analysis including: current industry trends, growth projections, target market size, user adoption patterns, emerging technologies, and competitive landscape shifts.`,
   499→    general: `Provide comprehensive research insights on this topic including: key facts, current best practices, relevant examples, and actionable recommendations.`,
   500→  };
   501→
   502→  const guidance = categoryGuidance[category] || categoryGuidance.general;
   503→
   504→  const result = {
   505→    query,
   506→    category,
   507→    researchGuidance: guidance,
   508→    note: "Use your extensive training knowledge to provide detailed, specific, and current insights for this research topic. Include real product names, specific technologies, concrete examples, and actionable data points.",
   509→  };
   510→
   511→  return {
   512→    toolCallId: "",
   513→    toolName: "knowledge_research",
   514→    success: true,
   515→    result: JSON.stringify(result),
   516→    displayMessage: `Researching: "${query.slice(0, 50)}" (${category})`,
   517→  };
   518→}

   540→async function saveGithubConfig(
   541→  args: { repoUrl: string; pat: string; patExpiry?: string },
   542→  ctx: ExecutorContext
   543→): Promise<AiToolResult> {
   544→  const { tursoClient, userRole } = ctx;
   545→
   546→  // RBAC: SUPERADMIN and ADMIN can save GitHub config
   547→  if (userRole !== "SUPERADMIN" && userRole !== "ADMIN") {
   548→    return {
   549→      toolCallId: "",
   550→      toolName: "save_github_config",
   551→      success: false,
   552→      result: "Permission denied: Only SUPERADMIN can save GitHub configuration.",
   553→      displayMessage: "Permission denied — only Super Admin can save GitHub config.",
   554→    };
   555→  }
   556→
   557→  if (!args.repoUrl || !args.pat) {
   558→    return {
   559→      toolCallId: "",
   560→      toolName: "save_github_config",
   561→      success: false,
   562→      result: "Both repoUrl and pat are required.",
   563→      displayMessage: "Missing GitHub URL or Personal Access Token.",
   564→    };
   565→  }
   566→
   567→  try {
   568→    const encryptedPat = encrypt(args.pat);
   569→
   570→    // Upsert GITHUB_REPO_URL, GITHUB_PAT, GITHUB_PAT_EXPIRY
   571→    await upsertSetting(tursoClient, "GITHUB_REPO_URL", args.repoUrl.trim());
   572→    await upsertSetting(tursoClient, "GITHUB_PAT", encryptedPat);
   573→
   574→    // Store expiry date (YYYY-MM-DD format, validated)
   575→    if (args.patExpiry) {
   576→      const expiryDate = new Date(args.patExpiry);
   577→      if (isNaN(expiryDate.getTime())) {
   578→        return {
   579→          toolCallId: "",
   580→          toolName: "save_github_config",
   581→          success: false,
   582→          result: "Invalid expiry date format. Use YYYY-MM-DD.",
   583→          displayMessage: "The expiry date format is invalid. Please provide a date like 2025-12-31.",
   584→        };
   585→      }
   586→      await upsertSetting(tursoClient, "GITHUB_PAT_EXPIRY", args.patExpiry);
   587→    }

   722→      toolCallId: "",
   723→      toolName: "github_pull",
   724→      success: false,
   725→      result: "GitHub PAT not configured — run /init first.",
   726→      displayMessage: "GitHub PAT is missing. Please run /init to set up GitHub credentials.",
   727→    };
   728→  }
   729→
   730→  const repoUrl = repoResult.rows[0].value as string;
   731→  let token: string;
   732→  try {
   733→    token = decrypt(patResult.rows[0].value as string);
   734→  } catch {
   735→    token = patResult.rows[0].value as string;
   736→  }
   737→
   738→  // Parse owner/repo from URL
   739→  const match = repoUrl.match(/github\.com[\/:]([^\/]+)\/([^\/\s#?]+)/i);
   740→  if (!match) {
   741→    return {
   742→      toolCallId: "",
   743→      toolName: "github_pull",
   744→      success: false,
   745→      result: "Invalid GitHub repository URL: " + repoUrl,
   746→      displayMessage: "The GitHub repository URL is invalid.",
   747→    };
   748→  }
   749→
   750→  const owner = match[1];
   751→  const repo = match[2];
   752→  const branch = args.branch || "main";
   753→
   754→  try {
   755→    const headers: Record<string, string> = {
   756→      Authorization: "Bearer " + token,
   757→      Accept: "application/vnd.github.v3+json",
   758→      "User-Agent": "KarmaBoard/1.0",
   759→    };
   760→
   761→    if (args.path) {
   762→      // Fetch specific file
   763→      const res = await fetch(
   764→        `https://api.github.com/repos/${owner}/${repo}/contents/${encodeURIComponent(args.path)}?ref=${branch}`,
   765→        { headers }
   766→      );
   767→
   768→      if (res.status === 404) {
   769→        return {
   770→          toolCallId: "",
   771→          toolName: "github_pull",
   772→          success: false,
   773→          result: `File not found: ${args.path}`,
   774→          displayMessage: `File '${args.path}' was not found in the repository.`,
   775→        };
   776→      }
   777→
   778→      if (!res.ok) {
   779→        const errText = await res.text().catch(() => "Unknown error");
   780→        if (res.status === 401 || res.status === 403) {
   781→          return {
   782→            toolCallId: "",
   783→            toolName: "github_pull",
   784→            success: false,
   785→            result: "GitHub PAT expired or invalid. Please update it in /init.",
   786→            displayMessage: "Your GitHub PAT appears to be expired or invalid. Please run /init to update it.",
   787→          };
   788→        }
   789→        return {
   790→          toolCallId: "",
   791→          toolName: "github_pull",
   792→          success: false,

   794→          displayMessage: `Failed to fetch from GitHub (${res.status}).`,
   795→        };
   796→      }
   797→

   802→        fileContent = Buffer.from(data.content.replace(/\n/g, ""), "base64").toString("utf-8");
   803→      } else {
   804→        fileContent = data.content || "";
   805→      }
   806→
   807→      return {
   808→        toolCallId: "",
   809→        toolName: "github_pull",
   810→        success: true,
   811→        result: JSON.stringify({ path: data.path, sha: data.sha, size: data.size, content: fileContent }),
   812→        displayMessage: `Fetched '${data.path}' (${data.size} bytes) from ${branch}.`,
   813→      };
   814→    } else {
   815→      // List repo tree (top-level)
   816→      const res = await fetch(
   817→        `https://api.github.com/repos/${owner}/${repo}/contents/?ref=${branch}`,
   818→        { headers }
   819→      );
   820→
   821→      if (!res.ok) {
   822→        const errText = await res.text().catch(() => "Unknown error");
   823→        if (res.status === 401 || res.status === 403) {
   824→          return {
   825→            toolCallId: "",
   826→            toolName: "github_pull",
   827→            success: false,
   828→            result: "GitHub PAT expired or invalid. Please update it in /init.",
   829→            displayMessage: "Your GitHub PAT appears to be expired or invalid. Please run /init to update it.",
   830→          };
   831→        }
   832→        return {
   833→          toolCallId: "",
   834→          toolName: "github_pull",
   835→          success: false,
   836→          result: `GitHub API error (${res.status}): ${errText}`,
   837→          displayMessage: `Failed to list repository tree (${res.status}).`,
   838→        };
   839→      }
   840→
   841→      const tree = await res.json();
   842→      const fileList = tree.map((item: any) => ({
   843→        name: item.name,
   844→        path: item.path,
   845→        type: item.type, // 'file' or 'dir'
   846→        size: item.size || 0,
   847→      }));
   848→
   849→      return {
   850→        toolCallId: "",
   851→        toolName: "github_pull",
   852→        success: true,
   853→        result: JSON.stringify({ branch, items: fileList }),
   854→        displayMessage: `Listed ${fileList.length} items in repository root (${branch}).`,
   855→      };
   856→    }
   857→  } catch (error) {
   858→    console.error("[githubPull tool] Error:", error);
   859→    return {
   860→      toolCallId: "",
   861→      toolName: "github_pull",
   862→      success: false,
   863→      result: `GitHub fetch error: ${error instanceof Error ? error.message : "Unknown error"}`,
   864→      displayMessage: "Failed to fetch from GitHub.",
   865→    };
   866→  }
   867→}

  1152→        headers: { ...headers, "Content-Type": "application/json" },
  1153→        body: JSON.stringify({ base_tree: baseSha, tree: treeEntries }),
  1154→      }
  1155→    );
  1156→    if (!treeRes.ok) {
  1157→      const errText = await treeRes.text().catch(() => "Unknown error");

  1167→

  1172→        method: "POST",
  1173→        headers: { ...headers, "Content-Type": "application/json" },

  1178→      const errText = await commitRes.text().catch(() => "Unknown error");

  1185→      };
  1186→    }

  1188→

  1190→    const updateRes = await fetch(
  1191→      `https://api.github.com/repos/${owner}/${repo}/git/refs/heads/${branch}`,
  1192→      {
  1193→        method: "PATCH",
  1194→        headers: { ...headers, "Content-Type": "application/json" },

  1206→      };
  1207→    }
  1208→
  1209→    // 6. Update all file statuses to 'pushed'
  1210→    for (const file of files) {

  1213→        args: [file.id],
  1214→      });
  1215→    }
  1216→
  1217→    const pushedPaths = files.map((f) => f.path);
  1218→    return {
  1219→      toolCallId: "",

  1223→      displayMessage: `Pushed ${files.length} file(s) to GitHub (${branch}). Commit: ${commitData.sha.slice(0, 7)}`,

  1237→// ===== Helper: Load GitHub Config =====
  1238→
  1239→interface GithubConfig {
  1240→  owner: string;
  1241→  repo: string;
  1242→  token: string;
  1243→  headers: Record<string, string>;
  1244→}
  1245→
  1246→async function loadGithubConfig(tursoClient: ExecutorContext["tursoClient"], branch?: string): Promise<GithubConfig | AiToolResult> {
  1247→  const repoResult = await tursoClient.execute({
  1248→    sql: `SELECT value FROM "Settings" WHERE key = 'GITHUB_REPO_URL'`,
  1249→    args: [],
  1250→  });
  1251→  const patResult = await tursoClient.execute({
  1252→    sql: `SELECT value FROM "Settings" WHERE key = 'GITHUB_PAT'`,
  1253→    args: [],
  1254→  });
  1255→
  1256→  if (repoResult.rows.length === 0) {
  1257→    return {
  1258→      toolCallId: "",
  1259→      toolName: "",
  1260→      success: false,
  1261→      result: "GitHub not configured — run /init first.",
  1262→      displayMessage: "GitHub is not configured. Please run /init to set up GitHub credentials.",
  1263→    };
  1264→  }
  1265→
  1266→  if (patResult.rows.length === 0) {
  1267→    return {
  1268→      toolCallId: "",
  1269→      toolName: "",
  1270→      success: false,
  1271→      result: "GitHub PAT not configured — run /init first.",
  1272→      displayMessage: "GitHub PAT is missing. Please run /init to set up GitHub credentials.",
  1273→    };
  1274→  }
  1275→
  1276→  const repoUrl = repoResult.rows[0].value as string;
  1277→  let token: string;
  1278→  try {
  1279→    token = decrypt(patResult.rows[0].value as string);
  1280→  } catch {
  1281→    token = patResult.rows[0].value as string;
  1282→  }
  1283→
  1284→  const match = repoUrl.match(/github\.com[\/:]([^\/]+)\/([^\/\s#?]+)/i);
  1285→  if (!match) {
  1286→    return {
  1287→      toolCallId: "",
  1288→      toolName: "",
  1289→      success: false,
  1290→      result: "Invalid GitHub repository URL: " + repoUrl,
  1291→      displayMessage: "The GitHub repository URL is invalid.",
  1292→    };
  1293→  }
  1294→
  1295→  const headers: Record<string, string> = {
  1296→    Authorization: "Bearer " + token,
  1297→    Accept: "application/vnd.github.v3+json",
  1298→    "User-Agent": "KarmaBoard/1.0",
  1299→  };
  1300→
  1301→  return { owner: match[1], repo: match[2], token, headers };
  1302→}

  1306→async function fsListDir(
  1307→  args: { path?: string; branch?: string },
  1308→  ctx: ExecutorContext
  1309→): Promise<AiToolResult> {
  1310→  const { tursoClient, userRole } = ctx;
  1311→
  1312→  if (userRole !== "SUPERADMIN" && userRole !== "ADMIN") {
  1313→    return {
  1314→      toolCallId: "", toolName: "fs_list_dir",
  1315→      success: false, result: "Permission denied: Only ADMIN and SUPERADMIN can list directories.",
  1316→      displayMessage: "Permission denied — only admins can list directories.",
  1317→    };
  1318→  }
  1319→
  1320→  const branch = args.branch || "main";
  1321→  const config = await loadGithubConfig(tursoClient);
  1322→  if (!("owner" in config)) return config as AiToolResult;
  1323→
  1324→  const { owner, repo, headers } = config;
  1325→
  1326→  try {
  1327→    let items: { name: string; path: string; type: string; size: number }[] = [];
  1328→
  1329→    if (args.path) {
  1330→      // Use contents API for a specific path
  1331→      const res = await fetch(
  1332→        `https://api.github.com/repos/${owner}/${repo}/contents/${encodeURIComponent(args.path)}?ref=${branch}`,
  1333→        { headers }
  1334→      );
  1335→
  1336→      if (!res.ok) {
  1337→        const errText = await res.text().catch(() => "Unknown error");
  1338→        return {
  1339→          toolCallId: "", toolName: "fs_list_dir",
  1340→          success: false, result: `GitHub API error (${res.status}): ${errText}`,
  1341→          displayMessage: `Failed to list directory '${args.path}' (${res.status}).`,
  1342→        };
  1343→      }
  1344→
  1345→      const data = await res.json();
  1346→      items = Array.isArray(data)
  1347→        ? data.map((item: any) => ({ name: item.name, path: item.path, type: item.type, size: item.size || 0 }))
  1348→        : [{ name: data.name, path: data.path, type: data.type, size: data.size || 0 }];
  1349→    } else {
  1350→      // Use recursive tree API for full listing
  1351→      const res = await fetch(
  1352→        `https://api.github.com/repos/${owner}/${repo}/git/trees/${branch}?recursive=1`,
  1353→        { headers }
  1354→      );
  1355→
  1356→      if (!res.ok) {
  1357→        const errText = await res.text().catch(() => "Unknown error");
  1358→        return {
  1359→          toolCallId: "", toolName: "fs_list_dir",
  1360→          success: false, result: `GitHub API error (${res.status}): ${errText}`,
  1361→          displayMessage: `Failed to list repository tree (${res.status}).`,
  1362→        };
  1363→      }
  1364→
  1365→      const data = await res.json();
  1366→      items = (data.tree || []).map((item: any) => ({
  1367→        name: item.path.split("/").pop() || item.path,
  1368→        path: item.path,
  1369→        type: item.type === "tree" ? "dir" : "file",
  1370→        size: item.size || 0,
  1371→      }));
  1372→    }
  1373→
  1374→    return {
  1375→      toolCallId: "", toolName: "fs_list_dir",
  1376→      success: true,
  1377→      result: JSON.stringify({ branch, path: args.path || "/", count: items.length, items }),
  1378→      displayMessage: `Listed ${items.length} item(s) in '${args.path || "/"}' (${branch}).`,
  1379→    };
  1380→  } catch (error) {
  1381→    console.error("[fsListDir tool] Error:", error);
  1382→    return {
  1383→      toolCallId: "", toolName: "fs_list_dir",
  1384→      success: false, result: `Error listing directory: ${error instanceof Error ? error.message : "Unknown error"}`,
  1385→      displayMessage: "Failed to list directory.",
  1386→    };
  1387→  }

  1853→async function extractTextFromZipBuffer(buffer: ArrayBuffer): Promise<string | null> {
  1854→  // Simple ZIP parser: find Local File Header entries and extract uncompressed content
  1855→  // ZIP format: PK\x03\x04 followed by file metadata and content
  1856→  const bytes = new Uint8Array(buffer);
  1857→  const parts: string[] = [];
  1858→  let offset = 0;
  1859→
  1860→  while (offset < bytes.length - 30) {
  1861→    // Look for Local File Header signature: PK\x03\x04
  1862→    if (bytes[offset] === 0x50 && bytes[offset + 1] === 0x4B &&
  1863→        bytes[offset + 2] === 0x03 && bytes[offset + 3] === 0x04) {
  1864→      const compressionMethod = bytes[offset + 8] | (bytes[offset + 9] << 8);
  1865→      const compressedSize = bytes[offset + 18] | (bytes[offset + 19] << 8) |
  1866→        (bytes[offset + 20] << 16) | (bytes[offset + 21] << 24);
  1867→      const uncompressedSize = bytes[offset + 22] | (bytes[offset + 23] << 8) |
  1868→        (bytes[offset + 24] << 16) | (bytes[offset + 25] << 24);
  1869→      const fileNameLength = bytes[offset + 26] | (bytes[offset + 27] << 8);
  1870→      const extraFieldLength = bytes[offset + 28] | (bytes[offset + 29] << 8);
  1871→
  1872→      const fileNameStart = offset + 30;
  1873→      const fileName = new TextDecoder().decode(bytes.slice(fileNameStart, fileNameStart + fileNameLength));
  1874→      const dataStart = fileNameStart + fileNameLength + extraFieldLength;
  1875→
  1876→      if (compressionMethod === 0) {
  1877→        // STORED (no compression)
  1878→        const content = new TextDecoder().decode(bytes.slice(dataStart, dataStart + uncompressedSize));
  1879→        if (fileName.endsWith(".txt")) {
  1880→          parts.push(content);
  1881→        }
  1882→        offset = dataStart + compressedSize;
  1883→      } else if (compressionMethod === 8) {
  1884→        // DEFLATE - use DecompressionStream if available (Edge Runtime)
  1885→        if (typeof DecompressionStream !== "undefined") {
  1886→          try {
  1887→            const deflateStream = new DecompressionStream("deflate-raw");
  1888→            const writer = deflateStream.writable.getWriter();
  1889→            writer.write(bytes.slice(dataStart, dataStart + compressedSize));
  1890→            writer.close();
  1891→
  1892→            const reader = deflateStream.readable.getReader();
  1893→            const chunks: Uint8Array[] = [];
  1894→            let totalSize = 0;
  1895→            while (true) {
  1896→              const { done, value } = await reader.read();
  1897→              if (done) break;
  1898→              chunks.push(value);
  1899→              totalSize += value.length;
  1900→            }
  1901→
  1902→            const result = new Uint8Array(totalSize);
  1903→            let pos = 0;
  1904→            for (const chunk of chunks) {
  1905→              result.set(chunk, pos);
  1906→              pos += chunk.length;
  1907→            }
  1908→
  1909→            const content = new TextDecoder().decode(result);
  1910→            if (fileName.endsWith(".txt")) {
  1911→              parts.push(content);
  1912→            }
  1913→          } catch {
  1914→            // Decompression failed, skip
  1915→          }
  1916→        }
  1917→        offset = dataStart + compressedSize;
  1918→      } else {
  1919→        offset = dataStart + compressedSize;
  1920→      }
  1921→    } else {
  1922→      offset++;
  1923→    }
  1924→  }
  1925→
  1926→  return parts.length > 0 ? parts.join("\n\n") : null;
  1927→}
