   117→

   124→
   125→export function ProjectTodos({ projectId, team, isAdmin, isSuperAdmin }: ProjectTodosProps) {
   126→  const { data: session } = useSession();
   127→  const user = session?.user as { id?: string; role?: string } | undefined;
   128→
   129→  const [todos, setTodos] = useState<TodoItem[]>([]);
   130→  const [summary, setSummary] = useState<TodoSummary | null>(null);
   131→  const [loading, setLoading] = useState(true);
   132→  const [statusFilter, setStatusFilter] = useState("ALL");
   133→  const [assigneeFilter, setAssigneeFilter] = useState("ALL");
   134→  const [expandedTodo, setExpandedTodo] = useState<string | null>(null);
   135→
   136→  // Add todo dialog
   137→  const [addOpen, setAddOpen] = useState(false);
   138→  const [addLoading, setAddLoading] = useState(false);
   139→  const [addForm, setAddForm] = useState({
   140→    title: "",
   141→    description: "",
   142→    assigneeId: "",
   143→    priority: "MEDIUM",
   144→    dueDate: "",
   145→  });
   146→
   147→  // Edit todo dialog
   148→  const [editOpen, setEditOpen] = useState(false);
   149→  const [editLoading, setEditLoading] = useState(false);
   150→  const [editTodoId, setEditTodoId] = useState<string | null>(null);
   151→  const [editForm, setEditForm] = useState({
   152→    title: "",
   153→    description: "",
   154→    assigneeId: "",
   155→    priority: "MEDIUM",
   156→    dueDate: "",
   157→    status: "PENDING",
   158→  });
   159→
   160→  const fetchTodos = useCallback(async () => {
   161→    try {
   162→      const params = new URLSearchParams();
   163→      if (statusFilter !== "ALL") params.set("status", statusFilter);
   164→      if (assigneeFilter !== "ALL") params.set("assigneeId", assigneeFilter);
   165→
   166→      const res = await fetch(`/api/projects/${projectId}/todos?${params.toString()}`);
   167→      const data = await res.json();
   168→      if (data.success) {
   169→        setTodos(data.data.todos);
   170→        // Always fetch unfiltered summary
   171→        const summaryRes = await fetch(`/api/projects/${projectId}/todos`);
   172→        const summaryData = await summaryRes.json();
   173→        if (summaryData.success) {
   174→          setSummary(summaryData.data.summary);
   175→        }
   176→      }
   177→    } catch {
   178→      // Silently fail
   179→    } finally {
   180→      setLoading(false);
   181→    }
   182→  }, [projectId, statusFilter, assigneeFilter]);
   183→
   184→  useEffect(() => {
   185→    fetchTodos();
   186→  }, [fetchTodos]);
   187→
   188→  // ── Determine what actions a user can take on a todo ──
   189→
   190→  const getNextStatus = (todo: TodoItem): string | null => {
   191→    if (isSuperAdmin) {
   192→      // SA can move anything to COMPLETED directly
   193→      if (todo.status !== "COMPLETED") return "COMPLETED";
   194→      return null;
   195→    }
   196→
   197→    // Members: PENDING → IN_PROGRESS → PENDING_REVIEW
   198→    switch (todo.status) {
   199→      case "PENDING": return "IN_PROGRESS";
   200→      case "IN_PROGRESS": return "PENDING_REVIEW";
   201→      default: return null;
   202→    }
   203→  };
   204→
   205→  const handleToggleStatus = async (todo: TodoItem) => {
   206→    const nextStatus = getNextStatus(todo);
   207→    if (!nextStatus) return;
   208→
   209→    // Optimistic update
   210→    setTodos((prev) =>
   211→      prev.map((t) => (t.id === todo.id ? { ...t, status: nextStatus } : t))
   212→    );
   213→
   214→    try {
   215→      const res = await fetch(`/api/projects/${projectId}/todos/${todo.id}`, {
   216→        method: "PATCH",
   217→        headers: { "Content-Type": "application/json" },
   218→        body: JSON.stringify({ status: nextStatus }),
   219→      });
   220→      const data = await res.json();
   221→      if (!data.success) {
   222→        // Revert optimistic update
   223→        setTodos((prev) =>
   224→          prev.map((t) => (t.id === todo.id ? { ...t, status: todo.status } : t))
   225→        );
   226→        errorToast({ error: data.error, title: "Failed to update task" });
   227→      } else {
   228→        fetchTodos();
   229→        if (nextStatus === "COMPLETED") {
   230→          toast.success("Task completed — client notified");
   231→        } else if (nextStatus === "PENDING_REVIEW") {
   232→          toast.success("Task submitted for review");
   233→        } else {
   234→          toast.success("Task status updated");
   235→        }
   236→      }
   237→    } catch {
   238→      setTodos((prev) =>
   239→        prev.map((t) => (t.id === todo.id ? { ...t, status: todo.status } : t))
   240→      );
   241→      errorToast({ error: "Failed to update task" });
   242→    }
   243→  };
   244→
   245→  // Super admin: reject a PENDING_REVIEW task back to IN_PROGRESS
   246→  const handleRejectReview = async (todo: TodoItem) => {
   247→    setTodos((prev) =>
   248→      prev.map((t) => (t.id === todo.id ? { ...t, status: "IN_PROGRESS" } : t))
   249→    );
   250→    try {
   251→      const res = await fetch(`/api/projects/${projectId}/todos/${todo.id}`, {
   252→        method: "PATCH",
   253→        headers: { "Content-Type": "application/json" },
   254→        body: JSON.stringify({ status: "IN_PROGRESS" }),
   255→      });
   256→      const data = await res.json();
   257→      if (!data.success) {
   258→        setTodos((prev) =>
   259→          prev.map((t) => (t.id === todo.id ? { ...t, status: todo.status } : t))
   260→        );
   261→        errorToast({ error: data.error, title: "Failed to reject task" });
   262→      } else {
   263→        fetchTodos();
   264→        toast.success("Task sent back for revisions");
   265→      }
   266→    } catch {
   267→      setTodos((prev) =>
   268→        prev.map((t) => (t.id === todo.id ? { ...t, status: todo.status } : t))
   269→      );
   270→      errorToast({ error: "Failed to reject task" });
   271→    }
   272→  };
   273→
   274→  // ── Handlers ──
   275→
   276→  const handleAddTodo = async () => {
   277→    if (!addForm.title.trim()) {
   278→      toast.error("Task title is required");
   279→      return;
   280→    }
   281→    setAddLoading(true);
   282→    try {
   283→      const res = await fetch(`/api/projects/${projectId}/todos`, {
   284→        method: "POST",
   285→        headers: { "Content-Type": "application/json" },
   286→        body: JSON.stringify({
   287→          title: addForm.title.trim(),
   288→          description: addForm.description.trim() || null,
   289→          assigneeId: addForm.assigneeId || null,
   290→          priority: addForm.priority,
   291→          dueDate: addForm.dueDate ? new Date(addForm.dueDate).toISOString() : null,
   292→        }),
   293→      });
   294→      const data = await res.json();
   295→      if (data.success) {
   296→        toast.success("Task added");
   297→        setAddOpen(false);
   298→        setAddForm({ title: "", description: "", assigneeId: "", priority: "MEDIUM", dueDate: "" });
   299→        fetchTodos();
   300→      } else {
   301→        errorToast({ error: data.error, title: "Failed to add task" });
   302→      }
   303→    } catch {
   304→      errorToast({ error: "Failed to add task" });
   305→    } finally {
   306→      setAddLoading(false);
   307→    }
   308→  };
   309→
   310→  const openEditDialog = (todo: TodoItem) => {
   311→    setEditTodoId(todo.id);
   312→    setEditForm({
   313→      title: todo.title,
   314→      description: todo.description || "",
   315→      assigneeId: todo.assigneeId || "",
   316→      priority: todo.priority,
   317→      dueDate: todo.dueDate ? todo.dueDate.split("T")[0] : "",
   318→      status: todo.status,
   319→    });
   320→    setEditOpen(true);
   321→  };
   322→
   323→  const handleEditSave = async () => {
   324→    if (!editTodoId || !editForm.title.trim()) {
   325→      toast.error("Task title is required");
   326→      return;
   327→    }
   328→    setEditLoading(true);
   329→    try {
   330→      const res = await fetch(`/api/projects/${projectId}/todos/${editTodoId}`, {
   331→        method: "PATCH",
   332→        headers: { "Content-Type": "application/json" },
   333→        body: JSON.stringify({
   334→          title: editForm.title.trim(),
   335→          description: editForm.description.trim() || null,
   336→          assigneeId: editForm.assigneeId || null,
   337→          priority: editForm.priority,
   338→          dueDate: editForm.dueDate ? new Date(editForm.dueDate).toISOString() : null,
   339→          status: editForm.status,
   340→        }),
   341→      });
   342→      const data = await res.json();
   343→      if (data.success) {
   344→        toast.success("Task updated");
   345→        setEditOpen(false);
   346→        fetchTodos();
   347→      } else {
   348→        errorToast({ error: data.error, title: "Failed to update task" });
   349→      }
   350→    } catch {
   351→      errorToast({ error: "Failed to update task" });
   352→    } finally {
   353→      setEditLoading(false);
   354→    }
   355→  };
   356→
   357→  const handleDeleteTodo = async (todo: TodoItem) => {
   358→    if (!confirm(`Delete "${todo.title}"? This cannot be undone.`)) return;
   359→    try {
   360→      const res = await fetch(`/api/projects/${projectId}/todos/${todo.id}`, { method: "DELETE" });
   361→      const data = await res.json();
   362→      if (data.success) {
   363→        toast.success("Task deleted");
   364→        fetchTodos();
   365→      } else {
   366→        errorToast({ error: data.error, title: "Failed to delete task" });
   367→      }
   368→    } catch {
   369→      errorToast({ error: "Failed to delete task" });
   370→    }
   371→  };
   372→
   373→  const handleMoveTodo = async (todo: TodoItem, direction: "up" | "down") => {
   374→    const sortedTodos = [...todos].sort((a, b) => a.sortOrder - b.sortOrder);
   375→    const idx = sortedTodos.findIndex((t) => t.id === todo.id);
   376→    if (direction === "up" && idx <= 0) return;
   377→    if (direction === "down" && idx >= sortedTodos.length - 1) return;
   378→
   379→    const swapWith = direction === "up" ? sortedTodos[idx - 1] : sortedTodos[idx + 1];
   380→
   381→    // Optimistic swap
   382→    setTodos((prev) =>
   383→      prev.map((t) => {
   384→        if (t.id === todo.id) return { ...t, sortOrder: swapWith.sortOrder };
   385→        if (t.id === swapWith.id) return { ...t, sortOrder: todo.sortOrder };
   386→        return t;
   387→      })
   388→    );
   389→
   390→    try {
   391→      await fetch(`/api/projects/${projectId}/todos/${todo.id}`, {
   392→        method: "PATCH",
   393→        headers: { "Content-Type": "application/json" },
   394→        body: JSON.stringify({ sortOrder: swapWith.sortOrder }),
   395→      });
   396→      await fetch(`/api/projects/${projectId}/todos/${swapWith.id}`, {
   397→        method: "PATCH",
   398→        headers: { "Content-Type": "application/json" },
   399→        body: JSON.stringify({ sortOrder: todo.sortOrder }),
   400→      });
   401→    } catch {
   402→      // Revert
   403→      fetchTodos();
   404→    }
   405→  };
   406→
   407→  function formatDueDate(dateStr: string | null) {
   408→    if (!dateStr) return null;
   409→    try {
   410→      return new Date(dateStr).toLocaleDateString("en-GB", { day: "numeric", month: "short" });
   411→    } catch {
   412→      return null;
   413→    }
   414→  }
   415→
   416→  function isOverdue(dateStr: string | null, status: string) {
   417→    if (!dateStr || status === "COMPLETED") return false;
   418→    return new Date(dateStr) < new Date();
   419→  }
   420→
   421→  // ── Render ──
   422→
   423→  const canManage = isAdmin || isSuperAdmin;
   424→  // Members can toggle status on their own assigned tasks or if they're admin/SA
   425→  const canToggle = (todo: TodoItem) => {
   426→    if (isSuperAdmin) return true;
   427→    if (isAdmin) return true;
   428→    // Regular members can toggle if it's not in PENDING_REVIEW or COMPLETED
   429→    if (todo.status === "PENDING_REVIEW" || todo.status === "COMPLETED") return false;
   430→    return canManage;
   431→  };
   432→
   433→  return (
   434→    <div>
   435→      {/* Header */}
   436→      <div className="flex items-center justify-between mb-4">
   437→        <h2 className="text-lg font-semibold flex items-center gap-2">
   438→          <ListTodo className="h-5 w-5" />
   439→          Project Todos
   440→        </h2>
   441→        {isAdmin && (
   442→          <Button size="sm" onClick={() => setAddOpen(true)}>
   443→            <Plus className="mr-2 h-4 w-4" />
   444→            Add Task
   445→          </Button>
   446→        )}
   447→      </div>
   448→
   449→      {/* Summary Progress */}
   450→      {summary && summary.total > 0 && (
   451→        <Card className="mb-4">
   452→          <CardContent className="p-4">
   453→            <div className="flex items-center justify-between mb-2">
   454→              <div className="flex items-center gap-3">
   455→                <span className="text-sm font-medium">{summary.completionPercent}% complete</span>
   456→                <div className="flex items-center gap-2 text-xs text-muted-foreground">
   457→                  <span className="flex items-center gap-1">
   458→                    <span className="h-2 w-2 rounded-full bg-emerald-500" />
   459→                    {summary.done} done
   460→                  </span>
   461→                  <span className="flex items-center gap-1">
   462→                    <span className="h-2 w-2 rounded-full bg-orange-500" />
   463→                    {summary.pendingReview} review
   464→                  </span>
   465→                  <span className="flex items-center gap-1">
   466→                    <span className="h-2 w-2 rounded-full bg-blue-500" />
   467→                    {summary.inProgress} in progress
   468→                  </span>
   469→                  <span className="flex items-center gap-1">
   470→                    <span className="h-2 w-2 rounded-full bg-slate-400" />
   471→                    {summary.pending} pending
   472→                  </span>
   473→                </div>
   474→              </div>
   475→              <span className="text-xs text-muted-foreground">{summary.total} total</span>
   476→            </div>
   477→            <Progress value={summary.completionPercent} className="h-2" />
   478→          </CardContent>
   479→        </Card>
   480→      )}
   481→
   482→      {/* Filters */}
   483→      {summary && summary.total > 0 && (
   484→        <div className="flex flex-wrap items-center gap-2 mb-4">
   485→          <Select value={statusFilter} onValueChange={setStatusFilter}>
   486→            <SelectTrigger className="h-8 w-[140px] text-xs">
   487→              <SelectValue placeholder="Status" />
   488→            </SelectTrigger>
   489→            <SelectContent>
   490→              <SelectItem value="ALL">All Status</SelectItem>
   491→              <SelectItem value="PENDING">Pending</SelectItem>
   492→              <SelectItem value="IN_PROGRESS">In Progress</SelectItem>
   493→              <SelectItem value="PENDING_REVIEW">Pending Review</SelectItem>
   494→              <SelectItem value="COMPLETED">Completed</SelectItem>
   495→            </SelectContent>
   496→          </Select>
   497→
   498→          <Select value={assigneeFilter} onValueChange={setAssigneeFilter}>
   499→            <SelectTrigger className="h-8 w-[150px] text-xs">
   500→              <SelectValue placeholder="Assignee" />
   501→            </SelectTrigger>
   502→            <SelectContent>
   503→              <SelectItem value="ALL">All Members</SelectItem>
   504→              <SelectItem value="unassigned">Unassigned</SelectItem>
   505→              {team.map((m) => (
   506→                <SelectItem key={m.userId} value={m.userId}>
   507→                  {m.user.name}
   508→                </SelectItem>
   509→              ))}
   510→            </SelectContent>
   511→          </Select>
   512→
   513→          {(statusFilter !== "ALL" || assigneeFilter !== "ALL") && (
   514→            <Button
   515→              variant="ghost"
   516→              size="sm"
   517→              className="h-8 text-xs"
   518→              onClick={() => {
   519→                setStatusFilter("ALL");
   520→                setAssigneeFilter("ALL");
   521→              }}
   522→            >
   523→              <X className="mr-1 h-3 w-3" />
   524→              Clear filters
   525→            </Button>
   526→          )}
   527→        </div>
   528→      )}
   529→
   530→      {/* Loading */}
   531→      {loading && (
   532→        <Card>
   533→          <CardContent className="flex items-center justify-center py-10">
   534→            <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   535→          </CardContent>
   536→        </Card>
   537→      )}
   538→
   539→      {/* Empty State */}
   540→      {!loading && todos.length === 0 && (
   541→        <Card>
   542→          <CardContent className="flex flex-col items-center py-10">
   543→            <ListTodo className="h-10 w-10 text-muted-foreground mb-3" />
   544→            <p className="text-sm text-muted-foreground mb-1">No tasks yet</p>
   545→            <p className="text-xs text-muted-foreground mb-3">
   546→              {(statusFilter !== "ALL" || assigneeFilter !== "ALL")
   547→                ? "No tasks match the current filters"
   548→                : "Create your first task to start tracking progress"}
   549→            </p>
   550→            {isAdmin && (statusFilter === "ALL" && assigneeFilter === "ALL") && (
   551→              <Button size="sm" variant="outline" onClick={() => setAddOpen(true)}>
   552→                <Plus className="mr-2 h-4 w-4" />
   553→                Add First Task
   554→              </Button>
   555→            )}
   556→          </CardContent>
   557→        </Card>
   558→      )}
   559→
   560→      {/* Todo List */}
   561→      {!loading && todos.length > 0 && (
   562→        <div className="space-y-2">
   563→          {todos.map((todo) => {
   564→            const priority = priorityConfig[todo.priority] || priorityConfig.MEDIUM;
   565→            const status = statusConfig[todo.status] || statusConfig.PENDING;
   566→            const StatusIcon = status.icon;
