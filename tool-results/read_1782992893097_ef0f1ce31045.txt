     1→import type { Resend as ResendType } from "resend";
     2→import { decrypt } from "@/lib/encryption";
     3→import { getTursoClient } from "@/lib/api-auth";
     4→
     5→type EmailProvider = "resend" | "gmail-smtp";
     6→
     7→interface EmailConfig {
     8→  provider: EmailProvider;
     9→  // Resend
    10→  resendApiKey: string | null;
    11→  resendFromEmail: string;
    12→  // Gmail SMTP
    13→  smtpUser: string | null; // your@gmail.com
    14→  smtpPass: string | null; // app password
    15→  // Shared
    16→  fromName: string;
    17→}

    19→/**
    20→ * Get email configuration — reads from Settings DB first, falls back to env vars.
    21→ * Provider is determined by EMAIL_PROVIDER env var (or DB setting), defaults to "gmail-smtp".
    22→ */
    23→async function getEmailConfig(): Promise<EmailConfig> {
    24→  const defaultProvider: EmailProvider =
    25→    (process.env.EMAIL_PROVIDER as EmailProvider) || "gmail-smtp";
    26→
    27→  try {
    28→    const tursoUrl = process.env.TURSO_DATABASE_URL;
    29→    const tursoToken = process.env.TURSO_AUTH_TOKEN;
    30→
    31→    if (tursoUrl && tursoToken) {
    32→      const client = getTursoClient();
    33→
    34→      // Fetch all email-related settings from DB (now includes SMTP_USER)
    35→      const result = await client.execute({
    36→        sql: `SELECT key, value FROM "Settings" WHERE key IN (?, ?, ?, ?, ?, ?)`,
    37→        args: [
    38→          "EMAIL_PROVIDER",
    39→          "RESEND_API_KEY",
    40→          "RESEND_FROM_EMAIL",
    41→          "EMAIL_FROM_NAME",
    42→          "SMTP_USER",
    43→          "SMTP_PASSWORD",
    44→        ],
    45→      });
    46→
    47→      let provider = defaultProvider;
    48→      let resendApiKey: string | null = null;
    49→      let resendFromEmail = "";
    50→      let fromName = process.env.EMAIL_FROM_NAME || "KarmaBoard";
    51→      let smtpUser: string | null = null;
    52→      let smtpPass: string | null = null;
    53→
    54→      for (const row of result.rows) {
    55→        const key = row.key as string;
    56→        const rawValue = row.value as string;
    57→
    58→        switch (key) {
    59→          case "EMAIL_PROVIDER":
    60→            if (rawValue === "resend" || rawValue === "gmail-smtp") {
    61→              provider = rawValue;
    62→            }
    63→            break;
    64→          case "RESEND_API_KEY":
    65→            try {
    66→              resendApiKey = decrypt(rawValue);
    67→            } catch {
    68→              resendApiKey = null;
    69→            }
    70→            break;
    71→          case "RESEND_FROM_EMAIL":
    72→            resendFromEmail = rawValue;
    73→            break;
    74→          case "EMAIL_FROM_NAME":
    75→            fromName = rawValue || fromName;
    76→            break;
    77→          case "RESEND_FROM_NAME": // legacy key name compatibility
    78→            fromName = rawValue || fromName;
    79→            break;
    80→          case "SMTP_USER":
    81→            smtpUser = rawValue;
    82→            break;
    83→          case "SMTP_PASSWORD":
    84→            try {
    85→              smtpPass = decrypt(rawValue);
    86→            } catch {
    87→              smtpPass = null;
    88→            }
    89→            break;
    90→        }
    91→      }
    92→
    93→      // Fall back to env vars if DB values are empty
    94→      if (!resendApiKey && process.env.RESEND_API_KEY) {
    95→        resendApiKey = process.env.RESEND_API_KEY;
    96→      }
    97→      if (!resendFromEmail && process.env.RESEND_FROM_EMAIL) {
    98→        resendFromEmail = process.env.RESEND_FROM_EMAIL;
    99→      }
   100→      if (!smtpUser && process.env.SMTP_USER) {
   101→        smtpUser = process.env.SMTP_USER;
   102→      }
   103→      if (!smtpPass && process.env.SMTP_PASSWORD) {
   104→        smtpPass = process.env.SMTP_PASSWORD;
   105→      }
   106→
   107→      console.log(`[email] Config loaded from DB+env. Provider: ${provider}, SMTP user: ${smtpUser ? "SET" : "NOT SET"}, SMTP pass: ${smtpPass ? "SET" : "NOT SET"}`);
   108→
   109→      return {
   110→        provider,
   111→        resendApiKey,
   112→        resendFromEmail,
   113→        smtpUser,
   114→        smtpPass,
   115→        fromName,
   116→      };
   117→    }
   118→  } catch (error) {
   119→    console.error("[email] Error reading settings from DB:", error);
   120→  }
   121→
   122→  // Final fallback to env vars only
   123→  const smtpUser = process.env.SMTP_USER || null;
   124→  const smtpPass = process.env.SMTP_PASSWORD || null;
   125→
   126→  console.log(`[email] Config from env vars only. Provider: ${defaultProvider}, SMTP user: ${smtpUser ? "SET" : "NOT SET"}, SMTP pass: ${smtpPass ? "SET" : "NOT SET"}`);
   127→
   128→  return {
   129→    provider: defaultProvider,
   130→    resendApiKey: process.env.RESEND_API_KEY || null,
   131→    resendFromEmail: process.env.RESEND_FROM_EMAIL || "",
   132→    smtpUser,
   133→    smtpPass,
   134→    fromName: process.env.EMAIL_FROM_NAME || "KarmaBoard",
   135→  };
   136→}

   145→async function sendViaResend(
   146→  config: EmailConfig,
   147→  to: string,
   148→  subject: string,
   149→  html: string
   150→): Promise<{ success: boolean; error?: string }> {
   151→  if (!config.resendApiKey) {
   152→    return { success: false, error: "Resend API key not configured" };
   153→  }
   154→  if (!config.resendFromEmail) {
   155→    return { success: false, error: "Resend from email not configured" };
   156→  }
   157→
   158→  const resend = await getResendClient(config.resendApiKey);
   159→
   160→  const { data, error } = await resend.emails.send({
   161→    from: `${config.fromName} <${config.resendFromEmail}>`,
   162→    to: [to],
   163→    subject,
   164→    html,
   165→  });
   166→
   167→  if (error) {
   168→    return { success: false, error: error.message };
   169→  }
   170→
   171→  console.log(`[email] Resend email sent to ${to}, emailId: ${data?.id}`);
   172→  return { success: true };
   173→}

   175→// ===== Gmail SMTP Transport =====
   176→
   177→async function sendViaGmailSmtp(
   178→  config: EmailConfig,
   179→  to: string,
   180→  subject: string,
   181→  html: string
   182→): Promise<{ success: boolean; error?: string }> {
   183→  if (!config.smtpUser || !config.smtpPass) {
   184→    return {
   185→      success: false,
   186→      error: `Gmail SMTP not configured (SMTP_USER: ${config.smtpUser ? "SET" : "NOT SET"}, SMTP_PASSWORD: ${config.smtpPass ? "SET" : "NOT SET"})`,
   187→    };
   188→  }
   189→
   190→  const nodemailer = await import("nodemailer");
   191→
   192→  const transporter = nodemailer.createTransport({
   193→    service: "gmail",
   194→    auth: {
   195→      user: config.smtpUser,
   196→      pass: config.smtpPass,
   197→    },
   198→  });
   199→
   200→  try {
   201→    const info = await transporter.sendMail({
   202→      from: `"${config.fromName}" <${config.smtpUser}>`,
   203→      to,
   204→      subject,
   205→      html,
   206→    });
   207→
   208→    console.log(`[email] Gmail SMTP email sent to ${to}, messageId: ${info.messageId}`);
   209→    return { success: true };
   210→  } catch (error) {
   211→    const msg = error instanceof Error ? error.message : String(error);
   212→    console.error("[email] Gmail SMTP error:", msg);
   213→    return { success: false, error: msg };
   214→  }
   215→}
